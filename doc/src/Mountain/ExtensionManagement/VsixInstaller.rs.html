1//! # VSIX Installer
2//!
3//! Unpacks a `.vsix` file (ZIP with `extension/` as the payload prefix) into
4//! Land's user-extensions directory and produces an
5//! `ExtensionDescriptionStateDTO` ready for insertion into the application
6//! state's `ScannedExtensionCollection`.
7//!
8//! ## Flow
9//!
10//! 1. `InstallVsix(VsixPath, InstallRoot)`:
11//!    - Open the `.vsix` as a zip archive.
12//!    - Read `extension/package.json`, parse minimal fields (publisher, name,
13//!      version). These three determine the install directory.
14//!    - Compute target: `<InstallRoot>/<publisher>.<name>-<version>/`.
15//!    - If target already exists, refuse (caller decides whether to reinstall).
16//!    - Stream every entry whose path begins with `extension/` into the target,
17//!      stripping that prefix.
18//!    - Re-parse the extracted `package.json` as a full
19//!      `ExtensionDescriptionStateDTO`, stamp `ExtensionLocation`,
20//!      `Identifier`, and `IsBuiltin=false`.
21//! 2. `UninstallExtension(InstallDir)`:
22//!    - Recursively delete the install directory.
23//!
24//! The caller (`WindServiceHandlers::extensions:install`) is responsible for
25//! `ScannedExtensionCollection::AddOrUpdate` and for broadcasting the
26//! `extensions:installed` Tauri event so Wind re-fetches the extension list.
27//!
28//! ## Why the minimal two-pass read?
29//!
30//! The first pass reads only `extension/package.json` to compute the install
31//! path (we need publisher+name+version *before* writing any files, so we can
32//! reject collisions without partial writes). The second pass streams
33//! everything to disk. This keeps memory low - we never hold the full archive
34//! in RAM, and we don't unpack to a temp dir just to move it.
35//!
36//! ## Why no gallery API?
37//!
38//! `extensions:install` in `WindServiceHandlers.rs` previously responded to
39//! both `install` (gallery) and `install-vsix` (local file). This installer
40//! handles the local-file case - VS Code's gallery contract requires an
41//! online marketplace which Land does not currently host. Gallery support
42//! can layer on later by resolving a publisher identifier + version to a
43//! VSIX URL, downloading to a temp file, and calling `InstallVsix`.
44

58/// Everything an IPC handler needs after a successful install.
59#[derive(Debug)]
60pub struct InstallOutcome {
61	/// `<publisher>.<name>` - the canonical identifier string.
62	pub Identifier:String,
63	/// Semver string from the manifest.
64	pub Version:String,
65	/// Extracted target directory on disk.
66	pub InstalledAt:PathBuf,
67	/// Fully-populated DTO, ready to `AddOrUpdate` in ScannedExtensions.
68	pub Description:ExtensionDescriptionStateDTO,
69}
70
71/// Manifest facts we need before we start writing files.
72struct ManifestFacts {
73	Publisher:String,
74	Name:String,
75	Version:String,
76}
77
78/// Errors distinct enough that the IPC handler can produce useful messages
79/// without a `CommonError` cast. Flattened to String at the handler boundary.
80#[derive(Debug, thiserror::Error)]
81pub enum InstallError {
82	#[error("VSIX path '{0}' does not exist")]
83	SourceMissing(PathBuf),
84
85	#[error("VSIX archive read failure: {0}")]
86	ArchiveRead(String),
87
88	#[error("VSIX manifest missing or unreadable: {0}")]
89	ManifestMissing(String),
90
91	#[error("VSIX manifest missing required field '{0}'")]
92	ManifestFieldMissing(&'static str),
93
94	#[error("Extension '{Identifier}' version {Version} is already installed at {InstalledAt}")]
95	AlreadyInstalled { Identifier:String, Version:String, InstalledAt:PathBuf },
96
97	#[error("Filesystem error during install: {0}")]
98	FilesystemIO(String),
99}
100
101const MANIFEST_ENTRY:&str = "extension/package.json";
102const PAYLOAD_PREFIX:&str = "extension/";
103
104/// Open `VsixPath` and install its payload under `InstallRoot`. On success the
105/// caller receives the new identifier, install directory, and a DTO ready
106/// for `ScannedExtensionCollection::AddOrUpdate`.
107pub fn InstallVsix(VsixPath:&Path, InstallRoot:&Path) -> Result<InstallOutcome, InstallError> {
108	if !VsixPath.exists() {
109		return Err(InstallError::SourceMissing(VsixPath.to_path_buf()));
110	}
111
112	let Facts = ReadManifestFacts(VsixPath)?;
113	let InstalledAt = InstallRoot.join(format!("{}.{}-{}", Facts.Publisher, Facts.Name, Facts.Version));
114
115	if InstalledAt.exists() {
116		return Err(InstallError::AlreadyInstalled {
117			Identifier:format!("{}.{}", Facts.Publisher, Facts.Name),
118			Version:Facts.Version,
119			InstalledAt,
120		});
121	}
122
123	CreateParent(&InstalledAt)?;
124	ExtractPayload(VsixPath, &InstalledAt)?;
125
126	let Description = BuildDescription(&InstalledAt)?;
127	let Identifier = format!("{}.{}", Facts.Publisher, Facts.Name);
128
129	dev_log!(
130		"extensions",
131		"[VsixInstaller] Installed '{}' v{} at {}",
132		Identifier,
133		Facts.Version,
134		InstalledAt.display()
135	);
136
137	Ok(InstallOutcome { Identifier, Version:Facts.Version, InstalledAt, Description })
138}

140/// Delete the install directory. Returns `Ok` if the path was already absent.
141pub fn UninstallExtension(InstallDir:&Path) -> Result<(), InstallError> {
142	if !InstallDir.exists() {
143		dev_log!(
144			"extensions",
145			"[VsixInstaller] Uninstall skipped - {} already absent",
146			InstallDir.display()
147		);
148
149		return Ok(());
150	}
151
152	fs::remove_dir_all(InstallDir).map_err(|Error| InstallError::FilesystemIO(Error.to_string()))?;
153
154	dev_log!("extensions", "[VsixInstaller] Uninstalled {}", InstallDir.display());
155
156	Ok(())
157}

161fn ReadManifestFacts(VsixPath:&Path) -> Result<ManifestFacts, InstallError> {
162	let Manifest = ReadFullManifest(VsixPath)?;
163
164	let Publisher = ReadStringField(&Manifest, "publisher")?;
165	let Name = ReadStringField(&Manifest, "name")?;
166	let Version = ReadStringField(&Manifest, "version")?;
167
168	Ok(ManifestFacts { Publisher, Name, Version })
169}

171/// Read the full `extension/package.json` from a `.vsix` without extracting
172/// the archive to disk. Used by the IPC `extensions:getManifest` handler so
173/// the "Install from VSIX…" preview dialog and drag-and-drop flow can inspect
174/// a manifest before the user confirms installation.
175///
176/// The returned value is the raw parsed JSON (`serde_json::Value`) - callers
177/// can project it into VS Code's `IExtensionManifest` shape. No NLS bundle
178/// resolution is performed here (the renderer only needs publisher/name/
179/// version/displayName for the preview UI, and NLS keys would require
180/// unpacking `package.nls.json` from the archive too).
181pub fn ReadFullManifest(VsixPath:&Path) -> Result<Value, InstallError> {
182	let Archive = File::open(VsixPath).map_err(|Error| InstallError::ArchiveRead(Error.to_string()))?;
183	let mut Archive = ZipArchive::new(Archive).map_err(|Error| InstallError::ArchiveRead(Error.to_string()))?;
184
185	let mut Entry = Archive
186		.by_name(MANIFEST_ENTRY)
187		.map_err(|Error| InstallError::ManifestMissing(Error.to_string()))?;
188
189	let mut Raw = String::new();
190
191	Entry
192		.read_to_string(&mut Raw)
193		.map_err(|Error| InstallError::ManifestMissing(Error.to_string()))?;
194
195	serde_json::from_str(&Raw).map_err(|Error| InstallError::ManifestMissing(Error.to_string()))
196}

198fn ReadStringField(Manifest:&Value, Field:&'static str) -> Result<String, InstallError> {
199	Manifest
200		.get(Field)
201		.and_then(|Value| Value.as_str())
202		.filter(|Value| !Value.is_empty())
203		.map(str::to_owned)
204		.ok_or(InstallError::ManifestFieldMissing(Field))
205}

207fn CreateParent(InstalledAt:&Path) -> Result<(), InstallError> {
208	if let Some(Parent) = InstalledAt.parent() {
209		fs::create_dir_all(Parent).map_err(|Error| InstallError::FilesystemIO(Error.to_string()))?;
210	}
211
212	Ok(())
213}

215fn ExtractPayload(VsixPath:&Path, InstalledAt:&Path) -> Result<(), InstallError> {
216	let Archive = File::open(VsixPath).map_err(|Error| InstallError::ArchiveRead(Error.to_string()))?;
217	let mut Archive = ZipArchive::new(Archive).map_err(|Error| InstallError::ArchiveRead(Error.to_string()))?;
218
219	fs::create_dir_all(InstalledAt).map_err(|Error| InstallError::FilesystemIO(Error.to_string()))?;
220
221	for Index in 0..Archive.len() {
222		let mut Entry = Archive
223			.by_index(Index)
224			.map_err(|Error| InstallError::ArchiveRead(Error.to_string()))?;
225
226		let EntryName = Entry.name().to_string();
227
228		// Only the `extension/...` subtree is the addon payload. Manifest-level
229		// files (`[Content_Types].xml`, `extension.vsixmanifest`, `assets/`,
230		// etc.) are VSIX packaging metadata and are not needed at runtime.
231		let Stripped = match EntryName.strip_prefix(PAYLOAD_PREFIX) {
232			Some(Path) if !Path.is_empty() => Path,
233			_ => continue,
234		};
235
236		// Guard against zip-slip: the archive must not reference `..` segments
237		// that escape the install dir. Reject any entry whose resolved path is
238		// outside `InstalledAt`.
239		let Target = InstalledAt.join(Stripped);
240
241		let CanonicalInstall = InstalledAt.to_path_buf();
242
243		let RejectTraversal = !Target.starts_with(&CanonicalInstall);
244
245		if RejectTraversal {
246			return Err(InstallError::ArchiveRead(format!("zip-slip entry rejected: {}", EntryName)));
247		}
248
249		if Entry.is_dir() {
250			fs::create_dir_all(&Target).map_err(|Error| InstallError::FilesystemIO(Error.to_string()))?;
251
252			continue;
253		}
254
255		if let Some(Parent) = Target.parent() {
256			fs::create_dir_all(Parent).map_err(|Error| InstallError::FilesystemIO(Error.to_string()))?;
257		}
258
259		let mut Output = File::create(&Target).map_err(|Error| InstallError::FilesystemIO(Error.to_string()))?;
260
261		io::copy(&mut Entry, &mut Output).map_err(|Error| InstallError::FilesystemIO(Error.to_string()))?;
262
263		// Preserve Unix executable bits recorded in the VSIX. Extensions
264		// that ship platform-native binaries (openai.chatgpt's `codex`,
265		// language-server launchers, etc.) rely on the `0o755` mode being
266		// carried through the zip. Without this, the child `spawn()`
267		// inside the extension fails with `EACCES` because the freshly
268		// written file has only the default `0o644` read/write mode.
269		#[cfg(unix)]
270		{
271			use std::os::unix::fs::PermissionsExt;
272			if let Some(Mode) = Entry.unix_mode() {
273				// Zip's unix_mode returns the full stat mode (type bits +
274				// permission bits). Mask to the permission bits only, then
275				// OR in `0o644` as a safety floor so we never drop read
276				// access. Respect the executable bit if the archive had it.
277				let Permissions = fs::Permissions::from_mode((Mode & 0o777) | 0o644);
278				let _ = fs::set_permissions(&Target, Permissions);
279			}
280		}
281	}
282
283	Ok(())
284}

286fn BuildDescription(InstalledAt:&Path) -> Result<ExtensionDescriptionStateDTO, InstallError> {
287	let ManifestPath = InstalledAt.join("package.json");
288
289	let Raw = fs::read_to_string(&ManifestPath).map_err(|Error| InstallError::ManifestMissing(Error.to_string()))?;
290
291	let mut ManifestValue:Value =
292		serde_json::from_str(&Raw).map_err(|Error| InstallError::ManifestMissing(Error.to_string()))?;
293
294	let mut Description:ExtensionDescriptionStateDTO = serde_json::from_value(ManifestValue.clone())
295		.map_err(|Error| InstallError::ManifestMissing(Error.to_string()))?;
296
297	Description.ExtensionLocation = serde_json::to_value(
298		url::Url::from_directory_path(InstalledAt)
299			.unwrap_or_else(|_| url::Url::parse("file:///").expect("file:/// is a valid URL")),
300	)
301	.unwrap_or(Value::Null);
302
303	if Description.Identifier == Value::Null || Description.Identifier == Value::Object(Default::default()) {
304		let Identifier = if Description.Publisher.is_empty() {
305			Description.Name.clone()
306		} else {
307			format!("{}.{}", Description.Publisher, Description.Name)
308		};
309
310		Description.Identifier = serde_json::json!({ "value": Identifier });
311	}
312
313	Description.IsBuiltin = false;
314
315	// Touch the mutable manifest so later tooling that re-serialises it sees
316	// the same canonical form we parsed from.
317	let _ = &mut ManifestValue;
318
319	Ok(Description)

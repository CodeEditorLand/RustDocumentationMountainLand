1//! # DevLog - Tag-filtered development logging
2//!
3//! Controlled by `LAND_DEV_LOG` environment variable.
4//! The same tags work in both Mountain (Rust) and Wind/Sky (TypeScript).
5//!
6//! ## Usage
7//! ```bash
8//! LAND_DEV_LOG=vfs,ipc ./Mountain          # only VFS + IPC
9//! LAND_DEV_LOG=all ./Mountain              # everything
10//! LAND_DEV_LOG=short ./Mountain            # everything, compressed + deduped
11//! LAND_DEV_LOG=terminal,exthost ./Mountain # terminal + extension host
12//! ./Mountain                               # nothing (only normal log!() output)
13//! ```
14//!
15//! ## Short Mode
16//!
17//! `LAND_DEV_LOG=short` enables all tags but compresses output:
18//! - Long app-data paths aliased to `$APP`
19//! - Consecutive duplicate messages counted (`(x14)` suffix)
20//! - Rust log targets compressed (`D::Binary::Main::Entry` → `Entry`)
21//!
22//! ## File sink
23//!
24//! When `LAND_DEV_LOG_FILE=1` (or, in debug builds, `LAND_DEV_LOG` is set),
25//! every emitted line is mirrored to:
26//!
27//! ```
28//! ~/Library/Application Support/<bundle>/logs/<YYYYMMDDTHHMMSS>/Mountain.dev.log
29//! ```
30//!
31//! The timestamp directory follows Tauri's `tauri-plugin-log` format, so
32//! the dev log sits next to the plugin's own log file for the same boot
33//! (one directory per process start). Use `LAND_DEV_LOG_FILE=0` to force-
34//! disable even in debug. File writes are flushed per line so `tail -f`
35//! shows live output.
36//!
37//! ## Tags (38 granular tags across all Elements)

118//! ### Batch 4 diagnostic tags
119//!
120//! Added 2026-04-24 alongside the `workspace.fs` tier-split refactor.
121//! Cocoon's `WorkspaceNamespace/FileSystemRoute.ts` now chooses between
122//! Tier A (`node:fs/promises` in-process) and Tier C (Mountain `FileSystem.*`
123//! gRPC) per URI scheme + custom-provider claim. The tag surfaces every
124//! decision so empirical workload profiling confirms the split is paying
125//! off - `grep 'route=native'` / `grep 'route=mountain'` buckets per run.
126//! Emitted from Cocoon stdout, picked up by Mountain's `[DEV:COCOON]`
127//! stdout tail with the standard `[DEV:FS-ROUTE]` prefix.
128//!
129//! | Tag                | Scope                                                                   |
130//! |--------------------|-------------------------------------------------------------------------|
131//! | `fs-route`         | `workspace.fs.*` + `openTextDocument` native-vs-mountain routing        |
132//! | `cmd-route`        | `commands.executeCommand` local-vs-mountain routing                     |
133//! | `dual-track`       | Mountain-first / Node-fallback progressive Rust migration dispatches    |
134//!
135//! ### `dual-track` tag usage pattern
136//!
137//! Every Cocoon shim that wraps `TryMountainThenNode` logs one of four
138//! decisions per dispatch:
139//!
140//!   - `route=mountain`      Mountain handled it (Rust code path served the call)
141//!   - `route=node-fallback` Build-time manifest says no Mountain handler (or
142//!                           runtime confirmed "Unknown method: X"); Cocoon's
143//!                           Node / stock-VS-Code implementation served it
144//!   - `route=unavailable`   Tier-4: no tier covers the method. Extension
145//!                           receives typed `NotImplementedError`; build
146//!                           surfaces this as a feature-gap
147//!   - `route=error`         Tier failed unexpectedly; error propagated
148//!
149//! Grep the stream to see which `vscode.*` methods Mountain doesn't yet
150//! cover - that set IS the Rust migration TODO list. A method moves from
151//! `node-fallback` to `mountain` automatically the moment Mountain lands
152//! its Rust handler AND the manifest is regenerated. `route=unavailable`
153//! rows are the apologise-to-user list.
154//!
155//! A single `[DEV:DUAL-TRACK] manifest mountain=N stockLift=M bespoke=K`
156//! line prints at Cocoon boot with the build-time tier coverage
157//! generated by `Maintain/Script/GenerateRouteManifest.sh`.

174// ── File sink ────────────────────────────────────────────────────────────
175//
176// Mirrors every `dev_log!` line to a timestamped file under the app's
177// `logs/<YYYYMMDDTHHMMSS>/` directory so long sessions can be inspected
178// post-mortem without scrolling terminal history. Enable with:
179//
180//     LAND_DEV_LOG_FILE=1     # explicit opt-in
181//     LAND_DEV_LOG_FILE=0     # explicit opt-out (wins over defaults)
182//
183// When unset, file logging is auto-enabled in debug builds iff
184// `LAND_DEV_LOG` itself is set to at least one tag. Release builds stay
185// silent unless the opt-in flag is present, to avoid surprise writes in
186// shipped binaries.
187//
188// Directory layout matches Tauri's tauri-plugin-log output (same parent
189// `logs/` and same `YYYYMMDDTHHMMSS` subdir format) so the two logs sit
190// side by side when the user greps one timestamp.
191
192static LOG_FILE:OnceLock<Mutex<Option<BufWriter<File>>>> = OnceLock::new();
193
194/// Decide whether the file sink should be active. Returns the final flag
195/// once per process; subsequent calls are cached.
196fn FileSinkEnabled() -> bool {
197	static ENABLED:OnceLock<bool> = OnceLock::new();
198	*ENABLED.get_or_init(|| {
199		match std::env::var("LAND_DEV_LOG_FILE") {
200			Ok(Value) => matches!(Value.as_str(), "1" | "true" | "yes" | "on"),
201			Err(_) => {
202				// Auto-enable when LAND_DEV_LOG is set in a debug build.
203				cfg!(debug_assertions) && std::env::var("LAND_DEV_LOG").is_ok()
204			},
205		}
206	})
207}

209/// Resolve the log directory for this session:
210///   ~/Library/Application Support/<bundle>/logs/<YYYYMMDDTHHMMSS>/
211///
212/// The timestamp is picked once per process at first call. If the app-data
213/// prefix can't be detected (Tauri hasn't spawned yet, say), fall back to
214/// the system temp directory with the same naming - dev_log still works
215/// and the file ends up somewhere predictable.
216fn ResolveLogDirectory() -> PathBuf {
217	let Stamp = FormatTimestamp();
218	let Base = match AppDataPrefix() {
219		Some(Prefix) => PathBuf::from(Prefix).join("logs"),
220		None => std::env::temp_dir().join("land-editor-logs"),
221	};
222	Base.join(Stamp)
223}

225/// Session timestamp in local time, cached once per process. MUST match
226/// whatever `WindServiceHandlers.rs::"nativeHost:getEnvironmentPaths"`
227/// builds, because VS Code's file service writes `window1/output/*.log`
228/// into the directory that handler returns - if DevLog and VS Code use
229/// different timezones, `Mountain.dev.log` and the `window1/` subtree
230/// land in two sibling directories 2-3 hours apart, which makes every
231/// post-mortem investigation start with "which folder has the real
232/// log?". Picking `chrono::Local::now()` matches the VS Code convention
233/// (Tauri's tauri-plugin-log also writes local-time `YYYYMMDDTHHMMSS`).
234///
235/// The format string is deliberately identical to the handler's
236/// `"%Y%m%dT%H%M%S"`, and both sides pull from the same OnceLock via
237/// `SessionTimestamp()` so re-entrant calls from anywhere in the
238/// codebase produce the same string.
239pub fn SessionTimestamp() -> String {
240	static STAMP:OnceLock<String> = OnceLock::new();
241	STAMP
242		.get_or_init(|| chrono::Local::now().format("%Y%m%dT%H%M%S").to_string())
243		.clone()
244}
245
246fn FormatTimestamp() -> String { SessionTimestamp() }

248// `DaysToYMD` + `IsLeap` were previously used to build a UTC timestamp
249// string without pulling chrono into DevLog. Replaced by
250// `chrono::Local::now()` in `SessionTimestamp()` so this file agrees
251// with `WindServiceHandlers.rs::"nativeHost:getEnvironmentPaths"` on
252// the session-log directory. chrono is already a Mountain dependency,
253// so the vendored date math is dead weight now.

255/// Initialise the file sink on first call. Silently falls through to a
256/// disabled sink if the directory or file can't be created - the caller
257/// must never panic because of a log-file failure.
258fn InitFileSink() -> &'static Mutex<Option<BufWriter<File>>> {
259	LOG_FILE.get_or_init(|| {
260		if !FileSinkEnabled() {
261			return Mutex::new(None);
262		}
263		let Dir = ResolveLogDirectory();
264		if create_dir_all(&Dir).is_err() {
265			eprintln!("[DEV:LOG] Failed to create log directory {}", Dir.display());
266			return Mutex::new(None);
267		}
268		let Path = Dir.join("Mountain.dev.log");
269		match OpenOptions::new().create(true).append(true).open(&Path) {
270			Ok(File) => {
271				let mut Writer = BufWriter::with_capacity(64 * 1024, File);
272				// Header pins the boot-time context so every session's file
273				// is self-describing even without the surrounding terminal.
274				let Header = format!(
275					"# Land dev log - started {}, pid {}, tags={:?}, short={}\n",
276					FormatTimestamp(),
277					std::process::id(),
278					EnabledTags(),
279					IsShort(),
280				);
281				let _ = Writer.write_all(Header.as_bytes());
282				let _ = Writer.flush();
283				eprintln!("[DEV:LOG] File sink → {}", Path.display());
284				Mutex::new(Some(Writer))
285			},
286			Err(Error) => {
287				eprintln!("[DEV:LOG] Failed to open {}: {}", Path.display(), Error);
288				Mutex::new(None)
289			},
290		}
291	})
292}

294/// Force the file sink to initialize before any `dev_log!` has run.
295///
296/// `WriteToFile` is otherwise lazy - the log file is only opened the first
297/// time a tagged `dev_log!` call fires. When Mountain panics (or the
298/// webview traps the user with an early error) before the first enabled
299/// tag emits, the session log directory ends up with an empty shell and
300/// the post-mortem evidence is lost.
301///
302/// Call this once at the top of `Binary::Main::Fn()` - as early as the
303/// binary can reach - so the header line + `LAND_DEV_LOG_FILE=1` opt-in
304/// are honoured even when nothing else ever logs. Harmless to call
305/// multiple times; the `OnceLock` inside `InitFileSink` gates it.
306pub fn InitEager() { let _ = InitFileSink(); }
307
308/// Append a single formatted line to the session's log file if the file
309/// sink is active. Swallows every error - dev_log must never crash.
310pub fn WriteToFile(Line:&str) {
311	let Sink = InitFileSink();
312	if let Ok(mut Guard) = Sink.lock() {
313		if let Some(Writer) = Guard.as_mut() {
314			let _ = Writer.write_all(Line.as_bytes());
315			if !Line.ends_with('\n') {
316				let _ = Writer.write_all(b"\n");
317			}
318			// Flush on every line - ordering/tail-f matters more than throughput
319			// for dev logs, and the BufWriter coalesces partial writes anyway.
320			let _ = Writer.flush();
321		}
322	}
323}

329/// Produce an identity signature for THIS running binary derived from
330/// `CARGO_PKG_NAME` (which Maintain sets to the long PascalCase product
331/// name before `cargo build`). Each profile produces a distinct signature
332/// - `_Debug_Mountain` → `.debug.mountain`, `_Compile_Mountain` →
333/// `.compile.mountain`, `_Bundle_Clean_Debug_ElectronProfile_Mountain`
334/// → `.debug.electron.profile.mountain` - so a candidate directory in
335/// `~/Library/Application Support/` can be disambiguated against every
336/// other `land.editor.*.mountain` leftover from prior runs.
337///
338/// Only the last three underscore-delimited segments are used: the
339/// leading `DevelopmentNodeEnvironment_MicrosoftVSCodeDependency_
340/// 22NodeVersion_Bundle_Clean` prefix is identical across profiles and
341/// doesn't help disambiguate, while the tail (`Debug_Mountain` vs
342/// `Compile_Mountain` vs `Debug_ElectronProfile_Mountain`) is where the
343/// per-profile identity lives.
344fn BinarySignature() -> String {
345	let PackageName = env!("CARGO_PKG_NAME");
346	let Segments:Vec<&str> = PackageName.split('_').collect();
347	let Take = Segments.len().min(4);
348	let Start = Segments.len().saturating_sub(Take);
349	// Each underscore-delimited segment is PascalCase (e.g. `ElectronProfile`,
350	// `MountainProfile`, `RestCompiler`). The Tauri identifier Maintain
351	// generates splits every PascalCase word on its capital boundary, so
352	// `ElectronProfile` → `electron.profile`, not `electronprofile`. Without
353	// the per-segment split here the signature becomes `electronprofile`
354	// and `ends_with` never matches the real Tauri app-data dir
355	// `…clean.debug.electron.profile.mountain`, so DevLog silently falls
356	// back to whichever other `*.mountain` directory `read_dir` yielded
357	// first - the bug that sent the electron-profile binary's logs into
358	// the compile-profile directory.
359	let Dotted:String = Segments[Start..]
360		.iter()
361		.flat_map(|Segment| SplitPascalCaseIntoWords(Segment))
362		.collect::<Vec<String>>()
363		.join(".")
364		.to_ascii_lowercase();
365	Dotted
366}

368/// Split a PascalCase / UPPERCASE string into lowercase component words,
369/// matching the tokenisation Maintain's `Build/Process.rs` applies when it
370/// stamps the Tauri `identifier`. Example: `ElectronProfile` →
371/// `["electron", "profile"]`; `22NodeVersion` → `["22", "node", "version"]`.
372/// Empty segments are filtered out.
373fn SplitPascalCaseIntoWords(Segment:&str) -> Vec<String> {
374	let mut Words:Vec<String> = Vec::new();
375	let mut Current = String::new();
376	let mut PrevWasUpper = false;
377	let mut PrevWasDigit = false;
378	for Ch in Segment.chars() {
379		let IsUpper = Ch.is_ascii_uppercase();
380		let IsDigit = Ch.is_ascii_digit();
381		let NeedBreak = !Current.is_empty()
382			&& ((IsUpper && !PrevWasUpper) || (IsDigit != PrevWasDigit && !Current.is_empty()));
383		if NeedBreak {
384			Words.push(std::mem::take(&mut Current));
385		}
386		Current.push(Ch);
387		PrevWasUpper = IsUpper;
388		PrevWasDigit = IsDigit;
389	}
390	if !Current.is_empty() {
391		Words.push(Current);
392	}
393	Words.into_iter().filter(|Word| !Word.is_empty()).collect()
394}

396fn DetectAppDataPrefix() -> Option<String> {
397	let Home = std::env::var("HOME").ok()?;
398	let Base = format!("{}/Library/Application Support", Home);
399	let Signature = BinarySignature();
400
401	// Prefer a directory whose name ends with this binary's unique tail
402	// signature: that's the app-data directory Tauri created for THIS
403	// profile. Without this check, a user who has ever launched another
404	// profile (debug-electron, release-electron, …) will see DevLog
405	// writing into that stale directory while userdata still goes into
406	// the current one, producing an "empty logs folder" mystery.
407	let mut FirstMatchingMountain:Option<String> = None;
408	if let Ok(Entries) = std::fs::read_dir(&Base) {
409		for Entry in Entries.flatten() {
410			let Name = Entry.file_name();
411			let Name = Name.to_string_lossy().into_owned();
412			if !Name.starts_with("land.editor.") || !Name.contains("mountain") {
413				continue;
414			}
415			// Strict match: binary signature tail is a suffix of the dir name.
416			if Name.ends_with(&Signature) {
417				return Some(format!("{}/{}", Base, Name));
418			}
419			// Lossy match: some segment of the binary signature is contained
420			// in the dir name. Used only if no strict match exists.
421			if FirstMatchingMountain.is_none() {
422				FirstMatchingMountain = Some(format!("{}/{}", Base, Name));
423			}
424		}
425	}
426	FirstMatchingMountain
427}

429/// Get the app-data path prefix for aliasing (cached).
430pub fn AppDataPrefix() -> &'static Option<String> { APP_DATA_PREFIX.get_or_init(DetectAppDataPrefix) }

432/// Replace the long app-data path with `$APP` in a string.
433pub fn AliasPath(Input:&str) -> String {
434	if let Some(Prefix) = AppDataPrefix() {
435		Input.replace(Prefix.as_str(), "$APP")
436	} else {
437		Input.to_string()
438	}

443pub struct DedupState {
444	pub LastKey:String,
445	pub Count:u64,
446}
447
448pub static DEDUP:Mutex<DedupState> = Mutex::new(DedupState { LastKey:String::new(), Count:0 });
449
450/// Flush the dedup buffer - prints the pending count if > 1.
451pub fn FlushDedup() {
452	if let Ok(mut State) = DEDUP.lock() {
453		if State.Count > 1 {
454			let Tail = format!("  (x{})", State.Count);
455			eprintln!("{}", Tail);
456			WriteToFile(&Tail);
457		}
458		State.LastKey.clear();
459		State.Count = 0;
460	}
461}

538/// Emit the line exactly once per process, keyed on the supplied key. Later
539/// calls with the same key are silently dropped. The file sink still
540/// captures the very first occurrence so the probe is documented.
541pub fn DebugOnce(Tag:&str, Key:&str, Line:&str) {
542	if let Ok(mut Keys) = DebugOnceKeys().lock() {
543		if !Keys.insert(Key.to_string()) {
544			return;
545		}
546	}
547	if IsEnabled(Tag) || IsEnabled("all") {
548		let Formatted = format!("[DEV:{}] {}", Tag.to_uppercase(), Line);
549		eprintln!("{}", Formatted);
550		WriteToFile(&Formatted);
551	} else {
552		// Never echo to the console, but preserve in the file sink so
553		// post-mortems still see which probe paths fired.
554		let Formatted = format!("[DEV:{}/once] {}", Tag.to_uppercase(), Line);
555		WriteToFile(&Formatted);
556	}
557}

573/// Check if a tag is enabled.
574pub fn IsEnabled(Tag:&str) -> bool {
575	let Tags = EnabledTags();
576	if Tags.is_empty() {
577		return false;
578	}
579	let Lower = Tag.to_lowercase();
580	Tags.iter().any(|T| T == "all" || T == "short" || T == Lower.as_str())
581}

583/// Log a tagged dev message. Only prints if the tag is enabled via
584/// LAND_DEV_LOG.
585///
586/// In `short` mode: aliases long paths, deduplicates consecutive identical
587/// lines.
588#[macro_export]
589macro_rules! dev_log {
590	($Tag:expr, $($Arg:tt)*) => {
591		if $crate::IPC::DevLog::IsEnabled($Tag) {
592			let RawMessage = format!($($Arg)*);
593			let TagUpper = $Tag.to_uppercase();
594			if $crate::IPC::DevLog::IsShort() {
595				let Aliased = $crate::IPC::DevLog::AliasPath(&RawMessage);
596				let Key = format!("{}:{}", TagUpper, Aliased);
597				let ShouldPrint = {
598					if let Ok(mut State) = $crate::IPC::DevLog::DEDUP.lock() {
599						if State.LastKey == Key {
600							State.Count += 1;
601							false
602						} else {
603							let PrevCount = State.Count;
604							let HadPrev = !State.LastKey.is_empty();
605							State.LastKey = Key;
606							State.Count = 1;
607							if HadPrev && PrevCount > 1 {
608								let Tail = format!("  (x{})", PrevCount);
609								eprintln!("{}", Tail);
610								$crate::IPC::DevLog::WriteToFile(&Tail);
611							}
612							true
613						}
614					} else {
615						true
616					}
617				};
618				if ShouldPrint {
619					let Formatted = format!("[DEV:{}] {}", TagUpper, Aliased);
620					eprintln!("{}", Formatted);
621					$crate::IPC::DevLog::WriteToFile(&Formatted);
622				}
623			} else {
624				let Formatted = format!("[DEV:{}] {}", TagUpper, RawMessage);
625				eprintln!("{}", Formatted);
626				$crate::IPC::DevLog::WriteToFile(&Formatted);
627			}
628		}
629	};
630}

636static OTLP_AVAILABLE:AtomicBool = AtomicBool::new(true);
637static OTLP_TRACE_ID:OnceLock<String> = OnceLock::new();
638
639fn GetTraceId() -> &'static str {
640	OTLP_TRACE_ID.get_or_init(|| {
641		use std::{
642			collections::hash_map::DefaultHasher,
643			hash::{Hash, Hasher},
644		};
645		let mut H = DefaultHasher::new();
646		std::process::id().hash(&mut H);
647		SystemTime::now()
648			.duration_since(UNIX_EPOCH)
649			.unwrap_or_default()
650			.as_nanos()
651			.hash(&mut H);
652		format!("{:032x}", H.finish() as u128)
653	})
654}
655
656pub fn NowNano() -> u64 { SystemTime::now().duration_since(UNIX_EPOCH).unwrap_or_default().as_nanos() as u64 }
657
658/// Emit an OTLP span to the local collector (Jaeger at 127.0.0.1:4318).
659/// Fire-and-forget on a background thread. Stops trying after first failure.
660pub fn EmitOTLPSpan(Name:&str, StartNano:u64, EndNano:u64, Attributes:&[(&str, &str)]) {
661	if !cfg!(debug_assertions) {
662		return;
663	}
664	if !OTLP_AVAILABLE.load(Ordering::Relaxed) {
665		return;
666	}
667
668	let SpanId = format!("{:016x}", rand_u64());
669	let TraceId = GetTraceId().to_string();
670	let SpanName = Name.to_string();
671
672	let AttributesJson:Vec<String> = Attributes
673		.iter()
674		.map(|(K, V)| {
675			format!(
676				r#"{{"key":"{}","value":{{"stringValue":"{}"}}}}"#,
677				K,
678				V.replace('\\', "\\\\").replace('"', "\\\"")
679			)
680		})
681		.collect();
682
683	let IsError = SpanName.contains("error");
684
685	let StatusCode = if IsError { 2 } else { 1 };
686	let Payload = format!(
687		concat!(
688			r#"{{"resourceSpans":[{{"resource":{{"attributes":["#,
689			r#"{{"key":"service.name","value":{{"stringValue":"land-editor-mountain"}}}},"#,
690			r#"{{"key":"service.version","value":{{"stringValue":"0.0.1"}}}}"#,
691			r#"]}},"scopeSpans":[{{"scope":{{"name":"mountain.ipc","version":"1.0.0"}},"#,
692			r#""spans":[{{"traceId":"{}","spanId":"{}","name":"{}","kind":1,"#,
693			r#""startTimeUnixNano":"{}","endTimeUnixNano":"{}","#,
694			r#""attributes":[{}],"status":{{"code":{}}}}}]}}]}}]}}"#,
695		),
696		TraceId,
697		SpanId,
698		SpanName,
699		StartNano,
700		EndNano,
701		AttributesJson.join(","),
702		StatusCode,
703	);
704
705	// Fire-and-forget on a background thread
706	std::thread::spawn(move || {
707		use std::{
708			io::{Read as IoRead, Write as IoWrite},
709			net::TcpStream,
710			time::Duration,
711		};
712
713		let Ok(mut Stream) = TcpStream::connect_timeout(&"127.0.0.1:4318".parse().unwrap(), Duration::from_millis(200))
714		else {
715			OTLP_AVAILABLE.store(false, Ordering::Relaxed);
716			return;
717		};
718		let _ = Stream.set_write_timeout(Some(Duration::from_millis(200)));
719		let _ = Stream.set_read_timeout(Some(Duration::from_millis(200)));
720
721		let HttpReq = format!(
722			"POST /v1/traces HTTP/1.1\r\nHost: 127.0.0.1:4318\r\nContent-Type: application/json\r\nContent-Length: \
723			 {}\r\nConnection: close\r\n\r\n",
724			Payload.len()
725		);
726		if Stream.write_all(HttpReq.as_bytes()).is_err() {
727			return;
728		}
729		if Stream.write_all(Payload.as_bytes()).is_err() {
730			return;
731		}
732		let mut Buf = [0u8; 32];
733		let _ = Stream.read(&mut Buf);
734		if !(Buf.starts_with(b"HTTP/1.1 2") || Buf.starts_with(b"HTTP/1.0 2")) {
735			OTLP_AVAILABLE.store(false, Ordering::Relaxed);
736		}
737	});
738}

740fn rand_u64() -> u64 {
741	use std::{
742		collections::hash_map::DefaultHasher,
743		hash::{Hash, Hasher},
744	};
745	let mut H = DefaultHasher::new();
746	std::thread::current().id().hash(&mut H);
747	NowNano().hash(&mut H);
748	H.finish()
749}

752/// Usage: `otel_span!("file:readFile", StartNano, &[("path", &SomePath)]);`
753#[macro_export]
754macro_rules! otel_span {

760	};

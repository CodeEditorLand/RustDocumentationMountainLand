3//! Converts VS Code `Uri`-shaped arguments to platform-native paths.
4//! Co-locates percent-decoding, userdata remapping, and `/Static/Application`
5//! rewriting because each is a private helper of `extract_path_from_arg`.
6//! Percent-decoding is also re-exported for callers outside the VFS path
7//! (configuration loaders, etc.).

17/// Extract a filesystem path from a VS Code argument.
18/// VS Code sends URI objects `{ scheme: "file", path: "/C:/foo", fsPath:
19/// "C:\\foo" }` but Mountain handlers expect platform-native path strings.
20///
21/// Windows URI paths have a leading slash: `/C:/Users/...` → strip it.
22/// Unix paths start with `/` normally.
23pub fn extract_path_from_arg(Arg:&Value) -> Result<String, String> {
24	if let Some(Path) = Arg.as_str() {
25		return Ok(normalize_uri_path(Path));
26	}
27	if let Some(Object) = Arg.as_object() {
28		if let Some(FsPath) = Object.get("fsPath").and_then(|V| V.as_str()) {
29			if !FsPath.is_empty() {
30				return Ok(FsPath.to_string());
31			}
32		}
33		if let Some(Path) = Object.get("path").and_then(|V| V.as_str()) {
34			if !Path.is_empty() {
35				return Ok(normalize_uri_path(Path));
36			}
37		}
38		if let Some(External) = Object.get("external").and_then(|V| V.as_str()) {
39			if External.starts_with("file://") {
40				let Stripped = External.trim_start_matches("file://");
41				return Ok(normalize_uri_path(Stripped));
42			}
43		}
44	}
45	Err("File path must be a string or URI object with path/fsPath field".to_string())
46}

48fn normalize_uri_path(Path:&str) -> String {
49	let Decoded = percent_decode(Path);
50	let Resolved = resolve_userdata_path(&Decoded);
51	let Resolved = resolve_static_application_path(&Resolved);
52
53	#[cfg(target_os = "windows")]
54	{
55		let Trimmed = if Resolved.len() >= 3 && Resolved.starts_with('/') && Resolved.as_bytes().get(2) == Some(&b':') {
56			Resolved[1..].to_string()
57		} else {
58			Resolved
59		};
60		Trimmed.replace('/', "\\")
61	}
62
63	#[cfg(not(target_os = "windows"))]
64	{
65		Resolved
66	}
67}

69fn resolve_userdata_path(Path:&str) -> String {
70	if !Path.starts_with("/User/") && Path != "/User" {
71		return Path.to_string();
72	}
73
74	let UserDataBase = get_userdata_base_dir();
75	let Resolved = format!("{}{}", UserDataBase, Path);
76	dev_log!("vfs", "resolve_userdata: {} -> {}", Path, Resolved);
77	Resolved
78}

80/// Map paths starting with /Static/Application/ to the real Sky Target
81/// directory. Also accepts the leading-slash-less form - the WASM loader
82/// (`vscode-oniguruma` → `onig.wasm`) resolves asset URLs relative to the
83/// current document, which strips the leading slash before the path
84/// reaches `file:read`. Without this branch, `tokio::fs::read` would be
85/// called with a relative path and fail with ENOENT, breaking TextMate
86/// syntax highlighting.
87fn resolve_static_application_path(Path:&str) -> String {
88	let Normalized = if Path.starts_with("/Static/Application/") || Path == "/Static/Application" {
89		Path.to_string()
90	} else if Path.starts_with("Static/Application/") || Path == "Static/Application" {
91		format!("/{}", Path)
92	} else {
93		return Path.to_string();
94	};
95
96	if let Some(Root) = get_static_application_root() {
97		let Relative = Normalized.strip_prefix("/Static/Application").unwrap_or("");
98		let Resolved = format!("{}/Static/Application{}", Root, Relative);
99		dev_log!("vfs", "resolve_static: {} -> {}", Path, Resolved);
100		Resolved
101	} else {
102		Path.to_string()
103	}
104}

106/// Decode percent-encoded characters in URI paths.
107/// Handles: %20 (space), %23 (#), %25 (%), %5B ([), %5D (]), etc.
108pub fn percent_decode(Input:&str) -> String {
109	let mut Result = String::with_capacity(Input.len());
110	let Bytes = Input.as_bytes();
111	let mut I = 0;
112
113	while I < Bytes.len() {
114		if Bytes[I] == b'%' && I + 2 < Bytes.len() {
115			let High = hex_digit(Bytes[I + 1]);
116			let Low = hex_digit(Bytes[I + 2]);
117			if let (Some(H), Some(L)) = (High, Low) {
118				Result.push((H * 16 + L) as char);
119				I += 3;
120				continue;
121			}
122		}
123		Result.push(Bytes[I] as char);
124		I += 1;
125	}
126
127	Result
128}

130pub fn hex_digit(Byte:u8) -> Option<u8> {
131	match Byte {
132		b'0'..=b'9' => Some(Byte - b'0'),
133		b'a'..=b'f' => Some(Byte - b'a' + 10),
134		b'A'..=b'F' => Some(Byte - b'A' + 10),
135		_ => None,
136	}
137}

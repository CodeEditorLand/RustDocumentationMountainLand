3//! `file:open` - open a file and return an integer file descriptor.
4//!
5//! VS Code's `DiskFileSystemProvider.open(resource, opts)` uses fd-based
6//! access for large binary files and write operations. The fd table is a
7//! process-global `HashMap<u32, File>` guarded by a `Mutex`. FDs start
8//! from 1 and increment with each successful open.
9//!
10//! Arguments[0] = resource URI or path string
11//! Arguments[1] = options: `{ create?: boolean, unlock?: boolean }`
12//!
13//! Returns: integer fd number, or 0 on error (VS Code ignores the error
14//! for read-only opens and falls back to the full-read path).
15
16use std::{
17	collections::HashMap,
18	sync::{
19		Mutex,
20		OnceLock,
21		atomic::{AtomicU32, Ordering},
22	},
23};
24
25use serde_json::{Value, json};
26use tokio::fs::File;
27
28use crate::{IPC::WindServiceHandlers::Utilities::PathExtraction::extract_path_from_arg, dev_log};
29
30static NEXT_FD:AtomicU32 = AtomicU32::new(1);
31
32pub struct FdTable {
33	pub Files:Mutex<HashMap<u32, File>>,
34}
35
36static FD_TABLE:OnceLock<FdTable> = OnceLock::new();
37
38pub fn GetFdTable() -> &'static FdTable { FD_TABLE.get_or_init(|| FdTable { Files:Mutex::new(HashMap::new()) }) }
39
40pub async fn FileOpenFd(Arguments:Vec<Value>) -> Result<Value, String> {
41	let ResourceArg = Arguments.first().ok_or("file:open: missing resource")?;
42
43	let Path = extract_path_from_arg(ResourceArg)?;
44
45	let Opts = Arguments.get(1).and_then(Value::as_object);
46
47	let Create = Opts.and_then(|O| O.get("create")).and_then(Value::as_bool).unwrap_or(false);
48
49	let Truncate = Opts.and_then(|O| O.get("truncate")).and_then(Value::as_bool).unwrap_or(false);
50
51	let F = if Create {
52		let mut OpenOpts = tokio::fs::OpenOptions::new();
53
54		OpenOpts.write(true).create(true);
55
56		if Truncate {
57			OpenOpts.truncate(true);
58		}
59
60		OpenOpts
61			.open(&Path)
62			.await
63			.map_err(|E| format!("file:open create '{}': {}", Path, E))?
64	} else {
65		tokio::fs::File::open(&Path)
66			.await
67			.map_err(|E| format!("file:open '{}': {}", Path, E))?
68	};
69
70	let Fd = NEXT_FD.fetch_add(1, Ordering::Relaxed);
71
72	if let Ok(mut Table) = GetFdTable().Files.lock() {
73		Table.insert(Fd, F);
74	}
75
76	dev_log!("vfs", "file:open fd={} path={} create={}", Fd, Path, Create);
77
78	Ok(json!(Fd))
79}

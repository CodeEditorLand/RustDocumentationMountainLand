11/// Search text across all workspace files (line-by-line grep, max 1000 results).
12pub async fn handle_search_find_in_files(
13	runtime:Arc<ApplicationRunTime>,
14	args:Vec<Value>,
15) -> Result<Value, String> {
16	use globset::GlobBuilder;
17	use tokio::fs;
18
19	let Pattern = args
20		.first()
21		.and_then(|V| V.as_str())
22		.ok_or("search:findInFiles requires pattern".to_string())?
23		.to_string();
24	let IsRegex = args.get(1).and_then(|V| V.as_bool()).unwrap_or(false);
25	let IsCaseSensitive = args.get(2).and_then(|V| V.as_bool()).unwrap_or(false);
26	let _IsWordMatch = args.get(3).and_then(|V| V.as_bool()).unwrap_or(false);
27	let IncludeGlob = args.get(4).and_then(|V| V.as_str()).unwrap_or("**").to_string();
28	let ExcludeGlob = args.get(5).and_then(|V| V.as_str()).unwrap_or("").to_string();
29	let MaxResults = args.get(6).and_then(|V| V.as_u64()).unwrap_or(1000) as usize;
30
31	let WorkspaceFolders = runtime.Environment.ApplicationState.Workspace.GetWorkspaceFolders();
32
33	if WorkspaceFolders.is_empty() {
34		return Ok(json!([]));
35	}
36
37	let RootPath = PathBuf::from(&WorkspaceFolders[0].URI.to_string().replace("file://", ""));
38
39	// Build include matcher
40	let IncludeMatcher = GlobBuilder::new(&IncludeGlob)
41		.literal_separator(false)
42		.build()
43		.map(|G| G.compile_matcher())
44		.ok();
45
46	// Build exclude matcher
47	let ExcludeMatcher = if !ExcludeGlob.is_empty() {
48		GlobBuilder::new(&ExcludeGlob)
49			.literal_separator(false)
50			.build()
51			.map(|G| G.compile_matcher())
52			.ok()
53	} else {
54		None
55	};
56
57	let SearchText = Pattern.clone();
58	let mut Matches = Vec::new();
59
60	// Walk directory recursively
61	let mut Stack = vec![RootPath.clone()];
62	while let Some(Dir) = Stack.pop() {
63		let mut Entries = match fs::read_dir(&Dir).await {
64			Ok(E) => E,
65			Err(_) => continue,
66		};
67
68		while let Ok(Some(Entry)) = Entries.next_entry().await {
69			let Path = Entry.path();
70			let RelPath = Path.strip_prefix(&RootPath).unwrap_or(&Path).to_string_lossy().to_string();
71
72			// Skip hidden dirs
73			if Path.file_name().map(|N| N.to_string_lossy().starts_with('.')).unwrap_or(false) {
74				continue;
75			}
76
77			if Path.is_dir() {
78				Stack.push(Path);
79				continue;
80			}
81
82			// Check include/exclude globs
83			if let Some(Ref) = &IncludeMatcher {
84				if !Ref.is_match(&RelPath) {
85					continue;
86				}
87			}
88			if let Some(Ref) = &ExcludeMatcher {
89				if Ref.is_match(&RelPath) {
90					continue;
91				}
92			}
93
94			// Read file and search line by line
95			let Content = match fs::read_to_string(&Path).await {
96				Ok(C) => C,
97				Err(_) => continue,
98			};
99
100			for (LineIndex, Line) in Content.lines().enumerate() {
101				let Hit = if IsRegex {
102					// Simple contains fallback (no regex crate available here)
103					Line.contains(&SearchText)
104				} else if IsCaseSensitive {
105					Line.contains(&SearchText)
106				} else {
107					Line.to_lowercase().contains(&SearchText.to_lowercase())
108				};
109
110				if Hit {
111					let Uri = format!("file://{}", Path.to_string_lossy());
112					Matches.push(json!({
113						"uri": Uri,
114						"lineNumber": LineIndex + 1,
115						"preview": Line.trim(),
116					}));
117
118					if Matches.len() >= MaxResults {
119						return Ok(json!(Matches));
120					}
121				}
122			}
123		}
124	}
125
126	Ok(json!(Matches))
127}

129/// Search file paths by glob pattern in workspace.
130pub async fn handle_search_find_files(
131	runtime:Arc<ApplicationRunTime>,
132	args:Vec<Value>,
133) -> Result<Value, String> {
134	use globset::GlobBuilder;
135	use tokio::fs;
136
137	let Pattern = args
138		.first()
139		.and_then(|V| V.as_str())
140		.ok_or("search:findFiles requires pattern".to_string())?
141		.to_string();
142	let MaxResults = args.get(1).and_then(|V| V.as_u64()).unwrap_or(500) as usize;
143
144	let WorkspaceFolders = runtime.Environment.ApplicationState.Workspace.GetWorkspaceFolders();
145
146	if WorkspaceFolders.is_empty() {
147		return Ok(json!([]));
148	}
149
150	let RootPath = PathBuf::from(&WorkspaceFolders[0].URI.to_string().replace("file://", ""));
151
152	let Matcher = GlobBuilder::new(&Pattern)
153		.literal_separator(false)
154		.build()
155		.map(|G| G.compile_matcher())
156		.map_err(|Error| format!("Invalid glob pattern: {}", Error))?;
157
158	let mut Files = Vec::new();
159	let mut Stack = vec![RootPath.clone()];
160
161	while let Some(Dir) = Stack.pop() {
162		let mut Entries = match fs::read_dir(&Dir).await {
163			Ok(E) => E,
164			Err(_) => continue,
165		};
166
167		while let Ok(Some(Entry)) = Entries.next_entry().await {
168			let Path = Entry.path();
169
170			if Path.file_name().map(|N| N.to_string_lossy().starts_with('.')).unwrap_or(false) {
171				continue;
172			}
173
174			if Path.is_dir() {
175				Stack.push(Path);
176				continue;
177			}
178
179			let RelPath = Path.strip_prefix(&RootPath).unwrap_or(&Path).to_string_lossy().to_string();
180
181			if Matcher.is_match(&RelPath) {
182				Files.push(format!("file://{}", Path.to_string_lossy()));
183
184				if Files.len() >= MaxResults {
185					return Ok(json!(Files));
186				}
187			}
188		}
189	}
190
191	Ok(json!(Files))
192}

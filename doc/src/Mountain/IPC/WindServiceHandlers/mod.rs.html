119/// Internal dispatcher for the single front-end Tauri command
120/// `MountainIPCInvoke` (registered in `Binary/Main/Entry.rs::invoke_handler!`,
121/// implemented in `Binary/IPC/InvokeCommand.rs`). The outer Tauri command
122/// receives `(method: String, params: Value)`, unwraps `params` into a
123/// `Vec<Value>`, then delegates here.
124///
125/// This function is **not** a Tauri command itself - removing the previously
126/// present `#[tauri::command]` attribute avoids the false impression that
127/// `mountain_ipc_invoke` is reachable from the webview under its snake-case
128/// name. All front-end callers (Wind, Sky, Output) must `invoke(
129/// "MountainIPCInvoke", { method, params })` through `InvokeCommand::
130/// MountainIPCInvoke`; this inner function is pure Rust-side plumbing.
131///
132/// The local parameter names (`command` / `args`) are preserved for diff
133/// minimality; the frontend-facing contract (`method` / `params`) lives
134/// entirely in `InvokeCommand.rs`.

138	// not useful in forensic review. The workbench emits thousands of
139	// `logger:log` invocations per boot (every `console.*` call inside VS
140	// Code code becomes an IPC round-trip); keeping those lines only
141	// expands log volume without adding signal. The actual dispatch below
142	// still runs - this just skips the `[DEV:IPC] invoke:` line.
143	let IsHighFrequencyCommand = matches!(
144		command.as_str(),
145		"logger:log" | "logger:registerLogger" | "logger:createLogger" | "log:registerLogger" | "log:createLogger"

168	// Atom O1 + O3: every invoke flows through `SubmitToEcho` below so the

171	// decisions under load, not moving a single future across threads. This

1477				// VS Code emits `updateMenubar` every time a relevant state flips:
1478				// active editor, dirty marker, selection. A cold boot fires the call
1479				// ~20× in the first few seconds, and every one triggers an AppKit
1480				// re-render on macOS (≈ 200 ms each). We coalesce adjacent calls
1481				// through a 50 ms debouncer so only the last pending state actually
1482				// hits the native menu. Semantics match VS Code's
1483				// `ElectronMenubarControl._updateMenu` scheduler.
1484				"menubar:updateMenubar" => {
1485					use std::{
1486						sync::{Arc, Mutex as StandardMutex, OnceLock},
1487						time::Duration,
1488					};

1490					use tokio::task::JoinHandle;
1491					type MenubarCell = StandardMutex<(Option<JoinHandle<()>>, u64)>;
1492					static MENUBAR_DEBOUNCE:OnceLock<Arc<MenubarCell>> = OnceLock::new();
1493					let Cell = MENUBAR_DEBOUNCE.get_or_init(|| Arc::new(StandardMutex::new((None, 0)))).clone();
1494
1495					if let Ok(mut Guard) = Cell.lock() {
1496						if let Some(Pending) = Guard.0.take() {
1497							Pending.abort();
1498						}
1499						Guard.1 = Guard.1.saturating_add(1);
1500						let CellForTask = Cell.clone();
1501						Guard.0 = Some(tokio::spawn(async move {
1502							tokio::time::sleep(Duration::from_millis(50)).await;

1504								let N = Post.1;

1509								0

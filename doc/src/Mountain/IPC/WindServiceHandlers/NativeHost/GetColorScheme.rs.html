1//! Wire method: `nativeHost:getColorScheme`.
2//! Returns `{ dark, highContrast }`. Dark-mode probe covers macOS
3//! `AppleInterfaceStyle`, Windows `AppsUseLightTheme`, and a Linux ladder
4//! (GTK color-scheme → GTK theme name → `KDE_COLOR_SCHEME` → xfconf).
5//! High-contrast probe is Windows `HighContrast/Flags` and the GNOME a11y
6//! `high-contrast` key; other OSes return false.
7
8use std::sync::OnceLock;
9
10use serde_json::{Value, json};
11
12// Cache dark-mode result for the process lifetime. The system colour scheme
13// is queried by spawning `defaults`/`reg`/`gsettings` which adds ~5-15 ms
14// on cold start. The workbench calls `getOSColorScheme` during boot and again
15// on window focus; caching turns the second+ call into a sub-microsecond read.
16// If the user switches dark/light mode while editing, they are expected to
17// restart the editor (same behaviour as stock VS Code on Electron).
18static DARK_MODE_CACHE:OnceLock<bool> = OnceLock::new();
19
20pub async fn Fn() -> Result<Value, String> {
21	let Dark = *DARK_MODE_CACHE.get_or_init(detect_dark_mode);
22
23	let HighContrast = {
24		#[cfg(target_os = "windows")]
25		{
26			std::process::Command::new("reg")
27				.args(["query", "HKCU\\Control Panel\\Accessibility\\HighContrast", "/v", "Flags"])
28				.output()
29				.ok()
30				.map(|O| {
31					let Output = String::from_utf8_lossy(&O.stdout);
32					Output.contains("0x1") || Output.contains("REG_DWORD    1")
33				})
34				.unwrap_or(false)
35		}
36
37		#[cfg(not(target_os = "windows"))]
38		{
39			#[cfg(target_os = "linux")]
40			{
41				std::process::Command::new("gsettings")
42					.args(["get", "org.gnome.desktop.a11y.interface", "high-contrast"])
43					.output()
44					.ok()
45					.map(|O| String::from_utf8_lossy(&O.stdout).trim() == "true")
46					.unwrap_or(false)
47			}
48
49			#[cfg(not(target_os = "linux"))]
50			{
51				false
52			}
53		}
54	};
55
56	Ok(json!({ "dark": Dark, "highContrast": HighContrast }))
57}
58
59fn detect_dark_mode() -> bool {
60	// runs once then cached via OnceLock
61	#[cfg(target_os = "macos")]
62	{
63		std::process::Command::new("defaults")
64			.args(["read", "-g", "AppleInterfaceStyle"])
65			.output()
66			.ok()
67			.map(|O| String::from_utf8_lossy(&O.stdout).trim().to_lowercase().contains("dark"))
68			.unwrap_or(false)
69	}
70
71	#[cfg(target_os = "windows")]
72	{
73		std::process::Command::new("reg")
74			.args([
75				"query",
76				"HKCU\\Software\\Microsoft\\Windows\\CurrentVersion\\Themes\\Personalize",
77				"/v",
78				"AppsUseLightTheme",
79			])
80			.output()
81			.ok()
82			.map(|O| {
83				let Output = String::from_utf8_lossy(&O.stdout);
84				Output.contains("0x0") || Output.contains("REG_DWORD    0")
85			})
86			.unwrap_or(false)
87	}
88
89	#[cfg(target_os = "linux")]
90	{
91		let GtkDark = std::process::Command::new("gsettings")
92			.args(["get", "org.gnome.desktop.interface", "color-scheme"])
93			.output()
94			.ok()
95			.map(|O| String::from_utf8_lossy(&O.stdout).contains("dark"))
96			.unwrap_or(false);
97
98		if GtkDark {
99			return true;
100		}
101
102		let GtkTheme = std::process::Command::new("gsettings")
103			.args(["get", "org.gnome.desktop.interface", "gtk-theme"])
104			.output()
105			.ok()
106			.map(|O| String::from_utf8_lossy(&O.stdout).to_lowercase().contains("dark"))
107			.unwrap_or(false);
108
109		if GtkTheme {
110			return true;
111		}
112
113		let KdeDark = std::env::var("KDE_COLOR_SCHEME")
114			.ok()
115			.map(|V| V.to_lowercase().contains("dark"))
116			.unwrap_or(false);
117
118		if KdeDark {
119			return true;
120		}
121
122		let XfceDark = std::process::Command::new("xfconf-query")
123			.args(["-c", "xsettings", "-p", "/Net/ThemeName"])
124			.output()
125			.ok()
126			.map(|O| String::from_utf8_lossy(&O.stdout).to_lowercase().contains("dark"))
127			.unwrap_or(false);
128
129		XfceDark
130	}
131
132	#[cfg(not(any(target_os = "macos", target_os = "windows", target_os = "linux")))]
133	{
134		false
135	}
136}

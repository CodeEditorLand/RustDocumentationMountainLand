1//! # Compress
2//!
3//! ## File: IPC/Message/Compress/Compress.rs
4//!
5//! ## Role in Mountain Architecture
6//!
7//! Provides Gzip compression for IPC messages to optimize bandwidth usage
8//! between Mountain's Rust backend and Wind's TypeScript frontend.
9//!
10//! ## Primary Responsibility
11//!
12//! Compress and decompress IPC message payloads using Gzip algorithm to reduce
13//! transfer payload size and improve IPC communication performance.
14//!
15//! ## Secondary Responsibilities
16//!
17//! - Determine optimal batching strategy for multiple messages
18//! - Handle compression errors gracefully with fallback
19//! - Validate decompression results to prevent decompression bomb attacks
20//! - Provide size thresholds for determining when to compress
21//!
22//! ## Dependencies
23//!
24//! **External Crates:**
25//! - `flate2` - Gzip compression/decompression
26//! - `serde_json` - Message serialization
27//!
28//! **Internal Modules:**
29//! - `DefineMessage::TauriIPCMessage` - Message type being compressed
30//!
31//! ## Dependents
32//!
33//! - `TauriIPCServer` - Uses compression for message batches
34//! - `Send` - Compresses outgoing messages
35//! - `Receive` - Decompresses incoming messages
36//!
37//! ## VSCode Pattern Reference
38//!
39//! Inspired by VSCode's RPC message compression in
40//! `vs/base/parts/ipc/node/ipc.net.ts`
41//! - Adaptive compression based on payload size
42//! - Size threshold for deciding when to compress
43//! - Batching small messages for efficiency
44//!
45//! ## Security Considerations
46//!
47//! - Compression bomb protection: Maximum decompressed size enforced
48//! - Validation of decompressed size before processing
49//! - Size limits on both compression and decompression operations
50//! - Fallback to uncompressed on failure prevents DoS via corruption
51//!
52//! ## Performance Considerations
53//!
54//! - Compression performed synchronously (simpler, small payloads)
55//! - Compress only messages > 1KB to avoid overhead on small data
56//! - Adaptive compression level (default = 6, balanced speed/ratio)
57//! - Batch multiple small messages when ShouldBatch returns true
58//!
59//! ## Error Handling Strategy
60//!
61//! - Returns Result<T, CompressionError> for explicit error handling
62//! - Logs compression failures without propagating to caller (graceful
63//!   degradation)
64//! - Falls back to uncompressed on compression failure
65//! - Detailed error messages include context for debugging
66//!
67//! ## Thread Safety
68//!
69//! - All methods are `&self` and safe for concurrent access
70//! - No interior mutability, state is configuration only
71//!
72//! ## TODO Items
73//!
74//! - [ ] Add support for alternative compression algorithms (zstd, brotli)
75//! - [ ] Implement adaptive compression based on message type
76//! - [ ] Add compression statistics tracking

86/// Maximum decompressed size to prevent compression bomb attacks (10MB)
87const MAX_DECOMPRESSED_SIZE:usize = 10 * 1024 * 1024;
88
89/// Message compression utility for optimizing IPC message transfer
90///
91/// This struct provides Gzip-based compression for IPC messages, adapting the
92/// compression level based on payload size and providing graceful fallback on
93/// errors.
94pub struct Compressor {
95	/// Compression level (0-9), higher = better ratio but slower
96	/// 0 = no compression, 1 = fastest, 6 = balanced, 9 = best
97	CompressionLevel:u32,
98	/// Minimum number of messages required for batching
99	BatchSize:usize,
100	/// Minimum size in bytes before compressing a single message
101	SingleMessageThreshold:usize,
102}
103
104impl Compressor {
105	/// Create a new message compressor with specified parameters
106	///
107	/// # Arguments
108	/// * `CompressionLevel` - Gzip compression level (0-9)
109	/// * `BatchSize` - Minimum messages to batch
110	/// * `SingleMessageThreshold` - Min bytes to compress single message
111	///
112	/// # Returns
113	/// A new Compressor instance
114	///
115	/// # Defaults when not specified:
116	/// - CompressionLevel: 6 (balanced speed/ratio)
117	/// - BatchSize: 10 messages
118	/// - SingleMessageThreshold: 1024 bytes (1KB)
119	pub fn new(CompressionLevel:u32, BatchSize:usize, SingleMessageThreshold:usize) -> Self {
120		Self { CompressionLevel, BatchSize, SingleMessageThreshold }
121	}
122
123	/// Create compressor with default values
124	pub fn defaults() -> Self { Self::new(6, 10, 1024) }
125
126	/// Compress messages using Gzip for efficient transfer
127	///
128	/// # Arguments
129	/// * `Messages` - Vector of messages to compress
130	///
131	/// # Returns
132	/// Ok(Compressed bytes) or Err with error description
133	///
134	/// # Security
135	/// - Validates input size before compression
136	/// - Prevents memory exhaustion via oversized inputs
137	pub fn compress_messages(&self, Messages:&[TauriIPCMessage]) -> Result<Vec<u8>, String> {
138		// Validate input size to prevent memory exhaustion
139		let total_size = std::mem::size_of_val(Messages);
140		if total_size > MAX_DECOMPRESSED_SIZE {
141			return Err(format!(
142				"Input too large: {} bytes (max: {})",
143				total_size, MAX_DECOMPRESSED_SIZE
144			));
145		}
146
147		// Serialize messages to JSON
148		let SerializedMessages =
149			serde_json::to_vec(Messages).map_err(|e| format!("Failed to serialize messages: {}", e))?;
150
151		// Validate serialized size
152		if SerializedMessages.len() > MAX_DECOMPRESSED_SIZE {
153			return Err(format!(
154				"Serialized data too large: {} bytes (max: {})",
155				SerializedMessages.len(),
156				MAX_DECOMPRESSED_SIZE
157			));
158		}
159
160		// Check if compression is beneficial (only compress if size exceeds threshold)
161		if SerializedMessages.len() < self.SingleMessageThreshold {
162			debug!(
163				"[Compress] Skipping compression: data size {} < threshold {}",
164				SerializedMessages.len(),
165				self.SingleMessageThreshold
166			);
167			return Ok(SerializedMessages);
168		}
169
170		// Compress using Gzip
171		let mut encoder = GzEncoder::new(Vec::new(), Compression::new(self.CompressionLevel));
172		encoder
173			.write_all(&SerializedMessages)
174			.map_err(|e| format!("Failed to write to compressor: {}", e))?;
175
176		let compressed_data = encoder.finish().map_err(|e| format!("Failed to finish compression: {}", e))?;
177
178		// Only return compressed if it's smaller than original
179		let compression_ratio = (compressed_data.len() as f64 / SerializedMessages.len() as f64) * 100.0;
180		if compressed_data.len() >= SerializedMessages.len() {
181			debug!(
182				"[Compress] Compression not beneficial: {}% ({} bytes vs {})",
183				compression_ratio,
184				compressed_data.len(),
185				SerializedMessages.len()
186			);
187			return Ok(SerializedMessages);
188		}
189
190		debug!(
191			"[Compress] Compressed {} messages: {} -> {} bytes ({:.1}%)",
192			Messages.len(),
193			SerializedMessages.len(),
194			compressed_data.len(),
195			compression_ratio
196		);
197
198		Ok(compressed_data)
199	}
200
201	/// Decompress messages from compressed data
202	///
203	/// # Arguments
204	/// * `CompressedData` - Byte slice containing compressed data
205	///
206	/// # Returns
207	/// Ok(Vector of TauriIPCMessage) or Err with error description
208	///
209	/// # Security
210	/// - Enforces MAX_DECOMPRESSED_SIZE limit to prevent decompression bomb
211	/// - Validates decompressed data structure before returning
212	pub fn decompress_messages(&self, CompressedData:&[u8]) -> Result<Vec<TauriIPCMessage>, String> {
213		validate_decompression_input(CompressedData.len())?;
214
215		let mut decoder = GzDecoder::new(CompressedData);
216		let mut DecompressedData = Vec::with_capacity(65536); // Pre-allocate 64KB
217
218		// Read with size limit to prevent decompression bomb
219		let bytes_read = decoder
220			.read_to_end(&mut DecompressedData)
221			.map_err(|e| format!("Failed to decompress data: {}", e))?;
222
223		validate_decompressed_size(bytes_read)?;
224
225		// Deserialize messages
226		serde_json::from_slice(&DecompressedData).map_err(|e| format!("Failed to deserialize messages: {}", e))
227	}
228
229	/// Check if messages should be batched for compression
230	///
231	/// # Arguments
232	/// * `MessagesCount` - Number of messages to consider
233	///
234	/// # Returns
235	/// true if batching should be used, false otherwise
236	pub fn should_batch(&self, MessagesCount:usize) -> bool { MessagesCount >= self.BatchSize }
237
238	/// Check if a single message should be compressed
239	///
240	/// # Arguments
241	/// * `Message` - The message to check
242	///
243	/// # Returns
244	/// true if compression should be applied, false otherwise
245	pub fn should_compress_single(&self, Message:&TauriIPCMessage) -> Result<bool, String> {
246		// Serialize to JSON to check size
247		let serialized = serde_json::to_vec(Message).map_err(|e| format!("Failed to serialize message: {}", e))?;
248
249		Ok(serialized.len() >= self.SingleMessageThreshold)
250	}
251}
252
253/// Validate decompression input size
254fn validate_decompression_input(CompressedSize:usize) -> Result<(), String> {
255	// Reject unreasonably large compressed inputs (even though decompression has
256	// limit)
257	const MAX_COMPRESSED_INPUT:usize = 5 * 1024 * 1024; // 5MB
258
259	if CompressedSize > MAX_COMPRESSED_INPUT {
260		return Err(format!(
261			"Compressed input too large: {} bytes (max: {})",
262			CompressedSize, MAX_COMPRESSED_INPUT
263		));
264	}
265	Ok(())
266}
267
268/// Validate decompressed size
269fn validate_decompressed_size(DecompressedSize:usize) -> Result<(), String> {
270	if DecompressedSize > MAX_DECOMPRESSED_SIZE {
271		return Err(format!(
272			"Decompressed data too large: {} bytes (max: {})",
273			DecompressedSize, MAX_DECOMPRESSED_SIZE
274		));
275	}
276	Ok(())
277}

279/// Compression error type for better error handling
280#[derive(Debug, thiserror::Error)]
281pub enum CompressionError {
282	#[error("Serialization failed: {0}")]
283	SerializationFailed(String),
284
285	#[error("Compression failed: {0}")]
286	CompressionFailed(String),
287
288	#[error("Decompression failed: {0}")]
289	DecompressionFailed(String),
290
291	#[error("Size limit exceeded: {0}")]
292	SizeLimitExceeded(String),
293
294	#[error("Invalid data: {0}")]
295	InvalidData(String),
296}
297
298impl From<CompressionError> for String {
299	fn from(error:CompressionError) -> Self { error.to_string() }
300}

342		let messages = vec![
343			TauriIPCMessage::new("channel1", json!({"d": 1}), None),
344			TauriIPCMessage::new("channel2", json!({"d": 2}), None),

348		// Compress

351		// Decompress

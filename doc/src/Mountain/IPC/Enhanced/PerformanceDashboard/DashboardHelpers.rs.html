27/// Human-readable display name for a `MetricType` variant.
28pub fn metric_type_name(metric_type:&MetricType) -> &'static str {
29	match metric_type {
30		MetricType::MessageProcessingTime => "Message Processing Time",
31		MetricType::ConnectionLatency => "Connection Latency",
32		MetricType::MemoryUsage => "Memory Usage",
33		MetricType::CpuUsage => "CPU Usage",
34		MetricType::NetworkThroughput => "Network Throughput",
35		MetricType::ErrorRate => "Error Rate",
36		MetricType::QueueSize => "Queue Size",
37	}

1//! # WorkspaceFileService (Workspace)
2//!
3//! RESPONSIBILITIES:
4//! - Parses `.code-workspace` configuration files (VSCode multi-root workspace format)
5//! - Resolves relative folder paths to absolute filesystem paths
6//! - Converts parsed workspace folders into `WorkspaceFolderStateDTO` instances
7//! - Handles path canonicalization and URI conversion
8//!
9//! ARCHITECTURAL ROLE:
10//! - Utility module for workspace configuration management
11//! - Used by [`MountainEnvironment`](crate::Environment::MountainEnvironment) during
12//!   workspace initialization and configuration loading
13//! - Integrates with [`ApplicationState`](crate::ApplicationState::ApplicationState)
14//!   for workspace folder state management
15//!
16//! FILE FORMAT:
17//! - Expects JSON format conforming to VSCode `.code-workspace` schema
18//! - Top-level object contains `folders` array (can also have `settings`, `extensions`)
19//! - Each folder entry has at least a `path` field (relative to workspace file)
20//! - Example:
21//!   ```json
22//!   {
23//!     "folders": [
24//!       { "path": "." },
25//!       { "path": "../other-project" }
26//!     ]
27//!   }
28//!   ```
29//!
30//! PARSING FLOW:
31//! 1. Read `.code-workspace` file content as string
32//! 2. Deserialize JSON into `WorkspaceFile` struct (using serde)
33//! 3. Determine workspace file's parent directory (base for relative paths)
34//! 4. For each folder entry:
35//!    - Join relative path with base directory
36//!    - Canonicalize to resolve symlinks and relative segments (`..`, `.`)
37//!    - Convert absolute path to `file://` URI via `Url::from_directory_path`
38//!    - Extract folder name from path (fallback to "untitled-folder")
39//!    - Assign sequential index based on declaration order
40//! 5. Return `Vec<WorkspaceFolderStateDTO>` with all resolved folders
41//!
42//! ERROR HANDLING:
43//! - Returns [`CommonError`](CommonLibrary::Error::CommonError) on any failure
44//! - JSON deserialization errors → `SerializationError`
45//! - Missing parent directory → `FileSystemIO`
46//! - Path canonicalization failure → `FileSystemNotFound`
47//! - URI conversion failure → `InvalidArgument`
48//!
49//! PERFORMANCE:
50//! - Synchronous function but should be called from async context
51//! - Each folder path undergoes I/O: join + canonicalize (can be slow on network drives)
52//! - Consider caching parsed workspace files if frequently accessed
53//!
54//! VS CODE REFERENCE:
55//! - `vs/workbench/workspaces/common/workspace.ts` - workspace file format
56//! - `vs/workbench/services/workspace/browser/workspaceService.ts` - workspace service
57//! - `vs/platform/workspace/common/workspace.ts` - workspace interfaces
58//!
59//! TODO:
60//! - Add validation for workspace file schema version
61//! - Support workspace file `settings` section (merge into configuration)
62//! - Parse workspace `extensions` section (recommend extensions)
63//! - Add support for workspace file glob patterns in folder paths
64//! - Implement workspace file change watching (reload on external edits)
65//! - Add workspace file templates for common multi-root configurations
66//! - Support workspace file encryption for sensitive paths
67//! - Handle workspace files on remote filesystems (SSH, containers)
68//! - Add workspace file migration tools (format upgrades)
69//! - Implement workspace file validation and linting
70//! - Support workspace file includes (composite workspaces)
71//!
72//! MODULE CONTENTS:
73//! - Structs: `WorkspaceFile`, `WorkspaceFolderEntry` (serde deserialization)
74//! - Function: `ParseWorkspaceFile` - main entry point
75//! - Data type: [`WorkspaceFolderStateDTO`](crate::ApplicationState::DTO::WorkspaceFolderStateDTO)

77use std::path::Path;
78
79use CommonLibrary::Error::CommonError::CommonError;
80use serde::Deserialize;
81use url::Url;
82
83use crate::ApplicationState::DTO::WorkspaceFolderStateDTO::WorkspaceFolderStateDTO;
84
85#[derive(Deserialize, Debug)]
86struct WorkspaceFile {
87	folders:Vec<WorkspaceFolderEntry>,
88	// Can also contain 'settings', 'extensions', etc.
89}
90
91#[derive(Deserialize, Debug)]
92struct WorkspaceFolderEntry {
93	path:String,
94}
95
96/// Parses a `.code-workspace` file content and resolves the folder paths.
97///
98/// # Parameters
99/// * `WorkspaceFilePath`: The absolute path to the `.code-workspace` file.
100/// * `FileContent`: The raw string content of the file.
101///
102/// # Returns
103/// A `Result` containing a vector of `WorkspaceFolderStateDTO`s.
104pub fn ParseWorkspaceFile(
105	WorkspaceFilePath:&Path,
106
107	FileContent:&str,
108) -> Result<Vec<WorkspaceFolderStateDTO>, CommonError> {
109	let Parsed:WorkspaceFile = serde_json::from_str(FileContent)
110		.map_err(|Error| CommonError::SerializationError { Description:Error.to_string() })?;
111
112	let WorkspaceFileDirectory = WorkspaceFilePath.parent().ok_or_else(|| {
113		CommonError::FileSystemIO {
114			Path:WorkspaceFilePath.to_path_buf(),
115
116			Description:"Cannot get parent directory of workspace file".to_string(),
117		}
118	})?;
119
120	let Folders:Result<Vec<WorkspaceFolderStateDTO>, CommonError> = Parsed
121		.folders
122		.into_iter()
123		.enumerate()
124		.map(|(Index, Entry)| {
125			let FolderPath = WorkspaceFileDirectory.join(Entry.path);
126
127			let CanonicalPath = FolderPath
128				.canonicalize()
129				.map_err(|_| CommonError::FileSystemNotFound(FolderPath.clone()))?;
130
131			let FolderURI = Url::from_directory_path(&CanonicalPath).map_err(|_| {
132				CommonError::InvalidArgument {
133					ArgumentName:"path".into(),
134
135					Reason:format!("Could not convert path '{}' to URL", CanonicalPath.display()),
136				}
137			})?;
138
139			let Name = CanonicalPath
140				.file_name()
141				.and_then(|n| n.to_str())
142				.unwrap_or("untitled-folder")
143				.to_string();
144
145			Ok(WorkspaceFolderStateDTO { URI:FolderURI, Name, Index })
146		})
147		.collect();
148
149	Folders
150}

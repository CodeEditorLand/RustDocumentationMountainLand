1// # CocoonServiceImpl Implementation
2//
3// This module implements the main gRPC service for Mountain-Cocoon
4// communication. It handles all requests from the Cocoon extension host
5// sidecar.
6//
7// ## Service Responsibilities
8//
9// - **Initialization**: Handshake and extension host initialization
10// - **Commands**: Register and execute extension commands
11// - **Language Features**: Hover, completion, definition, references, code
12// actions
13// - **File System**: Read, write, stat, and watch files
14// - **Terminal**: Manage terminal instances and I/O
15// - **Tree View**: Register providers and get tree children
16// - **SCM**: Source control management and git operations
17// - **Debug**: Debug adapter registration and session management
18// - **Save Participants**: Handle save events from extensions
19//
20// ## Architecture
21//
22// The service maintains references to:
23// - `MountainEnvironment`: Access to all Mountain services and providers
24// - `ActiveOperations`: Registry of cancellable operations
25//
26// ## Error Handling
27//
28// All methods return `tonic::Result<T>` and use proper error conversion
29// from internal errors to gRPC status codes.

261/// Implementation of the CocoonService gRPC server
262///
263/// This struct handles all incoming requests from the Cocoon extension host
264/// sidecar and dispatches them to the appropriate Mountain services.
265#[derive(Clone)]
266pub struct CocoonServiceImpl {
267	/// Mountain environment providing access to all services
268	environment:Arc<MountainEnvironment>,
269
270	/// Registry of active operations with their cancellation tokens
271	/// Maps request ID to cancellation token for operation cancellation
272	ActiveOperations:Arc<RwLock<HashMap<u64, tokio_util::sync::CancellationToken>>>,
273}
274
275impl CocoonServiceImpl {
276	/// Creates a new instance of the CocoonService server
277	///
278	/// # Parameters
279	/// - `environment`: Mountain environment with access to all services
280	///
281	/// # Returns
282	/// A new CocoonService instance
283	pub fn new(environment:Arc<MountainEnvironment>) -> Self {
284		info!("[CocoonService] New instance created");
285
286		Self { environment, ActiveOperations:Arc::new(RwLock::new(HashMap::new())) }
287	}
288
289	/// Registers an operation for potential cancellation
290	///
291	/// # Parameters
292	/// - `request_id`: The request identifier for the operation
293	///
294	/// # Returns
295	/// A cancellation token that can be used to cancel the operation
296	pub async fn RegisterOperation(&self, request_id:u64) -> tokio_util::sync::CancellationToken {
297		let token = tokio_util::sync::CancellationToken::new();
298		self.ActiveOperations.write().await.insert(request_id, token.clone());
299		debug!("[CocoonService] Registered operation {} for cancellation", request_id);
300		token
301	}
302
303	/// Unregisters an operation after completion
304	///
305	/// # Parameters
306	/// - `request_id`: The request identifier to unregister
307	pub async fn UnregisterOperation(&self, request_id:u64) {
308		self.ActiveOperations.write().await.remove(&request_id);
309		debug!("[CocoonService] Unregistered operation {}", request_id);
310	}
311
312	/// Registers a language feature provider in ApplicationState.
313	///
314	/// Converts the gRPC request fields into a `ProviderRegistrationDTO` and
315	/// stores it in `ApplicationState.Extension.ProviderRegistration`.
316	///
317	/// # Parameters
318	/// - `handle`: Unique provider handle
319	/// - `provider_type`: The type of language feature
320	/// - `language_selector`: Language scope (e.g. "typescript")
321	/// - `extension_id`: Extension that registered this provider
322	fn RegisterProvider(&self, handle:u32, provider_type:ProviderType, language_selector:&str, extension_id:&str) {
323		// SideCarIdentifier = "cocoon-main" so FeatureMethods::invoke_provider can
324		// route back via Vine::Client::SendRequestToSideCar("cocoon-main", ...).
325		// Selector stored as array so ProviderLookup::get_matching_provider's
326		// `.as_array()` call finds the language entry: [{ "language": "typescript" }].
327		let dto = ProviderRegistrationDTO {
328			Handle:handle,
329			ProviderType:provider_type,
330			Selector:json!([{ "language": language_selector }]),
331			SideCarIdentifier:"cocoon-main".to_string(),
332			ExtensionIdentifier:json!(extension_id),
333			Options:None,
334		};
335		self.environment
336			.ApplicationState
337			.Extension
338			.ProviderRegistration
339			.RegisterProvider(handle, dto);
340		debug!(
341			"[CocoonService] Provider {:?} registered: handle={}, language={}",
342			provider_type, handle, language_selector
343		);
344	}
345
346	/// Extracts a filesystem path from a URI proto message.
347	///
348	/// Handles both `file://` URIs and bare paths. Returns `None` if the URI
349	/// is absent or the path cannot be extracted.
350	fn UriToPath(uri_opt:Option<&Uri>) -> Option<std::path::PathBuf> {
351		let value = uri_opt?.value.as_str();
352		if value.is_empty() {
353			return None;
354		}
355		// Strip file:// prefix if present
356		let path_str = if let Some(Stripped) = value.strip_prefix("file://") {
357			Stripped
358		} else if value.starts_with('/') || (value.len() > 1 && value.as_bytes()[1] == b':') {
359			// Bare absolute path (Unix or Windows)
360			value
361		} else {
362			// Unknown scheme - return as-is
363			value
364		};
365		Some(std::path::PathBuf::from(path_str))
366	}
367}

369#[async_trait]
370impl CocoonService for CocoonServiceImpl {
371	/// Process Mountain requests from Cocoon (generic request-response).
372	///
373	/// Routes legacy `fs.*` / `commands.*` / `secrets.*` method names used by
374	/// Cocoon's `FileSystemService` and other services that call Mountain via
375	/// the generic `ProcessCocoonRequest` RPC instead of the typed methods.
376	///
377	/// Parameters are JSON-encoded bytes in `request.parameter`. Results are
378	/// JSON-encoded bytes in `response.result`.
379	async fn process_mountain_request(
380		&self,
381		request:Request<GenericRequest>,
382	) -> Result<Response<GenericResponse>, Status> {
383		let Req = request.into_inner();
384		let RequestId = Req.request_identifier;
385
386		debug!("[CocoonService] generic request: method={} id={}", Req.method, RequestId);
387
388		/// Serialise a value into the `result` bytes of a GenericResponse.
389		fn OkResponse(RequestId:u64, Value:&impl serde::Serialize) -> Response<GenericResponse> {
390			let Bytes = serde_json::to_vec(Value).unwrap_or_default();
391			Response::new(GenericResponse { request_identifier:RequestId, result:Bytes, error:None })
392		}
393
394		/// Build an error GenericResponse.
395		fn ErrResponse(RequestId:u64, Code:i32, Message:String) -> Response<GenericResponse> {
396			Response::new(GenericResponse {
397				request_identifier:RequestId,
398				result:Vec::new(),
399				error:Some(RpcError { code:Code, message:Message, data:Vec::new() }),
400			})
401		}
402
403		// Deserialise the generic parameter bytes as a JSON value
404		let Params:serde_json::Value = if Req.parameter.is_empty() {
405			serde_json::Value::Null
406		} else {
407			serde_json::from_slice(&Req.parameter).unwrap_or(serde_json::Value::Null)
408		};
409
410		match Req.method.as_str() {
411			// ---- File System ---- (Cocoon FileSystemService uses these paths)
412			"fs.readFile" | "file:read" => {
413				let Path = Params
414					.as_str()
415					.or_else(|| Params.get("path").and_then(|V| V.as_str()))
416					.unwrap_or("");
417				match tokio::fs::read(Path).await {
418					Ok(Content) => Ok(OkResponse(RequestId, &Content)),
419					Err(Error) => Ok(ErrResponse(RequestId, -32000, format!("fs.readFile: {}", Error))),
420				}
421			},
422			"fs.writeFile" | "file:write" => {
423				let Path = Params.get("path").and_then(|V| V.as_str()).unwrap_or("");
424				let Content:Vec<u8> = Params
425					.get("content")
426					.and_then(|V| V.as_array())
427					.map(|A| A.iter().filter_map(|B| B.as_u64().map(|N| N as u8)).collect())
428					.unwrap_or_default();
429				match tokio::fs::write(Path, &Content).await {
430					Ok(()) => Ok(OkResponse(RequestId, &serde_json::Value::Null)),
431					Err(Error) => Ok(ErrResponse(RequestId, -32000, format!("fs.writeFile: {}", Error))),
432				}
433			},
434			"fs.stat" | "file:stat" => {
435				let Path = Params
436					.as_str()
437					.or_else(|| Params.get("path").and_then(|V| V.as_str()))
438					.unwrap_or("");
439				match tokio::fs::metadata(Path).await {
440					Ok(Meta) => {
441						let Mtime = Meta
442							.modified()
443							.ok()
444							.and_then(|T| T.duration_since(UNIX_EPOCH).ok())
445							.map(|D| D.as_millis() as u64)
446							.unwrap_or(0);
447						Ok(OkResponse(
448							RequestId,
449							&json!({
450								"type": if Meta.is_dir() { 2 } else { 1 },
451								"is_file": Meta.is_file(),
452								"is_directory": Meta.is_dir(),
453								"size": Meta.len(),
454								"mtime": Mtime,
455							}),
456						))
457					},
458					Err(Error) => Ok(ErrResponse(RequestId, -32000, format!("fs.stat: {}", Error))),
459				}
460			},
461			"fs.listDir" | "fs.readdir" | "file:readdir" => {
462				let Path = Params
463					.as_str()
464					.or_else(|| Params.get("path").and_then(|V| V.as_str()))
465					.unwrap_or("");
466				match tokio::fs::read_dir(Path).await {
467					Ok(mut Entries) => {
468						// Return [{name, type}] where type 1=File 2=Directory
469						let mut Items:Vec<serde_json::Value> = Vec::new();
470						while let Ok(Some(Entry)) = Entries.next_entry().await {
471							if let Some(Name) = Entry.file_name().to_str() {
472								let IsDir = Entry.file_type().await.map(|T| T.is_dir()).unwrap_or(false);
473								Items.push(json!({ "name": Name, "type": if IsDir { 2u32 } else { 1u32 } }));
474							}
475						}
476						Ok(OkResponse(RequestId, &Items))
477					},
478					Err(Error) => Ok(ErrResponse(RequestId, -32000, format!("fs.listDir: {}", Error))),
479				}
480			},
481			"fs.createDir" | "file:mkdir" => {
482				let Path = Params
483					.as_str()
484					.or_else(|| Params.get("path").and_then(|V| V.as_str()))
485					.unwrap_or("");
486				match tokio::fs::create_dir_all(Path).await {
487					Ok(()) => Ok(OkResponse(RequestId, &serde_json::Value::Null)),
488					Err(Error) => Ok(ErrResponse(RequestId, -32000, format!("fs.createDir: {}", Error))),
489				}
490			},
491			"fs.delete" | "file:delete" => {
492				let Path = Params
493					.as_str()
494					.or_else(|| Params.get("path").and_then(|V| V.as_str()))
495					.unwrap_or("");
496				let Result = if std::path::Path::new(Path).is_dir() {
497					tokio::fs::remove_dir_all(Path).await
498				} else {
499					tokio::fs::remove_file(Path).await
500				};
501				match Result {
502					Ok(()) => Ok(OkResponse(RequestId, &serde_json::Value::Null)),
503					Err(Error) => Ok(ErrResponse(RequestId, -32000, format!("fs.delete: {}", Error))),
504				}
505			},
506			"fs.rename" | "file:move" => {
507				let From = Params.get("from").and_then(|V| V.as_str()).unwrap_or("");
508				let To = Params.get("to").and_then(|V| V.as_str()).unwrap_or("");
509				match tokio::fs::rename(From, To).await {
510					Ok(()) => Ok(OkResponse(RequestId, &serde_json::Value::Null)),
511					Err(Error) => Ok(ErrResponse(RequestId, -32000, format!("fs.rename: {}", Error))),
512				}
513			},
514			// ---- Commands ----
515			"commands.execute" => {
516				let CommandId = Params.get("id").and_then(|V| V.as_str()).unwrap_or("").to_string();
517				let Arg = Params.get("arg").cloned().unwrap_or(serde_json::Value::Null);
518				match self.environment.ExecuteCommand(CommandId, Arg).await {
519					Ok(Value) => Ok(OkResponse(RequestId, &Value)),
520					Err(Error) => Ok(ErrResponse(RequestId, -32000, Error.to_string())),
521				}
522			},
523			// ---- Commands (Cocoon MountainGRPCClient format) ----
524			"executeCommand" => {
525				let CommandId = Params.get("commandId").and_then(|V| V.as_str()).unwrap_or("").to_string();
526				let Arg = Params
527					.get("arguments")
528					.and_then(|A| A.as_array())
529					.and_then(|A| A.first())
530					.cloned()
531					.unwrap_or(serde_json::Value::Null);
532				match self.environment.ExecuteCommand(CommandId, Arg).await {
533					Ok(Value) => Ok(OkResponse(RequestId, &json!({ "result": Value }))),
534					Err(Error) => Ok(ErrResponse(RequestId, -32000, Error.to_string())),
535				}
536			},
537			"unregisterCommand" => {
538				let ExtensionId = Params.get("extensionId").and_then(|V| V.as_str()).unwrap_or("").to_string();
539				let CommandId = Params.get("commandId").and_then(|V| V.as_str()).unwrap_or("").to_string();
540				match self.environment.UnregisterCommand(ExtensionId, CommandId).await {
541					Ok(()) => Ok(OkResponse(RequestId, &json!({ "success": true }))),
542					Err(Error) => Ok(ErrResponse(RequestId, -32000, Error.to_string())),
543				}
544			},
545			// ---- Window dialogs (Window.ts method names) ----
546			"UserInterface.ShowOpenDialog" => {
547				use CommonLibrary::UserInterface::{
548					DTO::OpenDialogOptionsDTO::OpenDialogOptionsDTO,
549					UserInterfaceProvider::UserInterfaceProvider,
550				};
551				let Title = Params
552					.get(0)
553					.and_then(|V| V.get("title"))
554					.and_then(|T| T.as_str())
555					.map(|S| S.to_string());
556				let Options = OpenDialogOptionsDTO {
557					Base:CommonLibrary::UserInterface::DTO::DialogOptionsDTO::DialogOptionsDTO {
558						Title,
559						..Default::default()
560					},
561					..OpenDialogOptionsDTO::default()
562				};
563				match self.environment.ShowOpenDialog(Some(Options)).await {
564					Ok(Some(Paths)) => {
565						let Uris:Vec<String> = Paths.iter().map(|P| format!("file://{}", P.display())).collect();
566						Ok(OkResponse(RequestId, &json!(Uris)))
567					},
568					Ok(None) => Ok(OkResponse(RequestId, &json!(serde_json::Value::Array(vec![])))),
569					Err(Error) => Ok(ErrResponse(RequestId, -32000, Error.to_string())),
570				}
571			},
572			"UserInterface.ShowSaveDialog" => {
573				use CommonLibrary::UserInterface::{
574					DTO::SaveDialogOptionsDTO::SaveDialogOptionsDTO,
575					UserInterfaceProvider::UserInterfaceProvider,
576				};
577				let Title = Params
578					.get(0)
579					.and_then(|V| V.get("title"))
580					.and_then(|T| T.as_str())
581					.map(|S| S.to_string());
582				let Options = SaveDialogOptionsDTO {
583					Base:CommonLibrary::UserInterface::DTO::DialogOptionsDTO::DialogOptionsDTO {
584						Title,
585						..Default::default()
586					},
587					..SaveDialogOptionsDTO::default()
588				};
589				match self.environment.ShowSaveDialog(Some(Options)).await {
590					Ok(Some(Path)) => Ok(OkResponse(RequestId, &json!(format!("file://{}", Path.display())))),
591					Ok(None) => Ok(OkResponse(RequestId, &serde_json::Value::Null)),
592					Err(Error) => Ok(ErrResponse(RequestId, -32000, Error.to_string())),
593				}
594			},
595			"UserInterface.ShowInputBox" => {
596				use CommonLibrary::UserInterface::{
597					DTO::InputBoxOptionsDTO::InputBoxOptionsDTO,
598					UserInterfaceProvider::UserInterfaceProvider,
599				};
600				let Opts = Params.get(0);
601				let Options = InputBoxOptionsDTO {
602					Prompt:Opts
603						.and_then(|V| V.get("prompt"))
604						.and_then(|P| P.as_str())
605						.map(|S| S.to_string()),
606					PlaceHolder:Opts
607						.and_then(|V| V.get("placeHolder"))
608						.and_then(|P| P.as_str())
609						.map(|S| S.to_string()),
610					IsPassword:Some(Opts.and_then(|V| V.get("password")).and_then(|B| B.as_bool()).unwrap_or(false)),
611					Value:Opts
612						.and_then(|V| V.get("value"))
613						.and_then(|V| V.as_str())
614						.map(|S| S.to_string()),
615					Title:None,
616					IgnoreFocusOut:None,
617				};
618				match self.environment.ShowInputBox(Some(Options)).await {
619					Ok(Some(Text)) => Ok(OkResponse(RequestId, &json!(Text))),
620					Ok(None) => Ok(OkResponse(RequestId, &serde_json::Value::Null)),
621					Err(Error) => Ok(ErrResponse(RequestId, -32000, Error.to_string())),
622				}
623			},
624			// ---- Native shell operations ----
625			"openExternal" => {
626				use tauri::Emitter;
627				let Url = Params
628					.as_str()
629					.or_else(|| Params.get("url").and_then(|V| V.as_str()))
630					.unwrap_or("")
631					.to_string();

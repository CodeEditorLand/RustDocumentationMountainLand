2//! Secret Storage domain handlers for CocoonService.
3//!
4//! Typed gRPC RPCs: get_secret, store_secret, delete_secret.
5
6use CommonLibrary::Secret::SecretProvider::SecretProvider;
7use tonic::{Response, Status};
8
9use super::CocoonServiceImpl;
10use crate::{
11	Vine::Generated::{DeleteSecretRequest, Empty, GetSecretRequest, GetSecretResponse, StoreSecretRequest},
12	dev_log,
13};
14
15pub async fn GetSecret(
16	Service:&CocoonServiceImpl,
17	req:GetSecretRequest,
18) -> Result<Response<GetSecretResponse>, Status> {
19	dev_log!("cocoon", "[CocoonService] get_secret: key={}", req.key);
20
21	// The gRPC proto only carries `key`; we use the app name as the
22	// extension identifier (keyring service scoping).
23	match Service.environment.GetSecret(String::new(), req.key.clone()).await {
24		Ok(Some(Value)) => Ok(Response::new(GetSecretResponse { value:Value })),
25		Ok(None) => Ok(Response::new(GetSecretResponse { value:String::new() })),
26		Err(Error) => {
27			dev_log!("cocoon", "warn: [CocoonService] get_secret failed key={}: {}", req.key, Error);
28			Err(Status::internal(format!("get_secret: {}", Error)))
29		},
30	}
31}
32
33pub async fn StoreSecret(Service:&CocoonServiceImpl, req:StoreSecretRequest) -> Result<Response<Empty>, Status> {
34	dev_log!("cocoon", "[CocoonService] store_secret: key={}", req.key);
35
36	match Service.environment.StoreSecret(String::new(), req.key.clone(), req.value).await {
37		Ok(()) => Ok(Response::new(Empty {})),
38		Err(Error) => {
39			dev_log!("cocoon", "warn: [CocoonService] store_secret failed key={}: {}", req.key, Error);
40			Err(Status::internal(format!("store_secret: {}", Error)))
41		},
42	}
43}
44
45pub async fn DeleteSecret(Service:&CocoonServiceImpl, req:DeleteSecretRequest) -> Result<Response<Empty>, Status> {
46	dev_log!("cocoon", "[CocoonService] delete_secret: key={}", req.key);
47
48	match Service.environment.DeleteSecret(String::new(), req.key.clone()).await {
49		Ok(()) => Ok(Response::new(Empty {})),
50		Err(Error) => {
51			dev_log!(
52				"cocoon",
53				"warn: [CocoonService] delete_secret failed key={}: {}",
54				req.key,
55				Error
56			);
57			Err(Status::internal(format!("delete_secret: {}", Error)))
58		},
59	}
60}

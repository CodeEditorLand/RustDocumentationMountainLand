265/// Implementation of the CocoonService gRPC server
266///
267/// This struct handles all incoming requests from the Cocoon extension host
268/// sidecar and dispatches them to the appropriate Mountain services.
269#[derive(Clone)]
270pub struct CocoonServiceImpl {
271	/// Mountain environment providing access to all services
272	environment:Arc<MountainEnvironment>,
273
274	/// Registry of active operations with their cancellation tokens
275	/// Maps request ID to cancellation token for operation cancellation
276	ActiveOperations:Arc<RwLock<HashMap<u64, tokio_util::sync::CancellationToken>>>,
277}
278
279impl CocoonServiceImpl {
280	/// Creates a new instance of the CocoonService server
281	///
282	/// # Parameters
283	/// - `environment`: Mountain environment with access to all services
284	///
285	/// # Returns
286	/// A new CocoonService instance
287	pub fn new(environment:Arc<MountainEnvironment>) -> Self {
288		dev_log!("cocoon", "[CocoonService] New instance created");
289
290		Self { environment, ActiveOperations:Arc::new(RwLock::new(HashMap::new())) }
291	}
292
293	/// Registers an operation for potential cancellation
294	///
295	/// # Parameters
296	/// - `request_id`: The request identifier for the operation
297	///
298	/// # Returns
299	/// A cancellation token that can be used to cancel the operation
300	pub async fn RegisterOperation(&self, request_id:u64) -> tokio_util::sync::CancellationToken {
301		let token = tokio_util::sync::CancellationToken::new();
302		self.ActiveOperations.write().await.insert(request_id, token.clone());
303		dev_log!("cocoon", "[CocoonService] Registered operation {} for cancellation", request_id);
304		token
305	}
306
307	/// Unregisters an operation after completion
308	///
309	/// # Parameters
310	/// - `request_id`: The request identifier to unregister
311	pub async fn UnregisterOperation(&self, request_id:u64) {
312		self.ActiveOperations.write().await.remove(&request_id);
313		dev_log!("cocoon", "[CocoonService] Unregistered operation {}", request_id);
314	}
315
316	/// Registers a language feature provider in ApplicationState.
317	///
318	/// Converts the gRPC request fields into a `ProviderRegistrationDTO` and
319	/// stores it in `ApplicationState.Extension.ProviderRegistration`.
320	///
321	/// # Parameters
322	/// - `handle`: Unique provider handle
323	/// - `provider_type`: The type of language feature
324	/// - `language_selector`: Language scope (e.g. "typescript")
325	/// - `extension_id`: Extension that registered this provider
326	fn RegisterProvider(&self, handle:u32, provider_type:ProviderType, language_selector:&str, extension_id:&str) {
327		// SideCarIdentifier = "cocoon-main" so FeatureMethods::invoke_provider can
328		// route back via Vine::Client::SendRequestToSideCar("cocoon-main", ...).
329		// Selector stored as array so ProviderLookup::get_matching_provider's
330		// `.as_array()` call finds the language entry: [{ "language": "typescript" }].
331		let dto = ProviderRegistrationDTO {
332			Handle:handle,
333			ProviderType:provider_type,
334			Selector:json!([{ "language": language_selector }]),
335			SideCarIdentifier:"cocoon-main".to_string(),
336			ExtensionIdentifier:json!(extension_id),
337			Options:None,
338		};
339		self.environment
340			.ApplicationState
341			.Extension
342			.ProviderRegistration
343			.RegisterProvider(handle, dto);
344		dev_log!(
345			"cocoon",
346			"[CocoonService] Provider {:?} registered: handle={}, language={}",
347			provider_type,
348			handle,
349			language_selector
350		);
351	}

353	/// Extracts a filesystem path from a URI proto message.
354	///
355	/// Handles both `file://` URIs and bare paths. Returns `None` if the URI
356	/// is absent or the path cannot be extracted.
357	fn UriToPath(uri_opt:Option<&Uri>) -> Option<std::path::PathBuf> {
358		let value = uri_opt?.value.as_str();
359		if value.is_empty() {
360			return None;
361		}
362		// Strip file:// prefix if present
363		let path_str = if let Some(Stripped) = value.strip_prefix("file://") {
364			Stripped
365		} else if value.starts_with('/') || (value.len() > 1 && value.as_bytes()[1] == b':') {
366			// Bare absolute path (Unix or Windows)
367			value
368		} else {
369			// Unknown scheme - return as-is
370			value
371		};
372		Some(std::path::PathBuf::from(path_str))
373	}

379	/// Process Mountain requests from Cocoon (generic request-response).
380	///
381	/// Routes legacy `fs.*` / `commands.*` / `secrets.*` method names used by
382	/// Cocoon's `FileSystemService` and other services that call Mountain via
383	/// the generic `ProcessCocoonRequest` RPC instead of the typed methods.
384	///
385	/// Parameters are JSON-encoded bytes in `request.parameter`. Results are
386	/// JSON-encoded bytes in `response.result`.
387	async fn process_mountain_request(
388		&self,
389		request:Request<GenericRequest>,
390	) -> Result<Response<GenericResponse>, Status> {
391		let Req = request.into_inner();
392		let RequestId = Req.request_identifier;
393
394		dev_log!(
395			"cocoon",
396			"[CocoonService] generic request: method={} id={}",
397			Req.method,
398			RequestId
399		);
400
401		/// Serialise a value into the `result` bytes of a GenericResponse.
402		fn OkResponse(RequestId:u64, Value:&impl serde::Serialize) -> Response<GenericResponse> {
403			let Bytes = serde_json::to_vec(Value).unwrap_or_default();
404			Response::new(GenericResponse { request_identifier:RequestId, result:Bytes, error:None })
405		}
406
407		/// Build an error GenericResponse.
408		fn ErrResponse(RequestId:u64, Code:i32, Message:String) -> Response<GenericResponse> {
409			Response::new(GenericResponse {
410				request_identifier:RequestId,
411				result:Vec::new(),
412				error:Some(RpcError { code:Code, message:Message, data:Vec::new() }),
413			})
414		}
415
416		// Deserialise the generic parameter bytes as a JSON value
417		let Params:serde_json::Value = if Req.parameter.is_empty() {
418			serde_json::Value::Null
419		} else {
420			serde_json::from_slice(&Req.parameter).unwrap_or(serde_json::Value::Null)
421		};
422
423		match Req.method.as_str() {
424			// ---- File System ---- (Cocoon FileSystemService uses these paths)
425			"fs.readFile" | "file:read" => {
426				let Path = Params
427					.as_str()
428					.or_else(|| Params.get("path").and_then(|V| V.as_str()))
429					.unwrap_or("");
430				match tokio::fs::read(Path).await {
431					Ok(Content) => Ok(OkResponse(RequestId, &Content)),
432					Err(Error) => Ok(ErrResponse(RequestId, -32000, format!("fs.readFile: {}", Error))),
433				}
434			},
435			"fs.writeFile" | "file:write" => {
436				let Path = Params.get("path").and_then(|V| V.as_str()).unwrap_or("");
437				let Content:Vec<u8> = Params
438					.get("content")
439					.and_then(|V| V.as_array())
440					.map(|A| A.iter().filter_map(|B| B.as_u64().map(|N| N as u8)).collect())
441					.unwrap_or_default();
442				match tokio::fs::write(Path, &Content).await {
443					Ok(()) => Ok(OkResponse(RequestId, &serde_json::Value::Null)),
444					Err(Error) => Ok(ErrResponse(RequestId, -32000, format!("fs.writeFile: {}", Error))),
445				}
446			},
447			"fs.stat" | "file:stat" => {
448				let Path = Params
449					.as_str()
450					.or_else(|| Params.get("path").and_then(|V| V.as_str()))
451					.unwrap_or("");
452				match tokio::fs::metadata(Path).await {
453					Ok(Meta) => {
454						let Mtime = Meta
455							.modified()
456							.ok()
457							.and_then(|T| T.duration_since(UNIX_EPOCH).ok())
458							.map(|D| D.as_millis() as u64)
459							.unwrap_or(0);
460						Ok(OkResponse(
461							RequestId,
462							&json!({
463								"type": if Meta.is_dir() { 2 } else { 1 },
464								"is_file": Meta.is_file(),
465								"is_directory": Meta.is_dir(),
466								"size": Meta.len(),
467								"mtime": Mtime,
468							}),
469						))
470					},
471					Err(Error) => Ok(ErrResponse(RequestId, -32000, format!("fs.stat: {}", Error))),
472				}
473			},
474			"fs.listDir" | "fs.readdir" | "file:readdir" => {
475				let Path = Params
476					.as_str()
477					.or_else(|| Params.get("path").and_then(|V| V.as_str()))
478					.unwrap_or("");
479				match tokio::fs::read_dir(Path).await {
480					Ok(mut Entries) => {
481						// Return [{name, type}] where type 1=File 2=Directory
482						let mut Items:Vec<serde_json::Value> = Vec::new();
483						while let Ok(Some(Entry)) = Entries.next_entry().await {
484							if let Some(Name) = Entry.file_name().to_str() {
485								let IsDir = Entry.file_type().await.map(|T| T.is_dir()).unwrap_or(false);
486								Items.push(json!({ "name": Name, "type": if IsDir { 2u32 } else { 1u32 } }));
487							}
488						}
489						Ok(OkResponse(RequestId, &Items))
490					},
491					Err(Error) => Ok(ErrResponse(RequestId, -32000, format!("fs.listDir: {}", Error))),
492				}
493			},
494			"fs.createDir" | "file:mkdir" => {
495				let Path = Params
496					.as_str()
497					.or_else(|| Params.get("path").and_then(|V| V.as_str()))
498					.unwrap_or("");
499				match tokio::fs::create_dir_all(Path).await {
500					Ok(()) => Ok(OkResponse(RequestId, &serde_json::Value::Null)),
501					Err(Error) => Ok(ErrResponse(RequestId, -32000, format!("fs.createDir: {}", Error))),
502				}
503			},
504			"fs.delete" | "file:delete" => {
505				let Path = Params
506					.as_str()
507					.or_else(|| Params.get("path").and_then(|V| V.as_str()))
508					.unwrap_or("");
509				let Result = if std::path::Path::new(Path).is_dir() {
510					tokio::fs::remove_dir_all(Path).await
511				} else {
512					tokio::fs::remove_file(Path).await
513				};
514				match Result {
515					Ok(()) => Ok(OkResponse(RequestId, &serde_json::Value::Null)),
516					Err(Error) => Ok(ErrResponse(RequestId, -32000, format!("fs.delete: {}", Error))),
517				}
518			},
519			"fs.rename" | "file:move" => {
520				let From = Params.get("from").and_then(|V| V.as_str()).unwrap_or("");
521				let To = Params.get("to").and_then(|V| V.as_str()).unwrap_or("");
522				match tokio::fs::rename(From, To).await {
523					Ok(()) => Ok(OkResponse(RequestId, &serde_json::Value::Null)),
524					Err(Error) => Ok(ErrResponse(RequestId, -32000, format!("fs.rename: {}", Error))),
525				}
526			},
527			// ---- Commands ----
528			"commands.execute" => {
529				let CommandId = Params.get("id").and_then(|V| V.as_str()).unwrap_or("").to_string();
530				let Arg = Params.get("arg").cloned().unwrap_or(serde_json::Value::Null);
531				match self.environment.ExecuteCommand(CommandId, Arg).await {
532					Ok(Value) => Ok(OkResponse(RequestId, &Value)),
533					Err(Error) => Ok(ErrResponse(RequestId, -32000, Error.to_string())),
534				}
535			},
536			// ---- Commands (Cocoon MountainGRPCClient format) ----
537			"executeCommand" => {
538				let CommandId = Params.get("commandId").and_then(|V| V.as_str()).unwrap_or("").to_string();
539				let Arg = Params
540					.get("arguments")
541					.and_then(|A| A.as_array())
542					.and_then(|A| A.first())
543					.cloned()
544					.unwrap_or(serde_json::Value::Null);
545				match self.environment.ExecuteCommand(CommandId, Arg).await {
546					Ok(Value) => Ok(OkResponse(RequestId, &json!({ "result": Value }))),
547					Err(Error) => Ok(ErrResponse(RequestId, -32000, Error.to_string())),
548				}
549			},

552				let CommandId = Params.get("commandId").and_then(|V| V.as_str()).unwrap_or("").to_string();
553				match self.environment.UnregisterCommand(ExtensionId, CommandId).await {
554					Ok(()) => Ok(OkResponse(RequestId, &json!({ "success": true }))),
555					Err(Error) => Ok(ErrResponse(RequestId, -32000, Error.to_string())),
556				}
557			},

570					Base:CommonLibrary::UserInterface::DTO::DialogOptionsDTO::DialogOptionsDTO {

572						..Default::default()
573					},
574					..OpenDialogOptionsDTO::default()
575				};

577					Ok(Some(Paths)) => {
578						let Uris:Vec<String> = Paths.iter().map(|P| format!("file://{}", P.display())).collect();
579						Ok(OkResponse(RequestId, &json!(Uris)))
580					},
581					Ok(None) => Ok(OkResponse(RequestId, &json!(serde_json::Value::Array(vec![])))),
582					Err(Error) => Ok(ErrResponse(RequestId, -32000, Error.to_string())),
583				}
584			},
585			"UserInterface.ShowSaveDialog" => {
586				use CommonLibrary::UserInterface::{
587					DTO::SaveDialogOptionsDTO::SaveDialogOptionsDTO,
588					UserInterfaceProvider::UserInterfaceProvider,
589				};
590				let Title = Params
591					.get(0)
592					.and_then(|V| V.get("title"))
593					.and_then(|T| T.as_str())
594					.map(|S| S.to_string());
595				let Options = SaveDialogOptionsDTO {

597						Title,

599					},

601				};

603					Ok(Some(Path)) => Ok(OkResponse(RequestId, &json!(format!("file://{}", Path.display())))),
604					Ok(None) => Ok(OkResponse(RequestId, &serde_json::Value::Null)),
605					Err(Error) => Ok(ErrResponse(RequestId, -32000, Error.to_string())),
606				}

612				};

615					Prompt:Opts

617						.and_then(|P| P.as_str())

4// # TreeViewProvider Implementation
5//
6// Implements the `TreeViewProvider` trait for the `MountainEnvironment`.
7// This provider manages the lifecycle of custom tree views and orchestrates the
8// data flow between the extension host (`Cocoon`) and the UI (`Sky`).
9//
10// ## Key Features:
11// - Tree view registration and lifecycle management
12// - Tree data provider dispatching (native/proxied)
13// - Tree state persistence and restoration
14// - Lazy loading and selection handling
15// - Drag and drop support
16// - Badge and message management
17//
18// ## VSCode Reference:
19// - vs/workbench/api/browser/mainThreadTreeViews.ts
20// - vs/workbench/api/common/extHostTreeViews.ts
21//
22// ============================================================================

42	/// Registers a new tree data provider from an extension (e.g., Cocoon).
43	async fn RegisterTreeDataProvider(&self, ViewIdentifier:String, Options:Value) -> Result<(), CommonError> {
44		info!("[TreeViewProvider] Registering data provider for view: {}", ViewIdentifier);
45
46		let OptionsDTO:TreeViewOptionsDTO = serde_json::from_value(Options.clone()).map_err(|Error| {
47			CommonError::InvalidArgument { ArgumentName:"Options".into(), Reason:Error.to_string() }
48		})?;
49
50		// For now, assume all extension providers come from the main sidecar.
51		let SideCarIdentifier = "cocoon-main".to_string();
52
53		let NewState = TreeViewStateDTO {
54			ViewIdentifier:ViewIdentifier.clone(),
55
56			// This is a proxied provider, not native.
57			Provider:None,
58
59			SideCarIdentifier:Some(SideCarIdentifier),
60
61			CanSelectMany:OptionsDTO.CanSelectMany,
62
63			HasHandleDrag:OptionsDTO.HasHandleDrag,
64
65			HasHandleDrop:OptionsDTO.HasHandleDrop,
66
67			Message:None,
68
69			Title:None,
70
71			Description:None,
72		};
73
74		self.ApplicationState
75			.ActiveTreeViews
76			.lock()
77			.map_err(Utility::MapApplicationStateLockErrorToCommonError)?
78			.insert(ViewIdentifier.clone(), NewState);
79
80		self.ApplicationHandle
81			.emit(
82				"sky://tree-view/create",
83				json!({ "ViewIdentifier": ViewIdentifier, "Options": Options }),
84			)
85			.map_err(|Error| CommonError::UserInterfaceInteraction { Reason:Error.to_string() })?;
86
87		Ok(())
88	}

132	/// Gets the children for a given element. This method acts as a dispatcher.
133	async fn GetChildren(
134		&self,
135
136		ViewIdentifier:String,
137
138		ElementHandle:Option<String>,
139	) -> Result<Vec<Value>, CommonError> {
140		let ProviderInfo = self
141			.ApplicationState
142			.ActiveTreeViews
143			.lock()
144			.map_err(Utility::MapApplicationStateLockErrorToCommonError)?
145			.get(&ViewIdentifier)
146			.cloned();
147
148		if let Some(Info) = ProviderInfo {
149			if let Some(NativeProvider) = Info.Provider {
150				// Case 1: Native Rust provider (e.g., File Explorer)
151				return NativeProvider.GetChildren(ViewIdentifier, ElementHandle).await;
152			} else if let Some(SideCarId) = Info.SideCarIdentifier {
153				// Case 2: Proxied extension provider
154				let IPCProvider:Arc<dyn IPCProvider> = self.Require();
155
156				let RPCMethod = format!("{}$getChildren", ProxyTarget::ExtHostTreeView.GetTargetPrefix());
157
158				let RPCParams = json!([ViewIdentifier, ElementHandle]);
159
160				let Response = IPCProvider.SendRequestToSideCar(SideCarId, RPCMethod, RPCParams, 10000).await?;
161
162				return serde_json::from_value::<Vec<Value>>(Response).map_err(CommonError::from);
163			}
164		}
165		Err(CommonError::TreeViewProviderNotFound { ViewIdentifier })
166	}

194	/// Updates the tree view message displayed in the UI.
195	async fn SetTreeViewMessage(&self, ViewIdentifier:String, Message:Option<String>) -> Result<(), CommonError> {
196		info!(
197			"[TreeViewProvider] Setting message for view '{}': {:?}",
198			ViewIdentifier, Message
199		);
200
201		{
202			let mut TreeViewGuard = self
203				.ApplicationState
204				.ActiveTreeViews
205				.lock()
206				.map_err(Utility::MapApplicationStateLockErrorToCommonError)?;
207
208			if let Some(ViewState) = TreeViewGuard.get_mut(&ViewIdentifier) {
209				ViewState.Message = Message.clone();
210			}
211		}
212
213		self.ApplicationHandle
214			.emit(
215				"sky://tree-view/set-message",
216				json!({ "ViewIdentifier": ViewIdentifier, "Message": Message }),
217			)
218			.map_err(|Error| {
219				CommonError::UserInterfaceInteraction { Reason:format!("Failed to emit tree view message: {}", Error) }
220			})
221	}

358	/// Persists the current state of a tree view.
359	/// Saves the expansion, selection, and other state for later restoration.
360	/// Returns JSON representation of the persisted state.
361	async fn PersistTreeViewState(&self, ViewIdentifier:String) -> Result<Value, CommonError> {
362		info!("[TreeViewProvider] Persisting state for view '{}'", ViewIdentifier);
363
364		let TreeViews = self
365			.ApplicationState
366			.ActiveTreeViews
367			.lock()
368			.map_err(Utility::MapApplicationStateLockErrorToCommonError)?;
369
370		let State = TreeViews.get(&ViewIdentifier).map(|View| {
371			json!({
372				"ViewIdentifier": ViewIdentifier,
373				"Title": View.Title,
374				"Description": View.Description,
375				"CanSelectMany": View.CanSelectMany,
376				"Message": View.Message,
377				"HasHandleDrag": View.HasHandleDrag,
378				"HasHandleDrop": View.HasHandleDrop,
379			})
380		});
381
382		State.ok_or(CommonError::TreeViewProviderNotFound { ViewIdentifier })
383	}

385	/// Restores a previously persisted tree view state.
386	/// Restores expansion, selection, and other state from a JSON
387	/// representation.
388	async fn RestoreTreeViewState(&self, ViewIdentifier:String, StateValue:Value) -> Result<(), CommonError> {
389		info!("[TreeViewProvider] Restoring state for view '{}'", ViewIdentifier);
390
391		let mut TreeViews = self
392			.ApplicationState
393			.ActiveTreeViews
394			.lock()
395			.map_err(Utility::MapApplicationStateLockErrorToCommonError)?;
396
397		if let Some(ViewState) = TreeViews.get_mut(&ViewIdentifier) {
398			if let Some(Title) = StateValue.get("Title").and_then(|v| v.as_str()) {
399				ViewState.Title = Some(Title.to_string());
400			}
401			if let Some(Description) = StateValue.get("Description").and_then(|v| v.as_str()) {
402				ViewState.Description = Some(Description.to_string());
403			}
404			// TODO: Restore other state properties as needed
405
406			// Emit to frontend
407			self.ApplicationHandle
408				.emit(
409					"sky://tree-view/restore-state",
410					json!({
411						"ViewIdentifier": ViewIdentifier,
412						"State": StateValue
413					}),
414				)
415				.map_err(|Error| {
416					CommonError::UserInterfaceInteraction {
417						Reason:format!("Failed to emit restore state event: {}", Error),
418					}
419				})?;
420
421			Ok(())
422		} else {
423			Err(CommonError::TreeViewProviderNotFound { ViewIdentifier })
424		}

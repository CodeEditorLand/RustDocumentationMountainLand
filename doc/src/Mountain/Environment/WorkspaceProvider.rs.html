1//! # WorkspaceProvider (Environment)
2//!
3//! RESPONSIBILITIES:
4//! - Implements
5//!   [`WorkspaceProvider`](CommonLibrary::Workspace::WorkspaceProvider) and
6//!   [`WorkspaceEditApplier`](CommonLibrary::Workspace::WorkspaceEditApplier)
7//!   traits for [`MountainEnvironment`]
8//! - Manages multi-root workspace folder operations and configuration
9//! - Provides workspace trust management and file discovery capabilities
10//! - Handles workspace edit application and custom editor routing
11//!
12//! ARCHITECTURAL ROLE:
13//! - Core provider in the Environment system, exposing workspace-level
14//!   functionality to frontend via gRPC through the
15//!   [`AirService`](crate::Air::AirServiceProvider)
16//! - Workspace provider is one of the foundational services alongside Document,
17//!   Configuration, and Diagnostic providers
18//! - Integrates with
19//!   [`ApplicationState`](crate::ApplicationState::ApplicationState) for
20//!   persistent workspace folder storage
21//!
22//! ERROR HANDLING:
23//! - Uses [`CommonError`](CommonLibrary::Error::CommonError) for all operations
24//! - Application state lock errors are mapped using
25//!   [`Utility::MapApplicationStateLockErrorToCommonError`]
26//! - Some operations are stubbed with logging (FindFilesInWorkspace, OpenFile,
27//!   ApplyWorkspaceEdit)
28//!
29//! PERFORMANCE:
30//! - Workspace folder lookup uses O(n) linear search through folder list
31//! - Lock contention on `ApplicationState.Workspace.WorkspaceFolders` should be
32//!   minimized
33//! - File discovery and workspace edit application are not yet optimized
34//!
35//! VS CODE REFERENCE:
36//! - `vs/workbench/services/workspace/browser/workspaceService.ts` - workspace
37//!   service implementation
38//! - `vs/workbench/contrib/files/common/editors/textFileEditor.ts` - file
39//!   editor integration
40//! - `vs/platform/workspace/common/workspace.ts` - workspace types and
41//!   interfaces
42//!
43//! TODO:
44//! - Implement actual file search with glob pattern matching
45//! - Implement file opening with workspace-relative paths
46//! - Complete workspace edit application logic
47//! - Add workspace event propagation to subscribers
48//! - Implement custom editor routing by view type
49//!
50//! MODULE CONTENTS:
51//! - [`WorkspaceProvider`](CommonLibrary::Workspace::WorkspaceProvider)
52//!   implementation:
53//!   - [`GetWorkspaceFoldersInfo`](Self::GetWorkspaceFoldersInfo) - enumerate
54//!     all workspace folders
55//!   - [`GetWorkspaceFolderInfo`](Self::GetWorkspaceFolderInfo) - find folder
56//!     containing a URI
57//!   - [`GetWorkspaceName`](Self::GetWorkspaceName) - workspace identifier from
58//!     state
59//!   - [`GetWorkspaceConfigurationPath`](Self::GetWorkspaceConfigurationPath) -
60//!     .code-workspace path
61//!   - [`IsWorkspaceTrusted`](Self::IsWorkspaceTrusted) - trust status check
62//!   - [`RequestWorkspaceTrust`](Self::RequestWorkspaceTrust) - trust
63//!     acquisition (stub)
64//!   - [`FindFilesInWorkspace`](Self::FindFilesInWorkspace) - file discovery
65//!     (stub)
66//!   - [`OpenFile`](Self::OpenFile) - file opening (stub)
67//! - [`WorkspaceEditApplier`](CommonLibrary::Workspace::WorkspaceEditApplier)
68//!   implementation:
69//!   - [`ApplyWorkspaceEdit`](Self::ApplyWorkspaceEdit) - edit application
70//!     (stub)
71//! - Data types: [`(Url, String, usize)`] tuple for folder info (URI, name,
72//!   index)

88#[async_trait]
89impl WorkspaceProvider for MountainEnvironment {
90	/// Retrieves information about all currently open workspace folders.
91	async fn GetWorkspaceFoldersInfo(&self) -> Result<Vec<(Url, String, usize)>, CommonError> {
92		info!("[WorkspaceProvider] Getting workspace folders info.");
93		let FoldersGuard = self
94			.ApplicationState
95			.Workspace
96			.WorkspaceFolders
97			.lock()
98			.map_err(Utility::MapApplicationStateLockErrorToCommonError)?;
99		Ok(FoldersGuard.iter().map(|f| (f.URI.clone(), f.Name.clone(), f.Index)).collect())
100	}
101
102	/// Retrieves information for the specific workspace folder that contains a
103	/// given URI.
104	async fn GetWorkspaceFolderInfo(&self, URIToMatch:Url) -> Result<Option<(Url, String, usize)>, CommonError> {
105		let FoldersGuard = self
106			.ApplicationState
107			.Workspace
108			.WorkspaceFolders
109			.lock()
110			.map_err(Utility::MapApplicationStateLockErrorToCommonError)?;
111		for Folder in FoldersGuard.iter() {
112			if URIToMatch.as_str().starts_with(Folder.URI.as_str()) {
113				return Ok(Some((Folder.URI.clone(), Folder.Name.clone(), Folder.Index)));
114			}
115		}
116
117		Ok(None)
118	}
119
120	/// Gets the name of the current workspace.
121	async fn GetWorkspaceName(&self) -> Result<Option<String>, CommonError> {
122		self.ApplicationState.GetWorkspaceIdentifier().map(Some)
123	}
124
125	/// Gets the path to the workspace configuration file (`.code-workspace`).
126	async fn GetWorkspaceConfigurationPath(&self) -> Result<Option<PathBuf>, CommonError> {
127		Ok(self
128			.ApplicationState
129			.Workspace
130			.WorkspaceConfigurationPath
131			.lock()
132			.map_err(Utility::MapApplicationStateLockErrorToCommonError)?
133			.clone())
134	}
135
136	/// Checks if the current workspace is trusted.
137	async fn IsWorkspaceTrusted(&self) -> Result<bool, CommonError> {
138		Ok(self
139			.ApplicationState
140			.Workspace
141			.IsTrusted
142			.load(std::sync::atomic::Ordering::Relaxed))
143	}
144
145	/// Requests workspace trust from the user.
146	async fn RequestWorkspaceTrust(&self, _Options:Option<Value>) -> Result<bool, CommonError> {
147		warn!("[WorkspaceProvider] RequestWorkspaceTrust is not implemented; defaulting to trusted.");
148		Ok(true)
149	}
150
151	/// Finds files in the workspace matching the specified query.
152	async fn FindFilesInWorkspace(
153		&self,
154		_query:Value,
155		_:Option<Value>,
156		_:Option<usize>,
157		_:bool,
158		_:bool,
159	) -> Result<Vec<Url>, CommonError> {
160		info!("[WorkspaceProvider] FindFilesInWorkspace called");
161		// Scan all workspace folders to find files matching the query pattern. This
162		// integrates with FileSystemReader to traverse directories, apply glob and
163		// exclude patterns, and return matching file URIs. Respect query parameters
164		// including maxResults, excludePatterns, and .gitignore rules. The result
165		// set supports fuzzy search, symbol search, and quick file open features.
166		// Currently returns an empty result set.
167		Ok(Vec::new())
168	}
169
170	/// Opens a file in the workspace.
171	async fn OpenFile(&self, path:PathBuf) -> Result<(), CommonError> {
172		info!("[WorkspaceProvider] OpenFile called for: {:?}", path);
173		// Open a file in the editor by delegating to the Workbench or command system.
174		// Resolves the path relative to workspace roots, handles URI schemes (file://,
175		// untitled:), and triggers the 'workbench.action.files.open' command or
176		// equivalent. Creates a new document tab with the file contents, activating
177		// the editor and adding the file to the recently opened list. Currently a
178		// no-op.
179		Ok(())
180	}
181}

185	/// Applies a workspace edit to the workspace.
186	async fn ApplyWorkspaceEdit(&self, Edit:WorkspaceEditDTO) -> Result<bool, CommonError> {
187		info!("[WorkspaceEditApplier] Applying workspace edit");
188
189		// For now, just log the edit details
190		match Edit {
191			WorkspaceEditDTO { Edits } => {
192				for (DocumentURI, TextEdits) in Edits {
193					info!(
194						"[WorkspaceEditApplier] Would apply {} edits to document: {}",
195						TextEdits.len(),
196						DocumentURI
197					);
198				}
199			},
200		}
201
202		// Apply a collection of document edits and file operations to the workspace.
203		// Parses the WorkspaceEditDTO and performs text edits on documents, creates
204		// and deletes files, and handles renames with proper validation. Key aspects:
205		// validate document URIs and workspace trust, apply text edits with coordinate
206		// conversion (line/column), handle all operations atomically with rollback on
207		// failure, emit before/after events for extension observability, and return
208		// false if any edit fails with detailed error information. This enables
209		// multi-file refactorings, code actions, and automated fixes.
210		warn!("[WorkspaceEditApplier] ApplyWorkspaceEdit is not fully implemented");
211
212		Ok(true)
213	}

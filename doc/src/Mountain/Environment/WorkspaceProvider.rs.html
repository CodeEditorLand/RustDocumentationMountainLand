1//! # WorkspaceProvider (Environment)
2//!
3//! RESPONSIBILITIES:
4//! - Implements [`WorkspaceProvider`](CommonLibrary::Workspace::WorkspaceProvider) and
5//!   [`WorkspaceEditApplier`](CommonLibrary::Workspace::WorkspaceEditApplier) traits for [`MountainEnvironment`]
6//! - Manages multi-root workspace folder operations and configuration
7//! - Provides workspace trust management and file discovery capabilities
8//! - Handles workspace edit application and custom editor routing
9//!
10//! ARCHITECTURAL ROLE:
11//! - Core provider in the Environment system, exposing workspace-level functionality
12//!   to frontend via gRPC through the [`AirService`](crate::Air::AirServiceProvider)
13//! - Workspace provider is one of the foundational services alongside Document,
14//!   Configuration, and Diagnostic providers
15//! - Integrates with [`ApplicationState`](crate::ApplicationState::ApplicationState)
16//!   for persistent workspace folder storage
17//!
18//! ERROR HANDLING:
19//! - Uses [`CommonError`](CommonLibrary::Error::CommonError) for all operations
20//! - Application state lock errors are mapped using [`Utility::MapApplicationStateLockErrorToCommonError`]
21//! - Some operations are stubbed with logging (FindFilesInWorkspace, OpenFile, ApplyWorkspaceEdit)
22//!
23//! PERFORMANCE:
24//! - Workspace folder lookup uses O(n) linear search through folder list
25//! - Lock contention on `ApplicationState.WorkspaceFolders` should be minimized
26//! - File discovery and workspace edit application are not yet optimized
27//!
28//! VS CODE REFERENCE:
29//! - `vs/workbench/services/workspace/browser/workspaceService.ts` - workspace service implementation
30//! - `vs/workbench/contrib/files/common/editors/textFileEditor.ts` - file editor integration
31//! - `vs/platform/workspace/common/workspace.ts` - workspace types and interfaces
32//!
33//! TODO:
34//! - Implement actual file search with glob pattern matching
35//! - Implement file opening with workspace-relative paths
36//! - Complete workspace edit application logic
37//! - Add workspace event propagation to subscribers
38//! - Implement custom editor routing by view type
39//!
40//! MODULE CONTENTS:
41//! - [`WorkspaceProvider`](CommonLibrary::Workspace::WorkspaceProvider) implementation:
42//!   - [`GetWorkspaceFoldersInfo`](Self::GetWorkspaceFoldersInfo) - enumerate all workspace folders
43//!   - [`GetWorkspaceFolderInfo`](Self::GetWorkspaceFolderInfo) - find folder containing a URI
44//!   - [`GetWorkspaceName`](Self::GetWorkspaceName) - workspace identifier from state
45//!   - [`GetWorkspaceConfigurationPath`](Self::GetWorkspaceConfigurationPath) - .code-workspace path
46//!   - [`IsWorkspaceTrusted`](Self::IsWorkspaceTrusted) - trust status check
47//!   - [`RequestWorkspaceTrust`](Self::RequestWorkspaceTrust) - trust acquisition (stub)
48//!   - [`FindFilesInWorkspace`](Self::FindFilesInWorkspace) - file discovery (stub)
49//!   - [`OpenFile`](Self::OpenFile) - file opening (stub)
50//! - [`WorkspaceEditApplier`](CommonLibrary::Workspace::WorkspaceEditApplier) implementation:
51//!   - [`ApplyWorkspaceEdit`](Self::ApplyWorkspaceEdit) - edit application (stub)
52//! - Data types: [`(Url, String, usize)`] tuple for folder info (URI, name, index)

72#[async_trait]
73impl WorkspaceProvider for MountainEnvironment {
74	/// Retrieves information about all currently open workspace folders.
75	async fn GetWorkspaceFoldersInfo(&self) -> Result<Vec<(Url, String, usize)>, CommonError> {
76		info!("[WorkspaceProvider] Getting workspace folders info.");
77		let FoldersGuard = self
78			.ApplicationState
79			.WorkspaceFolders
80			.lock()
81			.map_err(Utility::MapApplicationStateLockErrorToCommonError)?;
82		Ok(FoldersGuard.iter().map(|f| (f.URI.clone(), f.Name.clone(), f.Index)).collect())
83	}
84
85	/// Retrieves information for the specific workspace folder that contains a
86	/// given URI.
87	async fn GetWorkspaceFolderInfo(&self, URIToMatch:Url) -> Result<Option<(Url, String, usize)>, CommonError> {
88		let FoldersGuard = self
89			.ApplicationState
90			.WorkspaceFolders
91			.lock()
92			.map_err(Utility::MapApplicationStateLockErrorToCommonError)?;
93		for Folder in FoldersGuard.iter() {
94			if URIToMatch.as_str().starts_with(Folder.URI.as_str()) {
95				return Ok(Some((Folder.URI.clone(), Folder.Name.clone(), Folder.Index)));
96			}
97		}
98
99		Ok(None)
100	}
101
102	/// Gets the name of the current workspace.
103	async fn GetWorkspaceName(&self) -> Result<Option<String>, CommonError> {
104		self.ApplicationState.GetWorkspaceIdentifier().map(Some)
105	}
106
107	/// Gets the path to the workspace configuration file (`.code-workspace`).
108	async fn GetWorkspaceConfigurationPath(&self) -> Result<Option<PathBuf>, CommonError> {
109		Ok(self
110			.ApplicationState
111			.WorkspaceConfigurationPath
112			.lock()
113			.map_err(Utility::MapApplicationStateLockErrorToCommonError)?
114			.clone())
115	}
116
117	/// Checks if the current workspace is trusted.
118	async fn IsWorkspaceTrusted(&self) -> Result<bool, CommonError> {
119		Ok(self.ApplicationState.IsTrusted.load(std::sync::atomic::Ordering::Relaxed))
120	}
121
122	/// Requests workspace trust from the user.
123	async fn RequestWorkspaceTrust(&self, _Options:Option<Value>) -> Result<bool, CommonError> {
124		warn!("[WorkspaceProvider] RequestWorkspaceTrust is not implemented; defaulting to trusted.");
125		Ok(true)
126	}
127
128	/// Finds files in the workspace matching the specified query.
129	async fn FindFilesInWorkspace(

138		// TODO: Implement file search
139		Ok(Vec::new())
140	}

144		info!("[WorkspaceProvider] OpenFile called for: {:?}", path);
145		// TODO: Implement file opening

147	}
148}

1//! # TestProvider (Environment)
2//!
3//! RESPONSIBILITIES:
4//! - Implements [`TestController`](CommonLibrary::Testing::TestController) for
5//!   [`MountainEnvironment`]
6//! - Manages test discovery, execution, and result reporting
7//! - Handles test controller registration and lifecycle
8//! - Tracks test run progress and aggregates results
9//! - Provides sidecar proxy for extension-provided test frameworks
10//!
11//! ARCHITECTURAL ROLE:
12//! - Environment provider for testing functionality
13//! - Uses controller pattern: each extension can register its own test
14//!   controller
15//! - Controllers identified by unique `ControllerIdentifier` and scoped to
16//!   extensions
17//! - Integrates with [`IPCProvider`](CommonLibrary::IPC::IPCProvider) for RPC
18//!   to test runners
19//! - Stores controller state in
20//!   `ApplicationState`
21//!
22//! TEST EXECUTION FLOW:
23//! 1. Extension registers test controller via `RegisterTestController`
24//! 2. Mountain calls `ResolveTests` to discover tests (async, emits
25//!    `TestItemAdded`)
26//! 3. Extension returns test tree structure with IDs, labels, children
27//! 4. UI requests to run tests via `RunTest` or `RunTests` with optional
28//!    `RunProfile`
29//! 5. Mountain forwards to extension's controller via RPC
30//! 6. Extension executes tests and reports progress via `TestRunStarted`,
31//!    `TestItemStarted`, `TestItemPassed`, `TestItemFailed`, `TestRunEnded`
32//!    events
33//! 7. Mountain aggregates results and emits to UI
34//!
35//! ERROR HANDLING:
36//! - Uses [`CommonError`](CommonLibrary::Error::CommonError) for all operations
37//! - Validates controller identifiers and run profile names (non-empty)
38//! - Controller state tracked in RwLock for thread-safe mutation
39//! - Unknown controller ID errors return `TestControllerNotFound`
40//! - Duplicate registration returns `InvalidArgument` error
41//!
42//! PERFORMANCE:
43//! - Test discovery is async and can be cancelled (drop sender)
44//! - Test run progress is streamed via events, avoiding blocking
45//! - Controller state uses RwLock for concurrent read access during test runs
46//! - TODO: Consider test result caching for quick re-runs
47//!
48//! VS CODE REFERENCE:
49//! - `vs/workbench/contrib/testing/common/testService.ts` - test service
50//!   architecture
51//! - `vs/workbench/contrib/testing/common/testController.ts` - test controller
52//!   interface
53//! - `vs/workbench/contrib/testing/common/testTypes.ts` - test data models
54//! - `vs/workbench/contrib/testing/browser/testingView.ts` - testing UI panel
55//!
56//! TODO:
57//! - Implement test run cancellation and timeout handling
58//! - Add test result caching and quick re-run support
59//! - Implement test tree filtering and search
60//! - Add test run configuration persistence
61//! - Support test run peeking (inline results in editor)
62//! - Implement test coverage integration
63//! - Add test run duration metrics and profiling
64//! - Support parallel test execution (multiple workers)
65//! - Implement test run retry logic for flaky tests
66//! - Add test run export (JUnit, TAP, custom formats)
67//! - Integrate with CodeLens for in-source test run buttons
68//!
69//! MODULE CONTENTS:
70//! - [`TestController`](CommonLibrary::Testing::TestController) implementation:
71//! - `RegisterTestController` - register extension's controller
72//! - `UnregisterTestController` - remove controller
73//! - `ResolveTests` - discover tests (async with cancellation)
74//! - `RunTest` - run single test by ID
75//! - `RunTests` - run multiple tests (by ID or all in parent)
76//! - `StopTestRun` - cancel ongoing test run
77//! - `DidTestItemDiscoveryStart` - discovery progress event
78//!   - `TestRunStarted`/`TestItemStarted`/`TestItemPassed`/`TestItemFailed`/
79//!     `TestRunEnded` - events
80//! - Data types: `TestControllerState`, `TestItemState`, `TestRunProfile`,
81//!   `TestResultState`

172#[async_trait]
173impl TestController for MountainEnvironment {
174	/// Registers a new test controller from an extension (e.g., Cocoon).
175	///
176	/// This method creates a TestControllerState entry and notifies the
177	/// frontend about the available test controller.
178	async fn RegisterTestController(&self, ControllerId:String, Label:String) -> Result<(), CommonError> {
179		info!(
180			"[TestProvider] Registering test controller '{}' with label '{}'",
181			ControllerId, Label
182		);
183
184		// For now, assume all extension providers come from the main sidecar
185		let SideCarIdentifier = Some("cocoon-main".to_string());
186
187		let ControllerState = TestControllerState {
188			ControllerIdentifier:ControllerId.clone(),
189
190			Label,
191
192			SideCarIdentifier,
193
194			IsActive:true,
195
196			SupportedTestTypes:vec!["unit".to_string(), "integration".to_string()],
197		};
198
199		// Store the controller state
200		let mut StateGuard = self.ApplicationState.TestProviderState.write().await;
201
202		StateGuard.Controllers.insert(ControllerId.clone(), ControllerState);
203
204		drop(StateGuard);
205
206		// Notify the frontend about the new test controller
207		self.ApplicationHandle
208			.emit("sky://test/registered", json!({ "ControllerIdentifier": ControllerId }))
209			.map_err(|Error| {
210				CommonError::IPCError { Description:format!("Failed to emit test registration event: {}", Error) }
211			})?;
212
213		debug!("[TestProvider] Test controller '{}' registered successfully", ControllerId);
214
215		Ok(())
216	}
217
218	/// Runs tests based on the test run request.
219	///
220	/// This implementation supports both native (Rust) and proxied (extension)
221	/// test controllers, with proper test discovery, execution, and result
222	/// reporting.
223	async fn RunTests(&self, ControllerIdentifier:String, TestRunRequest:Value) -> Result<(), CommonError> {
224		info!(
225			"[TestProvider] Running tests for controller '{}': {:?}",
226			ControllerIdentifier, TestRunRequest
227		);
228
229		// Get controller state
230		let ControllerState = {
231			let StateGuard = self.ApplicationState.TestProviderState.read().await;
232
233			StateGuard.Controllers.get(&ControllerIdentifier).cloned().ok_or_else(|| {
234				CommonError::TestControllerNotFound { ControllerIdentifier:ControllerIdentifier.clone() }
235			})?
236		};
237
238		// Create a new test run
239		let RunIdentifier = Uuid::new_v4().to_string();
240
241		let TestRun = TestRun {
242			RunIdentifier:RunIdentifier.clone(),
243
244			ControllerIdentifier:ControllerIdentifier.clone(),
245
246			Status:TestRunStatus::Queued,
247
248			StartedAt:std::time::Instant::now(),
249
250			Results:HashMap::new(),
251		};
252
253		{
254			let mut StateGuard = self.ApplicationState.TestProviderState.write().await;
255
256			StateGuard.ActiveRuns.insert(RunIdentifier.clone(), TestRun);
257		}
258
259		// Notify frontend about test run start
260		self.ApplicationHandle
261			.emit(
262				"sky://test/run-started",
263				json!({ "RunIdentifier": RunIdentifier, "ControllerIdentifier": ControllerIdentifier }),
264			)
265			.map_err(|Error| {
266				CommonError::IPCError { Description:format!("Failed to emit test run started event: {}", Error) }
267			})?;
268
269		// Execute tests based on controller type
270		if let Some(SideCarIdentifier) = &ControllerState.SideCarIdentifier {
271			// Proxied extension test controller
272			Self::RunProxiedTests(self, SideCarIdentifier, &RunIdentifier, TestRunRequest).await?;
273		} else {
274			// Native Rust test controller (currently not supported)
275			warn!(
276				"[TestProvider] Native test controllers not yet implemented for '{}'",
277				ControllerIdentifier
278			);
279
280			Self::UpdateRunStatus(self, &RunIdentifier, TestRunStatus::Skipped).await;
281		}
282
283		Ok(())
284	}
285}

291impl MountainEnvironment {
292	/// Runs tests via a proxied sidecar test controller.
293	async fn RunProxiedTests(
294		&self,
295
296		SideCarIdentifier:&str,
297
298		RunIdentifier:&str,
299
300		TestRunRequest:Value,
301	) -> Result<(), CommonError> {
302		info!(
303			"[TestProvider] Running proxied tests for run '{}' on sidecar '{}'",
304			RunIdentifier, SideCarIdentifier
305		);
306
307		// Update test run status to running
308		Self::UpdateRunStatus(self, RunIdentifier, TestRunStatus::Running).await;
309
310		let IPCProvider:Arc<dyn IPCProvider> = self.Require();
311
312		let RPCMethod = format!("{}$runTests", ProxyTarget::ExtHostTesting.GetTargetPrefix());
313
314		let RPCParams = json!({
315			"RunIdentifier": RunIdentifier,
316
317			"TestRunRequest": TestRunRequest,
318
319		});
320
321		match IPCProvider
322			.SendRequestToSideCar(SideCarIdentifier.to_string(), RPCMethod, RPCParams, 300000)
323			.await
324		{
325			Ok(Response) => {
326				// Parse test results from response
327				if let Ok(Results) = serde_json::from_value::<Vec<TestResult>>(Response) {
328					Self::StoreTestResults(self, RunIdentifier, Results).await;
329
330					// Determine final status based on results
331					let FinalStatus = Self::CalculateRunStatus(self, RunIdentifier).await;
332
333					Self::UpdateRunStatus(self, RunIdentifier, FinalStatus).await;
334
335					info!(
336						"[TestProvider] Test run '{}' completed with status {:?}",
337						RunIdentifier, FinalStatus
338					);
339				} else {
340					error!("[TestProvider] Failed to parse test results for run '{}'", RunIdentifier);
341
342					Self::UpdateRunStatus(self, RunIdentifier, TestRunStatus::Errored).await;
343				}
344				Ok(())
345			},
346
347			Err(Error) => {
348				error!("[TestProvider] Failed to run tests: {}", Error);
349
350				let _ = Self::UpdateRunStatus(self, RunIdentifier, TestRunStatus::Errored).await;
351
352				Err(Error)
353			},
354		}
355	}

357	/// Updates the status of a test run and notifies the frontend.
358	async fn UpdateRunStatus(&self, RunIdentifier:&str, Status:TestRunStatus) -> Result<(), CommonError> {
359		let mut StateGuard = self.ApplicationState.TestProviderState.write().await;
360
361		if let Some(TestRun) = StateGuard.ActiveRuns.get_mut(RunIdentifier) {
362			TestRun.Status = Status;
363
364			drop(StateGuard);
365
366			// Notify frontend about status change
367			self.ApplicationHandle
368				.emit(
369					"sky://test/run-status-changed",
370					json!({
371						"RunIdentifier": RunIdentifier,
372
373						"Status": Status,
374
375					}),
376				)
377				.map_err(|Error| {
378					CommonError::IPCError { Description:format!("Failed to emit test status change event: {}", Error) }
379				})?;
380
381			Ok(())
382		} else {
383			Err(CommonError::TestRunNotFound { RunIdentifier:RunIdentifier.to_string() })
384		}
385	}
386
387	/// Stores test results for a test run.
388	async fn StoreTestResults(&self, RunIdentifier:&str, Results:Vec<TestResult>) -> Result<(), CommonError> {
389		let mut StateGuard = self.ApplicationState.TestProviderState.write().await;
390
391		if let Some(TestRun) = StateGuard.ActiveRuns.get_mut(RunIdentifier) {
392			for Result in Results {
393				TestRun.Results.insert(Result.TestIdentifier.clone(), Result);
394			}
395			Ok(())
396		} else {
397			Err(CommonError::TestRunNotFound { RunIdentifier:RunIdentifier.to_string() })
398		}
399	}
400
401	/// Calculates the final status of a test run based on its results.
402	async fn CalculateRunStatus(&self, RunIdentifier:&str) -> TestRunStatus {
403		let StateGuard = self.ApplicationState.TestProviderState.read().await;
404
405		if let Some(TestRun) = StateGuard.ActiveRuns.get(RunIdentifier) {
406			if TestRun.Results.is_empty() {
407				TestRunStatus::Passed // No tests considered passed
408			} else {
409				let HasFailed = TestRun.Results.values().any(|r| r.Status == TestRunStatus::Failed);
410
411				let HasErrored = TestRun.Results.values().any(|r| r.Status == TestRunStatus::Errored);
412
413				if HasErrored {
414					TestRunStatus::Errored
415				} else if HasFailed {
416					TestRunStatus::Failed
417				} else {
418					TestRunStatus::Passed
419				}
420			}
421		} else {
422			TestRunStatus::Errored
423		}
424	}

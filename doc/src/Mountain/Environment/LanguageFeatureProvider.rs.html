1//! # LanguageFeatureProvider (Environment)
2//!
3//! RESPONSIBILITIES:
4//! - Implements [`LanguageFeatureProviderRegistry`](CommonLibrary::LanguageFeature::LanguageFeatureProviderRegistry)
5//!   for [`MountainEnvironment`]
6//! - Central hub for all language intelligence features (completion, hover, definition, etc.)
7//! - Routes language feature requests to appropriate extension providers in Cocoon
8//! - Manages provider registration, scoring, and selection based on document selectors
9//! - Handles JSON-RPC to LSP (Language Server Protocol) conversions
10//! - Supports all major LSP features: hover, completion, definition, references,
11//!   formatting, code actions, rename, semantic tokens, inlay hints, folding ranges
12//!
13//! ARCHITECTURAL ROLE:
14//! - Core provider enabling language intelligence across the editor
15//! - Uses provider registry pattern with numeric handles for efficient lookup
16//! - Each extension can register multiple providers for different file types
17//! - Integrates with [`IPCProvider`](CommonLibrary::IPC::IPCProvider) for RPC to Cocoon
18//! - Provider metadata stored in [`ApplicationState`](crate::ApplicationState::ApplicationState)
19//! - TODO: Consider fallback chain: Air (cached) → Cocoon (LSP) → Local (basic)
20//!
21//! PROVIDER SCORING:
22//! - Providers are scored based on selector specificity (language, scheme, pattern)
23//! - Higher scores win; ties broken by registration order
24//! - Selector matching uses language identifier, URI scheme, and glob patterns
25//! - TODO: Implement proper scoring algorithm with specificity weights
26//!
27//! ERROR HANDLING:
28//! - Uses [`CommonError`](CommonLibrary::Error::CommonError) for all operations
29//! - Validates provider type and selector presence during registration
30//! - Unknown provider handle errors return `TestControllerNotFound` (should be specialized)
31//! - Feature requests failing to find provider return `ProviderNotFound` errors
32//!
33//! PERFORMANCE:
34//! - Provider lookup should be O(1) via handle hash map in ApplicationState
35//! - Feature requests are async and non-blocking
36//! - TODO: Add provider caching for frequently used features
37//! - TODO: Implement request cancellation via CancellationToken for long-running operations
38//!
39//! VS CODE REFERENCE:
40//! - `vs/workbench/contrib/editor/browser/editor.contribution.ts` - language feature registration
41//! - `vs/workbench/services/languageserver/common/languageServer.ts` - LSP lifecycle
42//! - `vs/workbench/contrib/quickfix/browser/quickfix.ts` - code actions
43//! - `vs/workbench/contrib/hover/browser/hover.ts` - hover provider
44//! - `vs/workbench/contrib/completion/browser/completion.ts` - completion provider
45//!
46//! TODO:
47//! - Implement language server lifecycle management (start/stop/restart)
48//! - Add provider fallback chain: Air (cached) → Cocoon (LSP) → Local (basic)
49//! - Implement provider priority scoring based on selector specificity
50//! - Add scheme and pattern matching in document selectors
51//! - Add exclude pattern support
52//! - Handle LSP capability negotiation
53//! - Implement request cancellation through CancellationToken
54//! - Add performance metrics and logging for feature invocation
55//! - Support incremental synchronization for large files
56//! - Add semantic tokens support with proper token modifiers
57//! - Implement call hierarchy provider (calls to and from)
58//! - Add inlay hints support (parameter, type, inline variable hints)
59//! - Implement type hierarchy provider
60//! - Add workspace symbol support
61//! - Support linked editing range (rename multiple occurrences)
62//!
63//! MODULE CONTENTS:
64//! - [`LanguageFeatureProviderRegistry`](CommonLibrary::LanguageFeature::LanguageFeatureProviderRegistry) implementation:
65//!   - [`RegisterProvider`](Self::RegisterProvider) - register extension provider with selector
66//!   - [`UnregisterProvider`](Self::UnregisterProvider) - remove provider by handle
67//!   - [`GetMatchingProvider`](Self::GetMatchingProvider) - score and select provider for document
68//!   - LSP feature methods: [`Hover`](Self::Hover), [`Completion`](Self::Completion),
69//!     [`Definition`](Self::Definition), [`References`](Self::References),
70//!     [`Formatting`](Self::Formatting), [`CodeActions`](Self::CodeActions),
71//!     [`Rename`](Self::Rename), [`SemanticTokens`](Self::SemanticTokens), etc.
72//! - Data types: `ProviderRegistrationDTO`, `ProviderType`, `HoverResultDTO`,
73//!   `CompletionListDTO`, `CompletionContextDTO`, `LocationDTO`, `PositionDTO`

105	async fn RegisterProvider(
106		&self,
107
108		SideCarIdentifier:String,
109
110		ProviderType:ProviderType,
111
112		SelectorDTO:Value,
113
114		ExtensionIdentifierDTO:Value,
115
116		OptionsDTO:Option<Value>,
117	) -> Result<u32, CommonError> {
118		let Handle = self.ApplicationState.GetNextProviderHandle();
119
120		info!(
121			"[LangFeatureProvider] Registering {:?} provider from '{}' with new handle {}",
122			ProviderType, SideCarIdentifier, Handle
123		);
124
125		let NewRegistration = ProviderRegistrationDTO {
126			Handle,
127
128			ProviderType,
129
130			Selector:SelectorDTO,
131
132			SideCarIdentifier,
133
134			Options:OptionsDTO,
135
136			ExtensionIdentifier:ExtensionIdentifierDTO,
137		};
138
139		self.ApplicationState
140			.LanguageProviders
141			.lock()
142			.map_err(Utility::MapApplicationStateLockErrorToCommonError)?
143			.insert(Handle, NewRegistration);
144
145		Ok(Handle)
146	}
147
148	async fn UnregisterProvider(&self, Handle:u32) -> Result<(), CommonError> {
149		info!("[LangFeatureProvider] Unregistering provider with handle {}", Handle);
150
151		if self
152			.ApplicationState
153			.LanguageProviders
154			.lock()
155			.map_err(Utility::MapApplicationStateLockErrorToCommonError)?
156			.remove(&Handle)
157			.is_none()
158		{
159			warn!(
160				"[LangFeatureProvider] Attempted to unregister a provider with handle {} that was not found.",
161				Handle
162			);
163		}
164		Ok(())
165	}

293/// Finds the best provider for a given feature and document.
294fn FindBestProvider(
295	Environment:&MountainEnvironment,
296
297	ProviderType:ProviderType,
298
299	DocumentURI:&Url,
300) -> Result<Option<ProviderRegistrationDTO>, CommonError> {
301	let Providers = Environment
302		.ApplicationState
303		.LanguageProviders
304		.lock()
305		.map_err(Utility::MapApplicationStateLockErrorToCommonError)?;
306
307	let Document = Environment
308		.ApplicationState
309		.OpenDocuments
310		.lock()
311		.map_err(Utility::MapApplicationStateLockErrorToCommonError)?
312		.get(DocumentURI.as_str())
313		.cloned();
314
315	if let Some(doc) = Document {
316		// This is a simplified selector matching logic. A real implementation would
317		// score providers based on how well their DocumentSelector matches the
318		// document.
319		for Provider in Providers.values() {
320			if Provider.ProviderType == ProviderType {
321				if let Some(SelectorArray) = Provider.Selector.as_array() {
322					for Selector in SelectorArray {
323						if let Some(Lang) = Selector.get("language").and_then(Value::as_str) {
324							if Lang == doc.LanguageIdentifier {
325								debug!("Found provider with handle {} for document {}", Provider.Handle, DocumentURI);
326
327								return Ok(Some(Provider.clone()));
328							}
329						}
330						// TODO: Add scheme and pattern matching logic here.
331						// Current implementation only matches language
332						// identifier. Should also check:
333						// - Selector["scheme"] (e.g., "file", "untitled",
334						//   "custom")
335						// - Selector["pattern"] (e.g., "**/*.ts",
336						//   "src/**/*.rs")
337						// - Selector["exclude"] (e.g., "node_modules/**")
338						// Provider scoring should rank by specificity (pattern
339						// > language > all)
340					}
341				}
342			}
343		}
344	}
345	warn!("No provider found for {:?} on document {}", ProviderType, DocumentURI);
346
347	Ok(None)
348}

350/// A generic helper to find the best provider, invoke it via RPC, and
351/// deserialize the result.
352async fn InvokeProvider<TResponse:DeserializeOwned>(
353	Environment:&MountainEnvironment,
354
355	ProviderType:ProviderType,
356
357	DocumentURI:&Url,
358
359	mut ProviderArguments:Value,
360) -> Result<Option<TResponse>, CommonError> {
361	if let Some(Provider) = FindBestProvider(Environment, ProviderType, DocumentURI)? {
362		let RPCMethod = format!("$provide{}", Provider.ProviderType.to_string());
363
364		let URIComponents = json!({ "external": DocumentURI.to_string(), "$mid": 1 });
365
366		let ArgumentsVector = ProviderArguments.as_array_mut().ok_or_else(|| {
367			CommonError::InvalidArgument {
368				ArgumentName:"ProviderArguments".into(),
369
370				Reason:"Expected provider arguments to be a JSON array.".into(),
371			}
372		})?;
373
374		let mut FinalArgumentsVector = vec![json!(Provider.Handle), URIComponents];
375
376		FinalArgumentsVector.append(ArgumentsVector);
377
378		let FinalArguments = json!(FinalArgumentsVector);
379
380		let IPCProvider:Arc<dyn IPCProvider> = Environment.Require();
381
382		let Response = IPCProvider
383			.SendRequestToSideCar(Provider.SideCarIdentifier, RPCMethod, FinalArguments, 5000)
384			.await?;
385
386		if Response.is_null() {
387			return Ok(None);
388		}
389		serde_json::from_value(Response).map_err(|Error| {
390			CommonError::SerializationError {
391				Description:format!("Failed to deserialize response for {:?}: {}", ProviderType, Error),
392			}
393		})
394	} else {
395		Ok(None)
396	}

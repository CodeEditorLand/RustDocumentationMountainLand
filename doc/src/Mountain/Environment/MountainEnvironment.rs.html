3// # Architectural Role: Central DI Container and Application Context
4//
5// MountainEnvironment is the primary dependency injection (DI) container for
6// the Mountain application. It implements all provider traits defined in the
7// Common crate, acting as the central orchestrator that provides access to all
8// platform services.
9//
10// # Responsibilities
11//
12// 1. **Dependency Injection Container**: Implements Requires<T> for all 19+
13//    provider traits, enabling other components to request dependencies through
14//    the Require() method.
15//
16// 2. **Application Lifecycle Management**: Holds references to Tauri AppHandle
17//    and ApplicationState, managing the core application context and state.
18//
19// 3. **Air Integration**: Optionally manages the Air gRPC client for
20//    cloud-based services when the AirIntegration feature is enabled. Enables
21//    dynamic switching between local and cloud services.
22//
23// 4. **Extension Management**: Implements ExtensionManagementService for
24//    discovering, scanning, and managing extensions in the system.
25//
26// 5. **Service Orchestration**: Acts as the central coordinator between all
27//    providers (FileSystem, Document, Command, Configuration, IPC, etc.),
28//    ensuring proper initialization and interaction.
29//
30// # Initialization Sequence
31//
32// 1. Create MountainEnvironment instance via Create() or CreateWithAir()
33// 2. Provider instances are created lazily through Requires<T> traits
34// 3. Each provider can access ApplicationState and AppHandle through self
35// 4. Inter-provider communication is handled via IPCProvider or direct Rust
36//    calls
37//
38// # Dependency Wiring
39//
40// All providers implement their respective traits from the Common crate.
41// MountainEnvironment implements Requires<T> for each trait, returning an
42// Arc-wrapped clone of itself. This enables circular dependencies and lazy
43// initialization while maintaining type safety.
44//
45// # Patterns Borrowed from VSCode
46//
47// - **ServiceCollection Pattern**: Like VSCode's ServiceCollection,
48//   MountainEnvironment registers and provides all services in a centralized
49//   location.
50//
51// - **Lifecycle Management**: Similar to VSCode's IDisposable pattern,
52//   resources are automatically managed through Arc reference counting.
53//
54// - **Extension Points**: Extension management follows VSCode's activation
55//   event pattern, enabling lazy loading of extension services.
56//

127	#[allow(unused_mut)]
128	pub fn Create(ApplicationHandle:AppHandle<Wry>) -> Self {
129		info!("[MountainEnvironment] New instance created.");
130
131		let ApplicationState = ApplicationHandle.state::<Arc<ApplicationState>>().inner().clone();
132
133		#[cfg(feature = "AirIntegration")]
134		{
135			Self { ApplicationHandle, ApplicationState, AirClient:None }

138		#[cfg(not(feature = "AirIntegration"))]
139		{
140			Self { ApplicationHandle, ApplicationState }

171	/// Returns whether Air is available and ready.
172	#[cfg(feature = "AirIntegration")]
173	pub async fn IsAirAvailable(&self) -> bool {
174		if let Some(AirClient) = &self.AirClient {
175			use tonic::Request;
176			use AirLibrary::Vine::Generated::Air::HealthCheckRequest;
177
178			match AirClient.health_check(Request::new(HealthCheckRequest {})).await {
179				Ok(response) => {
180					let is_healthy = response.into_inner().healthy;
181
182					if !is_healthy {
183						warn!("[MountainEnvironment] Air health check returned unhealthy");
184					}
185
186					is_healthy
187				},
188				Err(error) => {
189					warn!("[MountainEnvironment] Air health check failed: {}", error);
190					false
191				},
192			}
193		} else {
194			info!("[MountainEnvironment] No Air client configured");
195			false
196		}
197	}

203	/// Scans a directory for extensions and returns their package.json data
204	async fn ScanExtensionDirectory(&self, path:&std::path::PathBuf) -> Result<Vec<serde_json::Value>, CommonError> {
205		use std::fs;
206
207		use serde_json::Value;
208
209		let mut extensions = Vec::new();
210
211		// Check if directory exists
212		if !path.exists() || !path.is_dir() {
213			warn!("[ExtensionManagementService] Extension directory does not exist: {:?}", path);
214			return Ok(extensions);
215		}
216
217		// Read directory contents
218		let entries = fs::read_dir(path).map_err(|error| {
219			CommonError::FileSystemIO {
220				Path:path.clone(),
221				Description:format!("Failed to read extension directory: {}", error),
222			}
223		})?;
224
225		for entry in entries {
226			let entry = entry.map_err(|error| {
227				CommonError::FileSystemIO {
228					Path:path.clone(),
229					Description:format!("Failed to read directory entry: {}", error),
230				}
231			})?;
232
233			let entry_path = entry.path();
234			if entry_path.is_dir() {
235				// Look for package.json in the extension directory
236				let package_json_path = entry_path.join("package.json");
237				if package_json_path.exists() {
238					match fs::read_to_string(&package_json_path) {
239						Ok(content) => {
240							match serde_json::from_str::<Value>(&content) {
241								Ok(mut package_json) => {
242									// Add extension location information
243									if let Some(obj) = package_json.as_object_mut() {
244										obj.insert(
245											"ExtensionLocation".to_string(),
246											Value::String(entry_path.to_string_lossy().to_string()),
247										);
248									}
249									extensions.push(package_json);
250									info!("[ExtensionManagementService] Found extension at: {:?}", entry_path);
251								},
252								Err(error) => {
253									warn!(
254										"[ExtensionManagementService] Failed to parse package.json at {:?}: {}",
255										package_json_path, error
256									);
257								},
258							}
259						},
260						Err(error) => {
261							warn!(
262								"[ExtensionManagementService] Failed to read package.json at {:?}: {}",
263								package_json_path, error
264							);
265						},
266					}
267				}
268			}
269		}
270
271		Ok(extensions)
272	}

277#[async_trait]
278impl ExtensionManagementService for MountainEnvironment {
279	async fn ScanForExtensions(&self) -> Result<(), CommonError> {
280		info!("[ExtensionManagementService] Scanning for extensions...");
281
282		// Get the extension scan paths from ApplicationState
283		let ScanPaths:Vec<std::path::PathBuf> = {
284			let ScanPathsGuard = self
285				.ApplicationState
286				.ExtensionScanPaths
287				.lock()
288				.map_err(|Error| CommonError::StateLockPoisoned { Context:Error.to_string() })?;
289			ScanPathsGuard.clone()
290		};
291
292		let mut extensions = Vec::new();
293
294		// Scan each extension directory
295		for path in ScanPaths {
296			if let Ok(mut scan_result) = self.ScanExtensionDirectory(&path).await {
297				extensions.append(&mut scan_result);
298			}
299		}
300
301		// Update ApplicationState with scanned extensions
302		let mut ScannedExtensionsGuard = self
303			.ApplicationState
304			.ScannedExtensions
305			.lock()
306			.map_err(|Error| CommonError::StateLockPoisoned { Context:Error.to_string() })?;
307
308		ScannedExtensionsGuard.clear();
309
310		for extension in extensions {
311			if let Some(identifier) = extension.get("Identifier").and_then(|v| v.as_str()) {
312				// Convert the extension DTO to ExtensionDescriptionStateDTO
313				let extension_dto = ExtensionDescriptionStateDTO {
314					Identifier:serde_json::Value::String(identifier.to_string()),
315					Name:extension.get("Name").and_then(|v| v.as_str()).unwrap_or("Unknown").to_string(),
316					Version:extension.get("Version").and_then(|v| v.as_str()).unwrap_or("0.0.0").to_string(),
317					Publisher:extension
318						.get("Publisher")
319						.and_then(|v| v.as_str())
320						.unwrap_or("Unknown")
321						.to_string(),
322					Engines:extension.get("Engines").cloned().unwrap_or(serde_json::Value::Null),
323					Main:extension.get("Main").and_then(|v| v.as_str()).map(|s| s.to_string()),
324					Browser:extension.get("Browser").and_then(|v| v.as_str()).map(|s| s.to_string()),
325					ModuleType:extension.get("ModuleType").and_then(|v| v.as_str()).map(|s| s.to_string()),
326					IsBuiltin:extension.get("IsBuiltin").and_then(|v| v.as_bool()).unwrap_or(false),
327					IsUnderDevelopment:extension.get("IsUnderDevelopment").and_then(|v| v.as_bool()).unwrap_or(false),
328					ExtensionLocation:extension.get("ExtensionLocation").cloned().unwrap_or(serde_json::Value::Null),
329					ActivationEvents:extension
330						.get("ActivationEvents")
331						.and_then(|v| v.as_array())
332						.map(|arr| arr.iter().filter_map(|v| v.as_str().map(|s| s.to_string())).collect()),
333					Contributes:extension.get("Contributes").cloned(),
334				};
335
336				ScannedExtensionsGuard.insert(identifier.to_string(), extension_dto);
337			}
338		}
339
340		info!("[ExtensionManagementService] Found {} extensions", ScannedExtensionsGuard.len());
341		Ok(())
342	}
343
344	async fn GetExtensions(&self) -> Result<Vec<Value>, CommonError> {
345		let ScannedExtensionsGuard = self
346			.ApplicationState
347			.ScannedExtensions
348			.lock()
349			.map_err(|Error| CommonError::StateLockPoisoned { Context:Error.to_string() })?;
350
351		let Extensions:Vec<Value> = ScannedExtensionsGuard
352			.values()
353			.map(|ext| serde_json::to_value(ext).unwrap_or(Value::Null))
354			.collect();
355
356		Ok(Extensions)
357	}
358
359	async fn GetExtension(&self, id:String) -> Result<Option<Value>, CommonError> {
360		let ScannedExtensionsGuard = self
361			.ApplicationState
362			.ScannedExtensions
363			.lock()
364			.map_err(|Error| CommonError::StateLockPoisoned { Context:Error.to_string() })?;
365
366		if let Some(extension_dto) = ScannedExtensionsGuard.get(&id) {
367			// Convert ExtensionDescriptionStateDTO back to JSON Value
368			let mut extension_value = serde_json::Map::new();
369			extension_value.insert("Identifier".to_string(), extension_dto.Identifier.clone());
370			extension_value.insert("Name".to_string(), Value::String(extension_dto.Name.clone()));
371			extension_value.insert("Version".to_string(), Value::String(extension_dto.Version.clone()));
372			extension_value.insert("Publisher".to_string(), Value::String(extension_dto.Publisher.clone()));
373			extension_value.insert("Engines".to_string(), extension_dto.Engines.clone());
374
375			if let Some(main) = &extension_dto.Main {
376				extension_value.insert("Main".to_string(), Value::String(main.clone()));
377			}
378
379			if let Some(browser) = &extension_dto.Browser {
380				extension_value.insert("Browser".to_string(), Value::String(browser.clone()));
381			}
382
383			if let Some(module_type) = &extension_dto.ModuleType {
384				extension_value.insert("ModuleType".to_string(), Value::String(module_type.clone()));
385			}
386
387			extension_value.insert("IsBuiltin".to_string(), Value::Bool(extension_dto.IsBuiltin));
388			extension_value.insert("IsUnderDevelopment".to_string(), Value::Bool(extension_dto.IsUnderDevelopment));
389			extension_value.insert("ExtensionLocation".to_string(), extension_dto.ExtensionLocation.clone());
390
391			if let Some(activation_events) = &extension_dto.ActivationEvents {
392				let events:Vec<Value> = activation_events.iter().map(|e| Value::String(e.clone())).collect();
393				extension_value.insert("ActivationEvents".to_string(), Value::Array(events));
394			}
395
396			if let Some(contributes) = &extension_dto.Contributes {
397				extension_value.insert("Contributes".to_string(), contributes.clone());
398			}
399
400			Ok(Some(Value::Object(extension_value)))
401		} else {
402			Ok(None)
403		}
404	}
405}

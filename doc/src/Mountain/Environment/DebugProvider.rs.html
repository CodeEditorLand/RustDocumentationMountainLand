1//! # DebugProvider (Environment)
2//!
3//! RESPONSIBILITIES:
4//! - Implements [`DebugService`](CommonLibrary::Debug::DebugService) for [`MountainEnvironment`]
5//! - Manages complete debugging session lifecycle from configuration to termination
6//! - Orchestrates between extension host (Cocoon), debug adapter, and UI
7//! - Handles DAP (Debug Adapter Protocol) message mediation
8//!
9//! ARCHITECTURAL ROLE:
10//! - Core provider for debugging functionality, analogous to VSCode's debug service
11//! - Uses two-stage registration: configuration providers and adapter descriptor factories
12//! - Each debug type (node, java, rust) can have its own configuration and adapter
13//! - Integrates with [`IPCProvider`](CommonLibrary::IPC::IPCProvider) for RPC to Cocoon
14//!
15//! DEBUG SESSION FLOW:
16//! 1. UI calls `StartDebugging` with folder URI and configuration
17//! 2. Mountain RPCs to Cocoon to resolve debug configuration (variable substitution)
18//! 3. Mountain RPCs to Cocoon to create debug adapter descriptor
19//! 4. Mountain spawns debug adapter process or connects to TCP server
20//! 5. Mountain mediates DAP messages between UI and debug adapter
21//! 6. UI sends DAP commands via `SendCommand` which forwards to adapter
22//! 7. Debug adapter sends DAP events/notifications back through Mountain to UI
23//! 8. Session ends on stop request or adapter process exit
24//!
25//! ERROR HANDLING:
26//! - Uses [`CommonError`](CommonLibrary::Error::CommonError) for all operations
27//! - Validates debug type is non-empty (InvalidArgument error)
28//! - TODO: Implement proper session lookup, timeout handling, and error recovery
29//!
30//! PERFORMANCE:
31//! - Debug adapter spawning should be async with timeout protection (5000ms in current RPC)
32//! - DAP message routing needs efficient session lookup (TODO: O(1) hash map)
33//! - Multiple simultaneous debug sessions require careful resource management
34//!
35//! VS CODE REFERENCE:
36//! - `vs/workbench/contrib/debug/browser/debugService.ts` - debug service main logic
37//! - `vs/workbench/contrib/debug/common/debug.ts` - debug interfaces and models
38//! - `vs/workbench/contrib/debug/browser/adapter/descriptorFactory.ts` - adapter descriptor factories
39//! - `vs/debugAdapter/common/debugProtocol.ts` - DAP protocol specification
40//!
41//! TODO:
42//! - Store debug adapter registrations in ApplicationState
43//! - Implement proper debug session tracking and management
44//! - Add debug adapter process spawning and lifecycle management
45//! - Implement proper DAP message routing and serialization
46//! - Add debug session state persistence across UI reloads
47//! - Implement debug console and variable inspection integration
48//! - Add support for multiple simultaneous debug sessions
49//! - Implement debug adapter termination and cleanup
50//! - Add debug session metrics and telemetry
51//! - Consider implementing debug configuration validation
52//! - Add support for debug adapters that communicate via TCP sockets
53//! - Implement debug adapter crash detection and recovery
54//!
55//! MODULE CONTENTS:
56//! - [`DebugService`](CommonLibrary::Debug::DebugService) implementation:
57//!   - [`RegisterDebugConfigurationProvider`](Self::RegisterDebugConfigurationProvider) - register config resolver
58//!   - [`RegisterDebugAdapterDescriptorFactory`](Self::RegisterDebugAdapterDescriptorFactory) - register adapter factory
59//!   - [`StartDebugging`](Self::StartDebugging) - start debug session (partial)
60//!   - [`SendCommand`](Self::SendCommand) - send DAP command to adapter (stub)

138	async fn StartDebugging(&self, FolderURI:Option<Url>, Configuration:Value) -> Result<String, CommonError> {
139		let SessionID = uuid::Uuid::new_v4().to_string();
140		info!(
141			"[DebugProvider] Starting debug session '{}' with config: {:?}",
142			SessionID, Configuration
143		);
144
145		let IPCProvider:Arc<dyn IPCProvider> = self.Require();
146		let DebugType = Configuration
147			.get("type")
148			.and_then(Value::as_str)
149			.ok_or_else(|| {
150				CommonError::InvalidArgument {
151					ArgumentName:"Configuration".into(),
152
153					Reason:"Missing 'type' field in debug configuration.".into(),
154				}
155			})?
156			.to_string();
157
158		// TODO: Look up which sidecar handles this debug type
159		// For now, assume the main sidecar handles all debugging.
160		let TargetSideCar = "cocoon-main".to_string();
161
162		// 1. Resolve configuration (Reverse-RPC to Cocoon)
163		info!("[DebugProvider] Resolving debug configuration for type '{}'", DebugType);
164		info!("[DebugProvider] Resolving debug configuration...");
165		let ResolveConfigMethod = format!("{}$resolveDebugConfiguration", ProxyTarget::ExtHostDebug.GetTargetPrefix());
166		let ResolvedConfig = IPCProvider
167			.SendRequestToSideCar(
168				TargetSideCar.clone(),
169				ResolveConfigMethod,
170				json!([DebugType.clone(), Configuration]),
171				5000,
172			)
173			.await?;
174
175		// 2. Get the Debug Adapter Descriptor (Reverse-RPC to Cocoon)
176		info!("[DebugProvider] Creating debug adapter descriptor...");
177		let CreateDescriptorMethod =
178			format!("{}$createDebugAdapterDescriptor", ProxyTarget::ExtHostDebug.GetTargetPrefix());
179		let Descriptor = IPCProvider
180			.SendRequestToSideCar(
181				TargetSideCar.clone(),
182				CreateDescriptorMethod,
183				json!([DebugType, &ResolvedConfig]),
184				5000,
185			)
186			.await?;
187
188		// 3. Spawn the Debug Adapter process based on the descriptor.
189		info!("[DebugProvider] Spawning Debug Adapter based on descriptor: {:?}", Descriptor);
190
191		// TODO: A full implementation would:
192		// - Parse the descriptor (executable path, command args, environment, or server
193		//   port)

195		// - Create a new DebugSession struct to manage the DAP communication stream
196		// - Establish JSON-RPC communication with the debug adapter
197		// - Store the session in ApplicationState with session_id as key
198		// - Implement proper session cleanup on termination

200		info!("[DebugProvider] Debug session '{}' started (simulation).", SessionID);

202	}

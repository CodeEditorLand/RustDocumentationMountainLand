1//! # DocumentProvider (Environment)
2//!
3//! Implements the `DocumentProvider` trait, managing the complete lifecycle of
4//! document operations including opening, saving, editing, and closing. It
5//! maintains document state, coordinates between the frontend (Sky), extension
6//! host (Cocoon), and filesystem, and handles both native file URIs and custom
7//! scheme URIs.
8//!
9//! ## RESPONSIBILITIES
10//!
11//! ### 1. Document State Management
12//! - Maintain all open documents in `ApplicationState.OpenDocuments`
13//! - Track document content, version, dirty status, and metadata
14//! - Support multiple documents open simultaneously
15//! - Handle document focus and activation
16//!
17//! ### 2. Document Persistence
18//! - Save documents to the filesystem (`Save`, `SaveAs`)
19//! - Handle file I/O errors gracefully
20//! - Support dirty state tracking and prompt on close
21//! - Implement atomic saves to prevent data loss
22//!
23//! ### 3. Change Tracking
24//! - Apply incremental text edits via LSP `DidChangeTextDocument`
25//! - Track version identifiers for synchronization
26//! - Maintain line count cache for performance
27//! - Support undo/redo integration (via version history)
28//!
29//! ### 4. URI Scheme Support
30//! - Native `file://` URIs: Read/write from filesystem
31//! - Custom schemes (untitled:, git:, vscode-vfs:): Via `TextDocumentContentProvider`
32//! - Extension-hosted documents: Request content from Cocoon
33//!
34//! ### 5. Event Orchestration
35//! - Emit events to Sky and Cocoon for document lifecycle:
36//!   - `onDidOpenDocument`: Document opened
37//!   - `onDidChangeDocument`: Content modified
38//!   - `onDidSaveDocument`: Document saved
39//!   - `onDidCloseDocument`: Document closed
40//!   - `onDidRenameDocument`: Document moved/renamed
41//!
42//! ### 6. Bulk Operations
43//! - `Save All`: Save all dirty documents efficiently
44//! - Batch close operations
45//! - Concurrent document operations with proper locking
46//!
47//! ## ARCHITECTURAL ROLE
48//!
49//! DocumentProvider is the **document lifecycle manager**:
50//!
51//! ```text
52//! Frontend ──► Open/Edit/Save ──► DocumentProvider ──► FileSystem / Cocoon
53//!               │                                   │
54//!               └─► Event Emit ──► Sky/Cocoon ◄─────┘
55//! ```
56//!
57//! ### Position in Mountain
58//! - `Environment` module: Document management capability
59//! - Implements `CommonLibrary::Document::DocumentProvider` trait
60//! - Accessible via `Environment.Require<dyn DocumentProvider>()`
61//!
62//! ### Document State Model
63//!
64//! Stored in `ApplicationState.OpenDocuments` as `HashMap<String, DocumentStateDTO>`:
65//! - `URI`: Document identifier
66//! - `LanguageIdentifier`: Syntax highlighting language
67//! - `Lines`: Vector of document lines (for efficient editing)
68//! - `Version`: Monotonic version for LSP sync
69//! - `IsDirty`: Unsaved changes flag
70//! - `Encoding`: Text encoding (usually UTF-8)
71//! - `EOL`: End-of-line sequence (LF or CRLF)
72//!
73//! ### Dependencies
74//! - `ApplicationState`: Document state storage
75//! - `FileSystemReader` / `FileSystemWriter`: File I/O
76//! - `IPCProvider`: For extension document providers
77//! - `Log`: Document change logging
78//!
79//! ### Dependents
80//! - Text editors: Load and modify document content
81//! - Language servers: Receive document change notifications
82//! - `Binary::Main`::`MountainGetWorkbenchConfiguration`: Initial document setup
83//! - Extension providers: Custom document handling
84//!
85//! ## DOCUMENT OPEN FLOW
86//!
87//! 1. Frontend calls `OpenDocument(uri, options)`
88//! 2. Provider checks if already open → return existing
89//! 3. If `file://` URI: `FileSystemReader::ReadFile` → content
90//! 4. If custom URI: `IPCProvider::SendRequest` to extension
91//! 5. Create `DocumentStateDTO` with content
92//! 6. Store in `ApplicationState.OpenDocuments`
93//! 7. Emit `onDidOpenDocument` event to Sky/Cocoon
94//! 8. Return document state to frontend
95//!
96//! ## DOCUMENT CHANGE FLOW (LSP-style)
97//!
98//! 1. Frontend sends `ApplyChanges(uri, version, changes)`
99//! 2. Provider looks up document in `OpenDocuments`
100//! 3. Apply delta changes or full text replacement
101//! 4. Increment `Version` and `VersionIdentifier`
102//! 5. Set `IsDirty = true`
103//! 6. Emit `onDidChangeDocument` event
104//! 7. Return updated document state
105//!
106//! ## DOCUMENT SAVE FLOW
107//!
108//! 1. Frontend calls `SaveDocument(uri)` or `SaveAll`
109//! 2. For each dirty document:
110//!    - Serialize document content from `Lines`
111//!    - Write to filesystem via `FileSystemWriter::WriteFile`
112//!    - Clear `IsDirty` flag
113//!    - Emit `onDidSaveDocument` event
114//! 3. Return save results (success/failure per document)
115//!
116//! ## ERROR HANDLING
117//!
118//! - Document not found: `CommonError::FileSystemNotFound`
119//! - Permission denied: `CommonError::FileSystemIO`
120//! - Already open: Return existing (not an error)
121//! - Unsaved changes: Prompt user or force close (configurable)
122//! - Extension provider failure: `CommonError::IPCError`
123//!
124//! ## PERFORMANCE
125//!
126//! - Document state stored in memory (fast access)
127//! - Large files may impact memory (consider streaming for >10MB)
128//! - Line-based storage enables efficient edits
129//! - Version numbers prevent race conditions in collaborative editing
130//!
131//! ## VS CODE REFERENCE
132//!
133//! Patterns from VS Code:
134//! - `vs/workbench/api/common/extHostDocumentProvider.ts` - Document provider interface
135//! - `vs/workbench/services/files/common/files.ts` - File document provider
136//! - `vs/workbench/services/textService/common/textService.ts` - Text document management
137//!

159//
160// 1. **Document State Management**: Maintains all open documents in
161//    ApplicationState, tracking content, version, dirty status, and metadata.
162//
163// 2. **Document Persistence**: Handles saving documents to the filesystem,
164//    including Save and Save As operations.
165//
166// 3. **Change Tracking**: Applies incremental text edits to documents, tracking
167//    version identifiers for collaboration and undo/redo.
168//
169// 4. **URI Scheme Support**: Supports both native file:// URIs and custom
170//    scheme URIs (e.g., untitled:, git:, vscode-vfs:) via
171//    TextDocumentContentProvider.
172//
173// 5. **Event Orchestration**: Emits events to Sky and Cocoon for document
174//    lifecycle changes (Opened, Saved, Changed, Closed, Renamed).
175//
176// 6. **Bulk Operations**: Supports Save All to handle multiple dirty documents
177//    efficiently.
178//
179// # Document State Model
180//
181// Each open document is represented in ApplicationState.OpenDocuments with:
182// - URI: Uniform Resource Identifier (file:// or custom scheme)
183// - LanguageIdentifier: Language ID (e.g., rust, typescript, markdown)
184// - TextContent: Full document text
185// - Version: Monotonically increasing version number (for LSP synchronization)
186// - LineCount: Cached line count for performance
187// - ChangeCount: Number of incremental changes applied
188// - IsDirty: Whether document has unsaved changes
189//
190// # Document Open Flow
191//
192// 1. Frontend requests to open a URI with optional content
193// 2. Mountain checks if document is already open - if yes, refocus existing
194//    model
195// 3. If file:// URI: read content from filesystem via FileSystemReader
196// 4. If custom URI: request content from extension's
197//    TextDocumentContentProvider via IPC
198// 5. Create DocumentStateDTO and store in ApplicationState.OpenDocuments
199// 6. Detect language from file extension or explicit parameter
200// 7. Emit "sky://documents/open" event to frontend
201// 8. Send $acceptModelAdded notification to Cocoon
202//
203// # Document Edit Flow
204//
205// 1. Frontend applies text edits with new version number
206// 2. Mountain applies changes to DocumentStateDTO text content
207// 3. Update version, line count, change count, and dirty status
208// 4. Send $acceptModelChanged notification to Cocoon
209// 5. Cocoon forwards to language servers for analysis
210//
211// # Document Save Flow
212//
213// 1. Frontend requests to save a URI
214// 2. Mountain retrieves document text content
215// 3. Write to filesystem via FileSystemWriter
216// 4. Clear dirty flag
217// 5. Emit "sky://documents/saved" event to frontend
218// 6. Send $acceptModelSaved notification to Cocoon
219// 7. Cocoon notifies language servers to update diagnostics
220//
221// # Patterns Borrowed from VSCode
222//
223// - **Text Model Service**: Inspired by VSCode's ITextModelService for managing
224//   text document instances with lifecycle and change events.
225//
226// - **TextDocumentContentProvider**: Like VSCode's pattern for custom URI
227//   schemes, allows extensions to provide document content dynamically.
228//
229// - **Synchronized Versions**: Mimics VSCode's versioning for seamless LSP
230//   synchronization with Language Servers.
231//
232// - **Change Events**: Emits granular change notifications like VSCode's
233//   onDidChangeTextDocument event.
234//

277	/// Opens a document. If the URI scheme is not native (`file`), it attempts
278	/// to resolve the content from a registered sidecar provider
279	/// (`TextDocumentContentProvider`).
280	async fn OpenDocument(
281		&self,
282
283		URIComponentsDTO:Value,
284
285		LanguageIdentifier:Option<String>,
286
287		Content:Option<String>,
288	) -> Result<Url, CommonError> {
289		let URI = Utility::GetURLFromURIComponentsDTO(&URIComponentsDTO)?;
290
291		info!("[DocumentProvider] Opening document: {}", URI);
292
293		// First, check if the document is already open.
294		if let Some(ExistingDocument) = self
295			.ApplicationState
296			.OpenDocuments
297			.lock()
298			.map_err(Utility::MapApplicationStateLockErrorToCommonError)?
299			.get(URI.as_str())
300		{
301			info!("[DocumentProvider] Document {} is already open.", URI);
302
303			match ExistingDocument.ToDTO() {
304				Ok(DTO) => {
305					if let Err(Error) = self.ApplicationHandle.emit("sky://documents/open", DTO) {
306						error!("[DocumentProvider] Failed to emit document open event: {}", Error);
307					}
308				},
309
310				Err(Error) => {
311					error!("[DocumentProvider] Failed to serialize existing document DTO: {}", Error);
312				},
313			}
314
315			return Ok(ExistingDocument.URI.clone());
316		}
317
318		// Resolve the content based on the URI scheme.
319		let FileContent = if let Some(c) = Content {
320			c
321		} else if URI.scheme() == "file" {
322			let FilePath = URI.to_file_path().map_err(|_| {
323				CommonError::InvalidArgument {
324					ArgumentName:"URI".into(),
325
326					Reason:"Cannot convert non-file URI to path".into(),
327				}
328			})?;
329
330			let RunTime = self.ApplicationHandle.state::<Arc<ApplicationRunTime>>().inner().clone();
331
332			let FileContentBytes = RunTime.Run(ReadFile(FilePath.clone())).await?;
333
334			String::from_utf8(FileContentBytes)
335				.map_err(|Error| CommonError::FileSystemIO { Path:FilePath, Description:Error.to_string() })?
336		} else {
337			// Custom scheme: attempt to resolve from a sidecar provider.
338			info!(
339				"[DocumentProvider] Non-native scheme '{}'. Attempting to resolve from sidecar.",
340				URI.scheme()
341			);
342
343			let IPCProvider:Arc<dyn IPCProvider> = self.Require();
344
345			let RpcResult = IPCProvider
346				.SendRequestToSideCar(
347					// In a multi-host world, we'd look this up
348					"cocoon-main".to_string(),
349					"$provideTextDocumentContent".to_string(),
350					json!([URIComponentsDTO]),
351					10000,
352				)
353				.await?;
354
355			RpcResult.as_str().map(String::from).ok_or_else(|| {
356				CommonError::IPCError {
357					Description:format!("Failed to get valid string content for custom URI scheme '{}'", URI.scheme()),
358				}
359			})?
360		};
361
362		// The rest of the flow is the same for all schemes.
363		let NewDocument = DocumentStateDTO::Create(URI.clone(), LanguageIdentifier, FileContent)?;
364
365		let DTOForNotification = NewDocument.ToDTO()?;
366
367		self.ApplicationState
368			.OpenDocuments
369			.lock()
370			.map_err(Utility::MapApplicationStateLockErrorToCommonError)?
371			.insert(URI.to_string(), NewDocument);
372
373		if let Err(Error) = self.ApplicationHandle.emit("sky://documents/open", DTOForNotification.clone()) {
374			error!("[DocumentProvider] Failed to emit document open event: {}", Error);
375		}
376
377		NotifyModelAdded(self, &DTOForNotification).await;
378
379		Ok(URI)
380	}

382	/// Saves the document at the given URI.
383	async fn SaveDocument(&self, URI:Url) -> Result<bool, CommonError> {
384		info!("[DocumentProvider] Saving document: {}", URI);
385
386		let (ContentBytes, FilePath) = {
387			let mut OpenDocumentsGuard = self
388				.ApplicationState
389				.OpenDocuments
390				.lock()
391				.map_err(Utility::MapApplicationStateLockErrorToCommonError)?;
392
393			if let Some(Document) = OpenDocumentsGuard.get_mut(URI.as_str()) {
394				// For non-file URIs, use temporary file location
395				if URI.scheme() != "file" {
396					info!("[DocumentProvider] Saving non-file URI '{}' to temporary location", URI);
397				}
398
399				Document.IsDirty = false;
400
401				(
402					Document.GetText().into_bytes(),
403					URI.to_file_path().map_err(|_| {
404						CommonError::InvalidArgument {
405							ArgumentName:"URI".into(),
406
407							Reason:"Cannot convert file URI to path".into(),
408						}
409					})?,
410				)
411			} else {
412				return Err(CommonError::FileSystemNotFound(URI.to_file_path().unwrap_or_default()));
413			}
414		};
415
416		let RunTime = self.ApplicationHandle.state::<Arc<ApplicationRunTime>>().inner().clone();
417
418		RunTime.Run(WriteFileBytes(FilePath, ContentBytes, true, true)).await?;
419
420		if let Err(Error) = self
421			.ApplicationHandle
422			.emit("sky://documents/saved", json!({ "uri": URI.to_string() }))
423		{
424			error!("[DocumentProvider] Failed to emit document saved event: {}", Error);
425		}
426
427		NotifyModelSaved(self, &URI).await;
428
429		Ok(true)
430	}

432	/// Saves a document to a new location.
433	async fn SaveDocumentAs(&self, OriginalURI:Url, NewTargetURI:Option<Url>) -> Result<Option<Url>, CommonError> {
434		info!("[DocumentProvider] Saving document as: {}", OriginalURI);
435
436		let RunTime = self.ApplicationHandle.state::<Arc<ApplicationRunTime>>().inner().clone();
437
438		let NewFilePath = match NewTargetURI {
439			Some(uri) => uri.to_file_path().ok(),
440
441			None => RunTime.Run(ShowSaveDialog(Some(SaveDialogOptionsDTO::default()))).await?,
442		};
443
444		let Some(NewPath) = NewFilePath else { return Ok(None) };
445
446		let NewURI = Url::from_file_path(&NewPath).map_err(|_| {
447			CommonError::InvalidArgument {
448				ArgumentName:"NewPath".into(),
449
450				Reason:"Could not convert new path to URI".into(),
451			}
452		})?;
453
454		let OriginalContent = {
455			let Guard = self
456				.ApplicationState
457				.OpenDocuments
458				.lock()
459				.map_err(Utility::MapApplicationStateLockErrorToCommonError)?;
460
461			Guard
462				.get(OriginalURI.as_str())
463				.map(|doc| doc.GetText())
464				.ok_or_else(|| CommonError::FileSystemNotFound(PathBuf::from(OriginalURI.path())))?
465		};
466
467		RunTime
468			.Run(WriteFileBytes(NewPath, OriginalContent.clone().into_bytes(), true, true))
469			.await?;
470
471		let NewDocumentState = {
472			let mut Guard = self
473				.ApplicationState
474				.OpenDocuments
475				.lock()
476				.map_err(Utility::MapApplicationStateLockErrorToCommonError)?;
477
478			let OldDocument = Guard.remove(OriginalURI.as_str());
479
480			let NewDocument =
481				DocumentStateDTO::Create(NewURI.clone(), OldDocument.map(|d| d.LanguageIdentifier), OriginalContent)?;
482
483			let DTO = NewDocument.ToDTO()?;
484
485			Guard.insert(NewURI.to_string(), NewDocument);
486
487			DTO
488		};
489
490		NotifyModelRemoved(self, &OriginalURI).await;
491
492		NotifyModelAdded(self, &NewDocumentState).await;
493
494		if let Err(Error) = self.ApplicationHandle.emit(
495			"sky://documents/renamed",
496			json!({ "oldUri": OriginalURI.to_string(), "newUri": NewURI.to_string() }),
497		) {
498			error!("[DocumentProvider] Failed to emit document renamed event: {}", Error);
499		}
500
501		Ok(Some(NewURI))
502	}

504	/// Saves all currently dirty documents.
505	async fn SaveAllDocuments(&self, IncludeUntitled:bool) -> Result<Vec<bool>, CommonError> {
506		info!(
507			"[DocumentProvider] SaveAllDocuments called (IncludeUntitled: {})",
508			IncludeUntitled
509		);
510
511		let URIsToSave:Vec<Url> = {
512			let OpenDocumentsGuard = self
513				.ApplicationState
514				.OpenDocuments
515				.lock()
516				.map_err(Utility::MapApplicationStateLockErrorToCommonError)?;
517
518			OpenDocumentsGuard
519				.values()
520				.filter(|Document| {
521					// Include documents that are dirty
522					if !Document.IsDirty {
523						return false;
524					}
525
526					// Include only file-scheme documents unless IncludeUntitled is true
527					if !IncludeUntitled && Document.URI.scheme() != "file" {
528						return false;
529					}
530
531					true
532				})
533				.map(|Document| Document.URI.clone())
534				.collect()
535		};
536
537		let mut Results = Vec::with_capacity(URIsToSave.len());
538
539		info!("[DocumentProvider] Saving {} dirty document(s)", URIsToSave.len());
540
541		for URI in URIsToSave {
542			let Result = self.SaveDocument(URI.clone()).await;
543
544			match &Result {
545				Ok(_) => {
546					info!("[DocumentProvider] Successfully saved {}", URI);
547				},
548				Err(Error) => {
549					error!("[DocumentProvider] Failed to save {}: {}", URI, Error);
550				},
551			}
552
553			Results.push(Result.is_ok());
554		}
555
556		Ok(Results)
557	}

559	/// Applies a collection of content changes to a document.
560	async fn ApplyDocumentChanges(
561		&self,
562
563		URI:Url,
564
565		NewVersionIdentifier:i64,
566
567		ChangesDTOCollection:Value,
568
569		_IsDirtyAfterChange:bool,
570
571		_IsUndoing:bool,
572
573		_IsRedoing:bool,
574	) -> Result<(), CommonError> {
575		trace!("[DocumentProvider] Applying changes to document: {}", URI);
576
577		{
578			let mut OpenDocumentsGuard = self
579				.ApplicationState
580				.OpenDocuments
581				.lock()
582				.map_err(Utility::MapApplicationStateLockErrorToCommonError)?;
583
584			if let Some(Document) = OpenDocumentsGuard.get_mut(URI.as_str()) {
585				Document.ApplyChanges(NewVersionIdentifier, &ChangesDTOCollection)?;
586			} else {
587				warn!("[DocumentProvider] Received changes for unknown document: {}", URI);
588
589				return Ok(());
590			}
591		}
592
593		NotifyModelChanged(self, &URI, NewVersionIdentifier, ChangesDTOCollection).await;
594
595		Ok(())
596	}

601/// Notifies Cocoon that a new document model has been added.
602async fn NotifyModelAdded(Environment:&MountainEnvironment, DocumentStateDTO:&Value) {
603	let URIString = DocumentStateDTO.get("URI").and_then(Value::as_str).unwrap_or("unknown");
604
605	info!("[DocumentProvider] Notifying ModelAdded for: {}", URIString);
606
607	let Payload = json!([DocumentStateDTO]);
608
609	let IPCProvider:Arc<dyn IPCProvider> = Environment.Require();
610
611	if let Err(Error) = IPCProvider
612		.SendNotificationToSideCar("cocoon-main".to_string(), "$acceptModelAdded".to_string(), Payload)
613		.await
614	{
615		error!(
616			"[DocumentProvider] Failed to send $acceptModelAdded for {}: {}",
617			URIString, Error
618		);
619	}
620}

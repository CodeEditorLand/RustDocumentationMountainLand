1//! # WorkspaceDelta
2//!
3//! Dispatches `$deltaWorkspaceFolders` notifications from Mountain to Cocoon
4//! whenever the open workspace folder set mutates. Called by every site that
5//! flips the folder list (boot-time seed, the `MountainWorkspaceOpen*`
6//! commands, pick-folder navigation, Wind add/remove, and the Cocoon-driven
7//! `$updateWorkspaceFolders` request).
8//!
9//! The delta is computed by
10//! [`WorkspaceState::SetWorkspaceFoldersReturnDelta`] and shipped as a
11//! fire-and-forget Vine notification: Cocoon's `NotificationHandler` converts
12//! it into a `didChangeWorkspaceFolders` event on
13//! `WorkspaceEventEmitter`, which powers every extension's
14//! `vscode.workspace.onDidChangeWorkspaceFolders` subscription. The same
15//! payload primes the local workspace snapshot in `WorkspaceNamespace` so
16//! `vscode.workspace.workspaceFolders` returns the fresh list on subsequent
17//! synchronous reads.
18
19use CommonLibrary::IPC::SkyEvent::SkyEvent;
20use serde_json::json;
21
22use crate::{ApplicationState::DTO::WorkspaceFolderStateDTO::WorkspaceFolderStateDTO, Vine::Client, dev_log};
23
24/// Serialisation shape matching the Cocoon-side Workspace shim. Mirrors the
25/// camelCase DTO Sky already serialises for `workspaces:getFolders`, so the
26/// Cocoon handler can pass the payload through to extension listeners without
27/// renaming fields.
28fn FolderToWire(Folder:&WorkspaceFolderStateDTO) -> serde_json::Value {
29	json!({
30		"uri": Folder.URI.to_string(),
31		"name": Folder.GetDisplayName(),
32		"index": Folder.Index,
33	})
34}
35
36/// Dispatch `$deltaWorkspaceFolders` to Cocoon. Returns immediately if both
37/// arrays are empty - no point waking the sidecar for a no-op mutation.
38///
39/// Errors are logged and swallowed: the workspace state is already updated by
40/// the caller, so a failed notification should not roll the mutation back. The
41/// log tag `[LandFix:WsDelta]` keeps the event grep-able in dev logs and is
42/// deliberately consistent with `[LandFix:WsNs]` on the Cocoon side.
43pub async fn DispatchDeltaWorkspaceFolders(Added:Vec<WorkspaceFolderStateDTO>, Removed:Vec<WorkspaceFolderStateDTO>) {
44	if Added.is_empty() && Removed.is_empty() {
45		return;
46	}
47
48	let AddedWire:Vec<serde_json::Value> = Added.iter().map(FolderToWire).collect();
49	let RemovedWire:Vec<serde_json::Value> = Removed.iter().map(FolderToWire).collect();
50
51	dev_log!(
52		"workspaces",
53		"[LandFix:WsDelta] $deltaWorkspaceFolders +{} -{} (first added={})",
54		AddedWire.len(),
55		RemovedWire.len(),
56		Added.first().map(|F| F.URI.as_str()).unwrap_or("<none>")
57	);
58
59	let Payload = json!({
60		"added": AddedWire,
61		"removed": RemovedWire,
62	});
63
64	if let Err(Error) =
65		Client::SendNotification("cocoon-main".to_string(), "$deltaWorkspaceFolders".to_string(), Payload).await
66	{
67		dev_log!(
68			"workspaces",
69			"warn: [LandFix:WsDelta] $deltaWorkspaceFolders notification failed: {}",
70			Error
71		);
72	}
73}

75/// Convenience wrapper: update the state and fire the delta in one call.
76///
77/// Spawns the notification on the current tokio runtime so callers in sync
78/// contexts (Tauri command handlers, boot-time seeding) don't have to build an
79/// async scope just to reach Cocoon. If no runtime is available (very early
80/// boot, unit tests), the notification is dropped - the state still mutates.
81pub fn UpdateWorkspaceFoldersAndNotify(
82	State:&crate::ApplicationState::State::WorkspaceState::WorkspaceState::State,
83	Folders:Vec<WorkspaceFolderStateDTO>,
84) {
85	let (Added, Removed) = State.SetWorkspaceFoldersReturnDelta(Folders);
86	if Added.is_empty() && Removed.is_empty() {
87		return;
88	}
89	if let Ok(Handle) = tokio::runtime::Handle::try_current() {
90		Handle.spawn(async move {
91			DispatchDeltaWorkspaceFolders(Added, Removed).await;
92		});
93	} else {
94		dev_log!(
95			"workspaces",
96			"warn: [LandFix:WsDelta] No tokio runtime available - delta dropped ({} added, {} removed)",
97			Added.len(),
98			Removed.len()
99		);
100	}
101}

103/// Variant that additionally emits a `sky://workspaces/changed` Tauri event
104/// so Wind/Sky can update their own caches (recent-folders list, sidebar
105/// breadcrumb) without polling `workspaces:getFolders`. Preferred call site
106/// whenever the caller already has an `AppHandle` in scope.
107pub fn UpdateWorkspaceFoldersAndBroadcast<R:tauri::Runtime>(
108	ApplicationHandle:&tauri::AppHandle<R>,
109	State:&crate::ApplicationState::State::WorkspaceState::WorkspaceState::State,
110	Folders:Vec<WorkspaceFolderStateDTO>,
111) {
112	use tauri::Emitter;
113
114	let (Added, Removed) = State.SetWorkspaceFoldersReturnDelta(Folders);
115	if Added.is_empty() && Removed.is_empty() {
116		return;
117	}
118	let AddedWire:Vec<serde_json::Value> = Added.iter().map(FolderToWire).collect();
119	let RemovedWire:Vec<serde_json::Value> = Removed.iter().map(FolderToWire).collect();
120	let BroadcastPayload = serde_json::json!({
121		"added": AddedWire.clone(),
122		"removed": RemovedWire.clone(),
123		"folders": State
124			.GetWorkspaceFolders()
125			.iter()
126			.map(FolderToWire)
127			.collect::<Vec<_>>(),
128	});
129	if let Err(Error) = ApplicationHandle.emit(SkyEvent::WorkspacesChanged.AsStr(), BroadcastPayload) {
130		dev_log!(
131			"workspaces",
132			"warn: [LandFix:WsDelta] sky://workspaces/changed emit failed: {}",
133			Error
134		);
135	}
136
137	// Persist the additions into the recently-opened list so the next boot's
138	// File → Open Recent menu and the Welcome screen can surface them.
139	// Mirrors VS Code's `ElectronMainWorkspacesMainService` behaviour.
140	PersistRecentlyOpened(&Added);
141
142	if let Ok(Handle) = tokio::runtime::Handle::try_current() {
143		Handle.spawn(async move {
144			DispatchDeltaWorkspaceFolders(Added, Removed).await;
145		});
146	}
147}

149/// Append every folder in `Added` to `~/.land/workspaces/RecentlyOpened.json`,
150/// deduping by URI and capping at 50 entries (the VS Code default). Swallows
151/// every error - a failed write must not prevent the workspace change.
152fn PersistRecentlyOpened(Added:&[WorkspaceFolderStateDTO]) {
153	if Added.is_empty() {
154		return;
155	}
156	let Home = std::env::var("HOME")
157		.or_else(|_| std::env::var("USERPROFILE"))
158		.unwrap_or_default();
159	if Home.is_empty() {
160		return;
161	}
162	let Path = std::path::PathBuf::from(Home)
163		.join(".land")
164		.join("workspaces")
165		.join("RecentlyOpened.json");
166	let mut Current:serde_json::Map<String, serde_json::Value> = std::fs::read_to_string(&Path)
167		.ok()
168		.and_then(|Contents| serde_json::from_str::<serde_json::Value>(&Contents).ok())
169		.and_then(|V| V.as_object().cloned())
170		.unwrap_or_default();
171	let mut Workspaces = Current
172		.get("workspaces")
173		.and_then(|V| V.as_array())
174		.cloned()
175		.unwrap_or_default();
176	for Folder in Added {
177		let Uri = Folder.URI.to_string();
178		Workspaces.retain(|Entry| Entry.get("uri").and_then(|V| V.as_str()).unwrap_or("") != Uri);
179		Workspaces.insert(
180			0,
181			serde_json::json!({
182				"uri": Uri,
183				"label": Folder.GetDisplayName(),
184			}),
185		);
186	}
187	Workspaces.truncate(50);
188	Current.insert("workspaces".into(), serde_json::Value::Array(Workspaces));
189	if !Current.contains_key("files") {
190		Current.insert("files".into(), serde_json::json!([]));
191	}
192	if let Some(Parent) = Path.parent() {
193		let _ = std::fs::create_dir_all(Parent);
194	}
195	if let Ok(Serialised) = serde_json::to_vec_pretty(&serde_json::Value::Object(Current)) {
196		let _ = std::fs::write(&Path, Serialised);
197	}
198}

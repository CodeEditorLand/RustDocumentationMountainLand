2//! Cocoon → Mountain `window.createTextEditorDecorationType` /
3//! `window.disposeTextEditorDecorationType` notifications. Forwards
4//! the payload on `sky://decoration/<suffix>`; Sky's editor renderer
5//! owns the Monaco-side decoration lifecycle so Mountain is a pure
6//! relay.
7//!
8//! Per session log audit `20260501T053137`: ~337 create + 317
9//! dispose calls, mostly from extension activation registering
10//! syntax-highlight decorations for new file types. Each emit hit
11//! Tauri's serialised WebKit channel that also delivers keystrokes.
12//! 16ms coalescer per method name buffers payloads and emits a
13//! single `{ batch: [...] }` per channel per frame; SkyBridge's
14//! listener demultiplexes back to per-decoration `cel:decoration:*`
15//! CustomEvents.
16
17use std::{
18	collections::HashMap,
19	sync::{
20		Arc,
21		Mutex,
22		OnceLock,
23		atomic::{AtomicBool, Ordering},
24	},
25	time::Duration,
26};

28use serde_json::{Value, json};
29use tauri::{AppHandle, Emitter};

31use crate::{Vine::Server::MountainVinegRPCService::MountainVinegRPCService, dev_log};

33struct DecorationEmitBatch {
34	Pending:Mutex<HashMap<String, Vec<Value>>>,
35	FlushScheduled:AtomicBool,
36}
37
38static DECORATION_EMIT_BATCH:OnceLock<Arc<DecorationEmitBatch>> = OnceLock::new();
39
40fn EnqueueDecorationEmit(Handle:&AppHandle, Channel:String, Payload:Value) {
41	let Batch = DECORATION_EMIT_BATCH.get_or_init(|| {
42		Arc::new(DecorationEmitBatch { Pending:Mutex::new(HashMap::new()), FlushScheduled:AtomicBool::new(false) })
43	});
44
45	{
46		let mut Guard = Batch.Pending.lock().unwrap();
47		Guard.entry(Channel).or_insert_with(Vec::new).push(Payload);
48	}
49
50	if !Batch.FlushScheduled.swap(true, Ordering::AcqRel) {
51		let BatchClone = Batch.clone();
52		let HandleClone = Handle.clone();
53		tokio::spawn(async move {
54			tokio::time::sleep(Duration::from_millis(16)).await;
55			let Drained:HashMap<String, Vec<Value>> = {
56				let mut Guard = BatchClone.Pending.lock().unwrap();
57				std::mem::take(&mut *Guard)
58			};
59			BatchClone.FlushScheduled.store(false, Ordering::Release);
60			for (ChannelName, Payloads) in Drained {
61				let Count = Payloads.len();
62				match HandleClone.emit(&ChannelName, json!({ "batch": Payloads })) {
63					Ok(()) => dev_log!("sky-emit", "[SkyEmit] ok channel={} batch={}", ChannelName, Count),
64					Err(Error) => {
65						dev_log!(
66							"sky-emit",
67							"[SkyEmit] fail channel={} batch={} error={}",
68							ChannelName,
69							Count,
70							Error
71						)
72					},
73				}
74			}
75		});
76	}
77}
78
79pub async fn DecorationTypeLifecycle(Service:&MountainVinegRPCService, MethodName:&str, Parameter:&Value) {
80	let EventName = format!("sky://decoration/{}", &MethodName["window.".len()..]);
81	EnqueueDecorationEmit(Service.ApplicationHandle(), EventName, Parameter.clone());
82}

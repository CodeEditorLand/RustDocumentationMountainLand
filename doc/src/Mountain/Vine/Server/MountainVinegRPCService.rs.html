1//! # MountainVinegRPCService
2//!
3//! Defines the gRPC service implementation for Mountain. This struct handles
4//! incoming RPC calls from the `Cocoon` sidecar, dispatches them to the
5//! application's core logic via the `Track` module, and returns the results.
6//!
7//! ## Service Methods
8//!
9//! - **process_cocoon_request**: Handles request-response calls from Cocoon
10//! - **send_cocoon_notification**: Handles fire-and-forget notifications from
11//!   Cocoon
12//! - **cancel_operation**: Cancels long-running operations requested by Cocoon
13//!
14//! ## Request Processing
15//!
16//! 1. Deserialize JSON parameters from request
17//! 2. Validate method name and parameters
18//! 3. Dispatch request to Track::DispatchLogic
19//! 4. Serialize response or error
20//! 5. Return gRPC response with proper status codes
21//!
22//! ## Error Handling
23//!
24//! All errors are converted to JSON-RPC compliant Error objects:
25//! - Parse errors: code -32700
26//! - Server errors: code -32000
27//! - Method not found: code -32601
28//! - Invalid params: code -32602
29//!
30//! ## Security
31//!
32//! - Parameter validation before processing
33//! - Message size limits enforced
34//! - Method name sanitization
35//! - Safe error messages (no sensitive data)
36

122	/// Registers an operation for potential cancellation

149	/// Validates a generic request before processing.
150	///
151	/// # Parameters
152	/// - `request`: The request to validate
153	///
154	/// # Returns
155	/// - `Ok(())`: Request is valid
156	/// - `Err(Status)`: Validation failed with appropriate gRPC status
157	fn ValidateRequest(&self, request:&GenericRequest) -> Result<(), Status> {
158		// Validate method name
159		if request.method.is_empty() {
160			return Err(Status::invalid_argument("Method name cannot be empty"));
161		}
162
163		if request.method.len() > ServiceConfig::MAX_METHOD_NAME_LENGTH {
164			return Err(Status::invalid_argument(format!(
165				"Method name exceeds maximum length of {} characters",
166				ServiceConfig::MAX_METHOD_NAME_LENGTH
167			)));
168		}
169
170		// Validate parameter size (rough estimate using JSON bytes)
171		if request.parameter.len() > 4 * 1024 * 1024 {
172			return Err(Status::resource_exhausted("Request parameter size exceeds limit"));
173		}
174
175		// Check for potentially malicious method names
176		if request.method.contains("../") || request.method.contains("::") {
177			return Err(Status::permission_denied("Invalid method name format"));
178		}
179
180		Ok(())
181	}

183	/// Creates a JSON-RPC compliant error response.
184	///
185	/// # Parameters
186	/// - `RequestIdentifier`: The request ID to echo back
187	/// - `code`: JSON-RPC error code
188	/// - `message`: Error message
189	/// - `data`: Optional error data (serialized)
190	///
191	/// # Returns
192	/// GenericResponse with error populated
193	fn CreateErrorResponse(RequestIdentifier:u64, code:i32, message:String, data:Option<Vec<u8>>) -> GenericResponse {
194		GenericResponse {
195			request_identifier:RequestIdentifier,
196			result:vec![],
197			error:Some(RPCError { code, message, data:data.unwrap_or_default() }),
198		}
199	}
200
201	/// Creates a successful JSON-RPC response.
202	///
203	/// # Parameters
204	/// - `RequestIdentifier`: The request ID to echo back
205	/// - `result`: Result value to serialize
206	///
207	/// # Returns
208	/// GenericResponse with result populated, or error if serialization fails
209	fn CreateSuccessResponse(RequestIdentifier:u64, result:&Value) -> GenericResponse {
210		let result_bytes = match serde_json::to_vec(result) {
211			Ok(bytes) => bytes,
212			Err(e) => {
213				dev_log!("grpc", "error: [MountainVinegRPCService] Failed to serialize result: {}", e);
214
215				// Return error response instead
216				return Self::CreateErrorResponse(
217					RequestIdentifier,
218					-32603, // Internal error
219					"Failed to serialize response".to_string(),
220					None,
221				);
222			},
223		};
224
225		GenericResponse { request_identifier:RequestIdentifier, result:result_bytes, error:None }
226	}
227}

229#[tonic::async_trait]
230impl MountainService for MountainVinegRPCService {
231	/// Handles generic request-response RPCs from Cocoon.
232	///
233	/// This is the main entry point for Cocoon to request operations from
234	/// Mountain. It validates the request, deserializes parameters, dispatches
235	/// to the Track module, and returns the result or error in JSON-RPC
236	/// format.
237	///
238	/// # Parameters
239	/// - `request`: GenericRequest containing method name and serialized
240	///   parameters
241	///
242	/// # Returns
243	/// - `Ok(Response<GenericResponse>)`: Response with result or error
244	/// - `Err(Status)`: gRPC status error (only for critical failures)
245	async fn process_cocoon_request(
246		&self,
247		request:Request<GenericRequest>,
248	) -> Result<Response<GenericResponse>, Status> {
249		let RequestData = request.into_inner();
250
251		let MethodName = RequestData.method.clone();
252
253		let RequestIdentifier = RequestData.request_identifier;
254		let ReceiveInstant = std::time::Instant::now();
255
256		dev_log!(
257			"grpc",
258			"[MountainVinegRPCService] Received gRPC Request [ID: {}]: Method='{}'",
259			RequestIdentifier,
260			MethodName
261		);
262
263		// Hot-path instrumentation (BATCH-16). Every RPC that shows up with
264		// uniform 700 ms latency (tree.register, Configuration.Inspect,
265		// Command.Execute) emits a `[LandFix:RPC]` marker here so p50/p95 can
266		// be derived from the log without patching every handler. The
267		// monotonic `t_ns` is a `SystemTime::UNIX_EPOCH` offset so Cocoon's
268		// `process.hrtime.bigint()` wire-send stamp can be diffed into three
269		// hops: wire → grpc-recv (transit), grpc-recv → dispatch-enter
270		// (Track resolve), dispatch-enter → registered (handler body).
271		let IsHotRpc = matches!(
272			MethodName.as_str(),
273			"$tree:register" | "tree.register" | "Configuration.Inspect" | "Command.Execute"
274		);
275		if IsHotRpc {
276			let InstrumentRecvNs = std::time::SystemTime::now()
277				.duration_since(std::time::UNIX_EPOCH)
278				.map(|D| D.as_nanos())
279				.unwrap_or(0);
280			dev_log!(
281				"grpc",
282				"[LandFix:RPC] grpc-recv method={} id={} size={} t_ns={}",
283				MethodName,
284				RequestIdentifier,
285				RequestData.parameter.len(),
286				InstrumentRecvNs
287			);
288		}
289
290		// Validate request before processing
291		if let Err(status) = self.ValidateRequest(&RequestData) {
292			dev_log!("grpc", "warn: [MountainVinegRPCService] Request validation failed: {}", status);
293
294			return Ok(Response::new(Self::CreateErrorResponse(
295				RequestIdentifier,
296				-32602, // Invalid params
297				status.message().to_string(),
298				None,
299			)));
300		}
301
302		// Deserialize JSON parameters
303		let ParametersValue:Value = match serde_json::from_slice(&RequestData.parameter) {
304			Ok(v) => {
305				// The previous `{:?}` Debug format serialised the full
306				// `Value` on every request - cheap for small payloads
307				// (`Diagnostic.Clear`), catastrophic for `tree.register` and
308				// `Configuration.Inspect` whose options blobs walk recursive
309				// structures. Only log param size at the default dev-log
310				// level and let the DevLog `all` target surface the body if
311				// the caller opts in.
312				dev_log!(
313					"grpc",
314					"[MountainVinegRPCService] Params for [ID: {}] ({} bytes)",
315					RequestIdentifier,
316					RequestData.parameter.len()
317				);
318				v
319			},
320			Err(e) => {
321				let msg = format!("Failed to deserialize parameters for method '{}': {}", MethodName, e);
322
323				dev_log!("grpc", "error: {}", msg);
324
325				return Ok(Response::new(Self::CreateErrorResponse(
326					RequestIdentifier,
327					-32700, // Parse error
328					msg,
329					None,
330				)));
331			},
332		};
333
334		dev_log!(
335			"grpc",
336			"[MountainVinegRPCService] Dispatching request [ID: {}] to Track::DispatchLogic",
337			RequestIdentifier
338		);
339
340		// Dispatch request to Track module for processing
341		let DispatchResult = Track::SideCarRequest::DispatchSideCarRequest(
342			self.ApplicationHandle.clone(),
343			self.RunTime.clone(),
344			// In the future, this could come from connection metadata
345			"cocoon-main".to_string(),
346			MethodName.clone(),
347			ParametersValue,
348		)
349		.await;
350
351		match DispatchResult {
352			Ok(SuccessfulResult) => {
353				if IsHotRpc {
354					dev_log!(
355						"grpc",
356						"[LandFix:RPC] dispatched method={} id={} elapsed={}ms",
357						MethodName,
358						RequestIdentifier,
359						ReceiveInstant.elapsed().as_millis()
360					);
361				}
362				dev_log!(
363					"grpc",
364					"[MountainVinegRPCService] Request [ID: {}] completed successfully",
365					RequestIdentifier
366				);
367
368				Ok(Response::new(Self::CreateSuccessResponse(RequestIdentifier, &SuccessfulResult)))
369			},
370
371			Err(ErrorString) => {
372				// `FileSystem.ReadFile` "Resource not found" is a routine
373				// occurrence - extensions probe for optional cache files on
374				// activate (terminal-suggest, json-language-features schema
375				// associations, etc.). Downgrade 404s to an info-level note
376				// so the error log reflects genuine failures only. The
377				// response itself is still returned with code -32000 so
378				// Cocoon's `readFile` shim can convert it into a proper
379				// `vscode.FileSystemError.FileNotFound`.
380				let LooksLike404 = MethodName == "FileSystem.ReadFile"
381					&& (ErrorString.to_lowercase().contains("resource not found")
382						|| ErrorString.to_lowercase().contains("not found"));
383				if LooksLike404 {
384					dev_log!(
385						"grpc",
386						"[LandFix:MountainVinegRPC] Request [ID: {}] {} 404 (benign): {}",
387						RequestIdentifier,
388						MethodName,
389						ErrorString
390					);
391				} else {
392					dev_log!(
393						"grpc",
394						"error: [MountainVinegRPCService] Request [ID: {}] failed: {}",
395						RequestIdentifier,
396						ErrorString
397					);
398				}
399
400				Ok(Response::new(Self::CreateErrorResponse(
401					RequestIdentifier,
402					-32000, // Server error
403					ErrorString,
404					None,
405				)))
406			},
407		}
408	}

410	/// Handles generic fire-and-forget notifications from Cocoon.
411	///
412	/// Notifications do not expect a response beyond acknowledgment.
413	/// They are used for status updates, events, and other asynchronous
414	/// notifications.
415	///
416	/// # Parameters
417	/// - `request`: GenericNotification with method name and parameters
418	///
419	/// # Returns
420	/// - `Ok(Response<Empty>)`: Notification was received and logged
421	/// - `Err(Status)`: Critical error during processing
422	///
423	/// # TODO
424	/// Future implementation should route notifications to dedicated handlers:
425	/// ```rust,ignore
426	/// let Parameter: Value = serde_json::from_slice(&notification.parameter)?;
427	/// NotificationHandler::Handle(MethodName, Parameter).await?;
428	/// ```
429	async fn send_cocoon_notification(&self, request:Request<GenericNotification>) -> Result<Response<Empty>, Status> {
430		let NotificationData = request.into_inner();
431
432		let MethodName = NotificationData.method;
433
434		dev_log!(
435			"grpc",
436			"[MountainVinegRPCService] Received gRPC Notification: Method='{}'",
437			MethodName
438		);
439
440		// Validate notification method name
441		if MethodName.is_empty() {
442			dev_log!(
443				"grpc",
444				"warn: [MountainVinegRPCService] Received notification with empty method name"
445			);
446			return Err(Status::invalid_argument("Method name cannot be empty"));
447		}
448
449		// Route notifications to appropriate handlers based on MethodName. Currently
450		// only logs known notification types and acknowledges all others. A complete
451		// implementation would maintain a registry of notification handlers per method,
452		// route notifications to registered handlers asynchronously, allow handlers
453		// to perform side effects (state updates, UI updates), support cancellation
454		// and timeouts for long-running handlers, and log unhandled notifications
455		// at debug level for diagnostics. Known notifications include:
456		// ExtensionActivated, ExtensionDeactivated, WebviewReady.
457
458		// Parse parameters for handlers that need them
459		let Parameter:Value = if NotificationData.parameter.is_empty() {
460			Value::Null
461		} else {
462			serde_json::from_slice(&NotificationData.parameter).unwrap_or(Value::Null)
463		};

472			},

475			},

477				super::Notification::ExtensionDeactivated::ExtensionDeactivated(self, &Parameter).await;
478			},
479			"WebviewReady" => {
480				super::Notification::WebviewReady::WebviewReady(self, &Parameter).await;
481			},

485			"progress.report" => {

862	/// Handles a request from Cocoon to cancel a long-running operation.
863	///
864	/// This method is called when Cocoon wants to cancel an operation that
865	/// was previously initiated via process_cocoon_request.
866	///
867	/// # Parameters
868	/// - `request`: CancelOperationRequest with the request ID to cancel
869	///
870	/// # Returns
871	/// - `Ok(Response<Empty>)`: Cancellation was initiated
872	/// - `Err(Status)`: Critical error during cancellation
873	async fn cancel_operation(&self, request:Request<CancelOperationRequest>) -> Result<Response<Empty>, Status> {
874		let cancel_request = request.into_inner();
875
876		let RequestIdentifierToCancel = cancel_request.request_identifier_to_cancel;
877
878		dev_log!(
879			"grpc",
880			"[MountainVinegRPCService] Received CancelOperation request for RequestID: {}",
881			RequestIdentifierToCancel
882		);
883
884		// Look up the operation in the active operations registry
885		let cancel_token = {
886			let operations = self.ActiveOperations.read().await;
887			operations.get(&RequestIdentifierToCancel).cloned()
888		};
889
890		match cancel_token {
891			Some(token) => {
892				// Trigger cancellation token to signal the operation to abort
893				token.cancel();
894
895				dev_log!(
896					"grpc",
897					"[MountainVinegRPCService] Successfully initiated cancellation for operation {}",
898					RequestIdentifierToCancel
899				);
900
901				// Note: We don't remove the token here - the operation itself should
902				// call UnregisterOperation when it completes. This allows the
903				// operation to detect the cancellation and clean up properly.
904
905				Ok(Response::new(Empty {}))
906			},
907			None => {
908				// Operation not found - it may have already completed
909				dev_log!(
910					"grpc",
911					"warn: [MountainVinegRPCService] Cannot cancel operation {}: operation not found (may have \
912					 already completed)",
913					RequestIdentifierToCancel
914				);
915
916				// Return success anyway - the operation is not running
917				Ok(Response::new(Empty {}))
918			},
919		}
920	}

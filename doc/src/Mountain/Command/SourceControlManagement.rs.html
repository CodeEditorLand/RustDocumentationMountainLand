1//! # SourceControlManagement (Command)
2//!
3//! RESPONSIBILITIES:
4//! - Defines Tauri command handlers for Source Control Management (SCM) operations
5//! - Exposes SCM functionality to the Sky frontend via IPC
6//! - Aggregates SCM provider state, resources, and groups for UI rendering
7//! - Routes SCM commands to appropriate providers (commit, push, pull, branch ops)
8//! - Manages branch listing, checkout, and commit history retrieval
9//! - Handles resource staging (git add equivalent)
10//!
11//! ARCHITECTURAL ROLE:
12//! - Command module that bridges Sky UI requests to [`SourceControlManagementProvider`]
13//!   implementations in the Environment layer
14//! - Uses Tauri's `#[command]` attribute for IPC exposure
15//! - Reads from [`ApplicationState.SourceControlManagement*`](crate::ApplicationState::ApplicationState)
16//!   fields to gather state
17//! - TODO: Should forward commands to provider methods via DI (Require trait)
18//!
19//! COMMAND REFERENCE (Tauri IPC):
20//! - [`GetAllSourceControlManagementState`](crate::Command::SourceControlManagement::GetAllSourceControlManagementState):
21//!   Returns complete snapshot of providers, groups, and resources for SCM view
22//! - [`GetSCMResourceChanges`](crate::Command::SourceControlManagement::GetSCMResourceChanges):
23//!   Get file changes for a specific provider
24//! - [`ExecuteSCMCommand`](crate::Command::SourceControlManagement::ExecuteSCMCommand):
25//!   Execute SCM operation (commit, push, pull, etc.)
26//! - [`GetSCMBranches`](crate::Command::SourceControlManagement::GetSCMBranches):
27//!   List branches for provider
28//! - [`CheckoutSCMBranch`](crate::Command::SourceControlManagement::CheckoutSCMBranch):
29//!   Switch to a different branch
30//! - [`GetSCMCommitHistory`](crate::Command::SourceControlManagement::GetSCMCommitHistory):
31//!   Retrieve commit log with optional limit
32//! - [`StageSCMResource`](crate::Command::SourceControlManagement::StageSCMResource):
33//!   Stage or unstage a file resource
34//!
35//! ERROR HANDLING:
36//! - Returns `Result<Value, String>` where error string is sent to frontend
37//! - Uses `MapLockError` to convert Mutex poisoning to error strings
38//! - Provider identifier parsing may fail (unwrap_or(0) fallback)
39//! - Unknown commands return error string
40//!
41//! PERFORMANCE:
42//! - State access uses RwLock reads; cloning entire state maps (may be heavy)
43//! - TODO: Consider pagination for large commit histories and resource lists
44//!

80/// Retrieves the complete state of all Source Control Management providers,
81/// groups, and resources for rendering in the UI.
82///
83/// This command is called by the frontend to get a full snapshot of the SCM
84/// view.
85#[command]
86pub async fn GetAllSourceControlManagementState(State:State<'_, Arc<ApplicationState>>) -> Result<Value, String> {
87	log::debug!("[SourceControlManagement Command] Getting all SCM state for UI.");
88
89	let Providers = State
90		.SourceControlManagementProviders
91		.lock()
92		.map_err(MapLockError)
93		.map_err(|Error| Error.to_string())?
94		.clone();
95
96	let Groups = State
97		.SourceControlManagementGroups
98		.lock()
99		.map_err(MapLockError)
100		.map_err(|Error| Error.to_string())?
101		.clone();
102
103	let Resources = State
104		.SourceControlManagementResources
105		.lock()
106		.map_err(MapLockError)
107		.map_err(|Error| Error.to_string())?
108		.clone();
109
110	Ok(json!({
111		"providers": Providers,
112		"groups": Groups,
113		"resources": Resources,
114	}))
115}

117#[command]
118pub async fn GetSCMResourceChanges(
119	State:State<'_, Arc<ApplicationState>>,
120
121	ProviderIdentifier:String,
122) -> Result<Value, String> {
123	log::debug!("[SCM Command] Getting resource changes for provider: {}", ProviderIdentifier);
124
125	let resources_map = State
126		.SourceControlManagementResources
127		.lock()
128		.map_err(MapLockError)
129		.map_err(|Error| Error.to_string())?
130		.clone();
131
132	// Filter resources by provider - Resources is HashMap<u32, HashMap<String,
133	// Vec<SourceControlManagementResourceDTO>>> We need to flatten and filter by
134	// ProviderHandle (u32) matching ProviderIdentifier (String)
135	let provider_handle_u32 = ProviderIdentifier.parse::<u32>().unwrap_or(0);
136	let ProviderResources:Vec<_> = resources_map
137		.iter()
138		.flat_map(|(_group_id, group_resources)| group_resources.values())
139		.flat_map(|vec_resources| vec_resources.iter())
140		.filter(|r| r.ProviderHandle == provider_handle_u32)
141		.cloned()
142		.collect();
143
144	Ok(json!({
145		"resources": ProviderResources,
146	}))
147}

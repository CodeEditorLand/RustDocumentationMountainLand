36/// A specific Tauri command handler for the UI to fetch the children of a tree
37/// view node. This handler dispatches to the correct provider (native or
38/// proxied).
39#[command]
40pub async fn GetTreeViewChildren(
41	ApplicationHandle:AppHandle<Wry>,
42
43	_State:State<'_, Arc<ApplicationState>>,
44
45	ViewId:String,
46
47	ElementHandle:Option<String>,
48) -> Result<Value, String> {
49	log::debug!(
50		"[DispatchLogic] Getting TreeView children for '{}', element: {:?}",
51		ViewId,
52		ElementHandle
53	);
54
55	let RunTime = ApplicationHandle.state::<Arc<ApplicationRunTime>>().inner().clone();
56
57	let Environment:Arc<MountainEnvironment> = RunTime.Environment.clone();
58
59	let TreeProvider:Arc<dyn CommonTreeViewProvider> = Environment.Require();
60
61	match TreeProvider.GetChildren(ViewId.clone(), ElementHandle).await {
62		Ok(Children) => Ok(json!(Children)),
63		Err(Error) => {
64			let ErrorMessage = format!("Failed to get children for tree view '{}': {}", ViewId, Error);
65			log::error!("{}", ErrorMessage);
66			Err(ErrorMessage)
67		},
68	}
69}

142/// Refreshes a tree view.
143#[command]
144pub async fn RefreshTreeView(
145	ApplicationHandle:AppHandle<Wry>,
146
147	_State:State<'_, Arc<ApplicationState>>,
148
149	ViewId:String,
150
151	ItemsToRefresh:Option<Vec<String>>,
152) -> Result<Value, String> {
153	log::debug!(
154		"[TreeView Command] Refreshing tree view '{}', items: {:?}",
155		ViewId,
156		ItemsToRefresh
157	);
158
159	let RunTime = ApplicationHandle.state::<Arc<ApplicationRunTime>>().inner().clone();
160
161	let Environment:Arc<MountainEnvironment> = RunTime.Environment.clone();
162
163	let RefreshValue:Option<Value> = ItemsToRefresh.and_then(|items| serde_json::to_value(items).ok());
164
165	match Environment.RefreshTreeView(ViewId.clone(), RefreshValue).await {
166		Ok(_) => Ok(json!({ "success": true })),
167		Err(Error) => {
168			let ErrorMessage = format!("Failed to refresh tree view '{}': {}", ViewId, Error);
169			log::error!("{}", ErrorMessage);
170			Err(ErrorMessage)
171		},
172	}
173}

175/// Reveals a specific tree item.
176#[command]
177pub async fn RevealTreeViewItem(
178	ApplicationHandle:AppHandle<Wry>,
179
180	_State:State<'_, Arc<ApplicationState>>,
181
182	ViewId:String,
183
184	ItemHandle:String,
185
186	Options:Option<Value>,
187) -> Result<Value, String> {
188	log::debug!("[TreeView Command] Revealing item '{}' in view '{}'", ItemHandle, ViewId);
189
190	let RunTime = ApplicationHandle.state::<Arc<ApplicationRunTime>>().inner().clone();
191
192	let Environment:Arc<MountainEnvironment> = RunTime.Environment.clone();
193
194	let OptionsValue = Options.unwrap_or(json!({}));
195
196	match Environment.RevealTreeItem(ViewId.clone(), ItemHandle, OptionsValue).await {
197		Ok(_) => Ok(json!({ "success": true })),
198		Err(Error) => {
199			let ErrorMessage = format!("Failed to reveal tree item in view '{}': {}", ViewId, Error);
200			log::error!("{}", ErrorMessage);
201			Err(ErrorMessage)
202		},
203	}
204}

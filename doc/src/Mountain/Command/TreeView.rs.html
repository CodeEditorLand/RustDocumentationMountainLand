1//! # TreeView (Command)
2//!
3//! RESPONSIBILITIES:
4//! - Defines Tauri command handlers for TreeView operations from Sky frontend
5//! - Bridges TreeView UI requests to
6//!   [`TreeViewProvider`](CommonLibrary::TreeView::TreeViewProvider)
7//! - Handles tree data fetching, expansion, selection, and refresh operations
8//! - Manages tree view state persistence and restoration (stubs)
9//!
10//! ARCHITECTURAL ROLE:
11//! - Command module exposing TreeView functionality via Tauri IPC
12//!   (`#[command]`)
13//! - Delegates to Environment's
14//!   [`TreeViewProvider`](crate::Environment::TreeViewProvider) via DI with
15//!   `Require()` trait from `MountainEnvironment`
16//! - Translates frontend requests to provider method calls with proper error
17//!   mapping
18//!
19//! COMMAND REFERENCE (Tauri IPC):
20//! - [`GetTreeViewChildren`](crate::Command::TreeView::GetTreeViewChildren):
21//!   Fetch child items for a tree node (by `ElementHandle`, null for root)
22//! - [`GetTreeViewItem`](crate::Command::TreeView::GetTreeViewItem): Get tree
23//!   item metadata (label, icon, description) by handle
24//! - [`OnTreeViewExpansionChanged`](crate::Command::TreeView::OnTreeViewExpansionChanged):
25//!   Notify when user expands/collapses a node (stub - trait method missing)
26//! - [`OnTreeViewSelectionChanged`](crate::Command::TreeView::OnTreeViewSelectionChanged):
27//!   Notify when user selects/deselects tree items (stub - trait method missing)
28//! - [`RefreshTreeView`](crate::Command::TreeView::RefreshTreeView): Request
29//!   tree view to refresh its data, optionally specific items
30//! - [`RevealTreeViewItem`](crate::Command::TreeView::RevealTreeViewItem):
31//!   Request to reveal/focus a specific tree item in the UI
32//! - [`PersistTreeView`](crate::Command::TreeView::PersistTreeView): Save tree
33//!   view state (scroll position, expansion) (stub)
34//! - [`RestoreTreeView`](crate::Command::TreeView::RestoreTreeView): Restore
35//!   previously saved tree view state (stub)
36//!
37//! ERROR HANDLING:
38//! - Returns `Result<Value, String>` with error strings sent to frontend
39//! - Provider errors are logged with context and converted to error strings
40//! - Missing trait methods return structured error indicating not implemented
41//!
42//! PERFORMANCE:
43//! - All commands are async and non-blocking
44//! - Tree data fetching should be efficient; provider may cache results
45//! - Refresh can target specific items to avoid full tree rebuild
46//!
47//! VS CODE REFERENCE:
48//! - `vs/workbench/api/browser/mainThreadTreeViews.ts` - main thread tree view
49//!   API
50//! - `vs/workbench/api/common/extHostTreeViews.ts` - extension host tree view
51//!   API
52//! - `vs/workbench/contrib/files/browser/explorerView.ts` - file explorer tree
53//!   view
54//! - `vs/workbench/contrib/tree/browser/treeView.ts` - generic tree view
55//!   component

95/// A specific Tauri command handler for the UI to fetch the children of a tree
96/// view node. This handler dispatches to the correct provider (native or
97/// proxied).
98#[command]
99pub async fn GetTreeViewChildren(
100	ApplicationHandle:AppHandle<Wry>,
101
102	_State:State<'_, Arc<ApplicationState>>,
103
104	ViewId:String,
105
106	ElementHandle:Option<String>,
107) -> Result<Value, String> {
108	log::debug!(
109		"[DispatchLogic] Getting TreeView children for '{}', element: {:?}",
110		ViewId,
111		ElementHandle
112	);
113
114	let RunTime = ApplicationHandle.state::<Arc<ApplicationRunTime>>().inner().clone();
115
116	let Environment:Arc<MountainEnvironment> = RunTime.Environment.clone();
117
118	let TreeProvider:Arc<dyn CommonTreeViewProvider> = Environment.Require();
119
120	match TreeProvider.GetChildren(ViewId.clone(), ElementHandle).await {
121		Ok(Children) => Ok(json!(Children)),
122		Err(Error) => {
123			let ErrorMessage = format!("Failed to get children for tree view '{}': {}", ViewId, Error);
124			log::error!("{}", ErrorMessage);
125			Err(ErrorMessage)
126		},
127	}
128}

208/// Refreshes a tree view.
209#[command]
210pub async fn RefreshTreeView(
211	ApplicationHandle:AppHandle<Wry>,
212
213	_State:State<'_, Arc<ApplicationState>>,
214
215	ViewId:String,
216
217	ItemsToRefresh:Option<Vec<String>>,
218) -> Result<Value, String> {
219	log::debug!(
220		"[TreeView Command] Refreshing tree view '{}', items: {:?}",
221		ViewId,
222		ItemsToRefresh
223	);
224
225	let RunTime = ApplicationHandle.state::<Arc<ApplicationRunTime>>().inner().clone();
226
227	let Environment:Arc<MountainEnvironment> = RunTime.Environment.clone();
228
229	let RefreshValue:Option<Value> = ItemsToRefresh.and_then(|items| serde_json::to_value(items).ok());
230
231	match Environment.RefreshTreeView(ViewId.clone(), RefreshValue).await {
232		Ok(_) => Ok(json!({ "success": true })),
233		Err(Error) => {
234			let ErrorMessage = format!("Failed to refresh tree view '{}': {}", ViewId, Error);
235			log::error!("{}", ErrorMessage);
236			Err(ErrorMessage)
237		},
238	}
239}

241/// Reveals a specific tree item.
242#[command]
243pub async fn RevealTreeViewItem(
244	ApplicationHandle:AppHandle<Wry>,
245
246	_State:State<'_, Arc<ApplicationState>>,
247
248	ViewId:String,
249
250	ItemHandle:String,
251
252	Options:Option<Value>,
253) -> Result<Value, String> {
254	log::debug!("[TreeView Command] Revealing item '{}' in view '{}'", ItemHandle, ViewId);
255
256	let RunTime = ApplicationHandle.state::<Arc<ApplicationRunTime>>().inner().clone();
257
258	let Environment:Arc<MountainEnvironment> = RunTime.Environment.clone();
259
260	let OptionsValue = Options.unwrap_or(json!({}));
261
262	match Environment.RevealTreeItem(ViewId.clone(), ItemHandle, OptionsValue).await {
263		Ok(_) => Ok(json!({ "success": true })),
264		Err(Error) => {
265			let ErrorMessage = format!("Failed to reveal tree item in view '{}': {}", ViewId, Error);
266			log::error!("{}", ErrorMessage);
267			Err(ErrorMessage)
268		},
269	}
270}

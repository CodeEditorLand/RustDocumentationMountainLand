1//! # TreeView (Command)
2//!
3//! RESPONSIBILITIES:
4//! - Defines Tauri command handlers for TreeView operations from Sky frontend
5//! - Bridges TreeView UI requests to [`TreeViewProvider`](CommonLibrary::TreeView::TreeViewProvider)
6//! - Handles tree data fetching, expansion, selection, and refresh operations
7//! - Manages tree view state persistence and restoration (stubs)
8//!
9//! ARCHITECTURAL ROLE:
10//! - Command module exposing TreeView functionality via Tauri IPC (`#[command]`)
11//! - Delegates to Environment's [`TreeViewProvider`](crate::Environment::TreeViewProvider)
12//!   via DI with `Require()` trait from `MountainEnvironment`
13//! - Translates frontend requests to provider method calls with proper error mapping
14//!
15//! COMMAND REFERENCE (Tauri IPC):
16//! - [`GetTreeViewChildren`](crate::Command::TreeView::GetTreeViewChildren):
17//!   Fetch child items for a tree node (by `ElementHandle`, null for root)
18//! - [`GetTreeViewItem`](crate::Command::TreeView::GetTreeViewItem):
19//!   Get tree item metadata (label, icon, description) by handle
20//! - [`OnTreeViewExpansionChanged`](crate::Command::TreeView::OnTreeViewExpansionChanged):
21//!   Notify when user expands/collapses a node (stub - trait method missing)
22//! - [`OnTreeViewSelectionChanged`](crate::Command::TreeView::OnTreeViewSelectionChanged):
23//!   Notify when user selects/deselects tree items (stub - trait method missing)
24//! - [`RefreshTreeView`](crate::Command::TreeView::RefreshTreeView):
25//!   Request tree view to refresh its data, optionally specific items
26//! - [`RevealTreeViewItem`](crate::Command::TreeView::RevealTreeViewItem):
27//!   Request to reveal/focus a specific tree item in the UI
28//! - [`PersistTreeView`](crate::Command::TreeView::PersistTreeView):
29//!   Save tree view state (scroll position, expansion) (stub)
30//! - [`RestoreTreeView`](crate::Command::TreeView::RestoreTreeView):
31//!   Restore previously saved tree view state (stub)
32//!
33//! ERROR HANDLING:
34//! - Returns `Result<Value, String>` with error strings sent to frontend
35//! - Provider errors are logged with context and converted to error strings
36//! - Missing trait methods return structured error indicating not implemented
37//!
38//! PERFORMANCE:
39//! - All commands are async and non-blocking
40//! - Tree data fetching should be efficient; provider may cache results
41//! - Refresh can target specific items to avoid full tree rebuild
42//!
43//! VS CODE REFERENCE:
44//! - `vs/workbench/api/browser/mainThreadTreeViews.ts` - main thread tree view API
45//! - `vs/workbench/api/common/extHostTreeViews.ts` - extension host tree view API
46//! - `vs/workbench/contrib/files/browser/explorerView.ts` - file explorer tree view
47//! - `vs/workbench/contrib/tree/browser/treeView.ts` - generic tree view component
48//!
49//! TODO:

51//! - Add tree view state persistence to ApplicationState
52//! - Implement drag and drop support for tree items
53//! - Add tree item validation and disabled states

63//!
64//! MODULE CONTENTS:
65//! - Tauri command functions (all `#[command] pub async fn`):
66//!   - Data retrieval: `GetTreeViewChildren`, `GetTreeViewItem`
67//!   - UI events: `OnTreeViewExpansionChanged`, `OnTreeViewSelectionChanged`
68//!   - Management: `RefreshTreeView`, `RevealTreeViewItem`

86/// A specific Tauri command handler for the UI to fetch the children of a tree
87/// view node. This handler dispatches to the correct provider (native or
88/// proxied).
89#[command]
90pub async fn GetTreeViewChildren(
91	ApplicationHandle:AppHandle<Wry>,
92
93	_State:State<'_, Arc<ApplicationState>>,
94
95	ViewId:String,
96
97	ElementHandle:Option<String>,
98) -> Result<Value, String> {
99	log::debug!(
100		"[DispatchLogic] Getting TreeView children for '{}', element: {:?}",
101		ViewId,
102		ElementHandle
103	);
104
105	let RunTime = ApplicationHandle.state::<Arc<ApplicationRunTime>>().inner().clone();
106
107	let Environment:Arc<MountainEnvironment> = RunTime.Environment.clone();
108
109	let TreeProvider:Arc<dyn CommonTreeViewProvider> = Environment.Require();
110
111	match TreeProvider.GetChildren(ViewId.clone(), ElementHandle).await {
112		Ok(Children) => Ok(json!(Children)),
113		Err(Error) => {
114			let ErrorMessage = format!("Failed to get children for tree view '{}': {}", ViewId, Error);
115			log::error!("{}", ErrorMessage);
116			Err(ErrorMessage)
117		},
118	}
119}

192/// Refreshes a tree view.
193#[command]
194pub async fn RefreshTreeView(
195	ApplicationHandle:AppHandle<Wry>,
196
197	_State:State<'_, Arc<ApplicationState>>,
198
199	ViewId:String,
200
201	ItemsToRefresh:Option<Vec<String>>,
202) -> Result<Value, String> {
203	log::debug!(
204		"[TreeView Command] Refreshing tree view '{}', items: {:?}",
205		ViewId,
206		ItemsToRefresh
207	);
208
209	let RunTime = ApplicationHandle.state::<Arc<ApplicationRunTime>>().inner().clone();
210
211	let Environment:Arc<MountainEnvironment> = RunTime.Environment.clone();
212
213	let RefreshValue:Option<Value> = ItemsToRefresh.and_then(|items| serde_json::to_value(items).ok());
214
215	match Environment.RefreshTreeView(ViewId.clone(), RefreshValue).await {
216		Ok(_) => Ok(json!({ "success": true })),
217		Err(Error) => {
218			let ErrorMessage = format!("Failed to refresh tree view '{}': {}", ViewId, Error);
219			log::error!("{}", ErrorMessage);
220			Err(ErrorMessage)
221		},
222	}
223}

225/// Reveals a specific tree item.
226#[command]
227pub async fn RevealTreeViewItem(
228	ApplicationHandle:AppHandle<Wry>,
229
230	_State:State<'_, Arc<ApplicationState>>,
231
232	ViewId:String,
233
234	ItemHandle:String,
235
236	Options:Option<Value>,
237) -> Result<Value, String> {
238	log::debug!("[TreeView Command] Revealing item '{}' in view '{}'", ItemHandle, ViewId);
239
240	let RunTime = ApplicationHandle.state::<Arc<ApplicationRunTime>>().inner().clone();
241
242	let Environment:Arc<MountainEnvironment> = RunTime.Environment.clone();
243
244	let OptionsValue = Options.unwrap_or(json!({}));
245
246	match Environment.RevealTreeItem(ViewId.clone(), ItemHandle, OptionsValue).await {
247		Ok(_) => Ok(json!({ "success": true })),
248		Err(Error) => {
249			let ErrorMessage = format!("Failed to reveal tree item in view '{}': {}", ViewId, Error);
250			log::error!("{}", ErrorMessage);
251			Err(ErrorMessage)
252		},
253	}
254}

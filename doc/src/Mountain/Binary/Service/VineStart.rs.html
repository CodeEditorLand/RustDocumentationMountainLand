1//! # Vine Start Module
2//!
3//! Initializes and starts the Vine gRPC server.
4
5use log::{error, info};
6
7/// Starts the Vine gRPC server at the specified addresses.
8///
9/// # Arguments
10///
11/// * `ApplicationHandle` - The Tauri application handle
12/// * `PrimaryAddress` - The primary server address (e.g., "\\[::1\\]:50051")
13/// * `SecondaryAddress` - The secondary server address (e.g.,
14///   "\\[::1\\]:50052")
15///
16/// # Returns
17///
18/// A `Result` indicating success or failure.
19///
20/// # Vine Server Functionality
21///
22/// The Vine gRPC server provides:
23/// - Inter-service communication infrastructure
24/// - gRPC method handling for various services
25/// - Multi-port support for different service types
26///
27/// # Addresses
28///
29/// - Primary: `[::1]:50051` - Main service communication
30/// - Secondary: `[::1]:50052` - Auxiliary service communication
31///
32/// # Errors
33///
34/// Returns an error if Vine server initialization fails.
35pub async fn VineStart(
36	ApplicationHandle:tauri::AppHandle,
37	PrimaryAddress:String,
38	SecondaryAddress:String,
39) -> Result<(), String> {
40	match crate::Vine::Server::Initialize::Initialize(ApplicationHandle, PrimaryAddress, SecondaryAddress) {
41		Ok(()) => {
42			info!("[Vine] [Start] Vine gRPC server started successfully.");
43			Ok(())
44		},
45		Err(e) => {
46			error!("[Vine] [Start] Failed to start: {}", e);
47			Err(format!("Failed to start Vine gRPC server: {}", e))
48		},
49	}
50}

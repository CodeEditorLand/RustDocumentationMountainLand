1//! # TLS Certificate Management Commands
2//!
3//! This module provides Tauri commands for managing TLS certificates from the
4//! webview.
5//!
6//! ## Available Commands
7//!
8//! - `tls_get_ca_cert` - Returns the CA certificate PEM for trust store
9//!   installation
10//! - `tls_get_server_cert_info` - Returns information about a server
11//!   certificate
12//! - `tls_renew_certificate` - Forces renewal of a server certificate
13//! - `tls_get_all_certs` - Lists all cached server certificates
14//! - `tls_initialize` - Initializes the certificate manager and loads/generates
15//!   CA
16//! - `tls_check_cert_status` - Checks if a certificate needs renewal
17//!
18//! ## Usage Example
19//!
20//! ```typescript
21//! // Get CA certificate for installation
22//! const caCert = await invoke('tls_get_ca_cert');
23//! console.log('CA Certificate:', caCert);
24//!
25//! // Get server certificate info
26//! const certInfo = await invoke('tls_get_server_cert_info', {
27//!   hostname: 'code.editor.land'
28//! });
29//! console.log('Valid until:', certInfo.valid_until);
30//!
31//! // Renew a certificate
32//! await invoke('tls_renew_certificate', {
33//!   hostname: 'code.editor.land'
34//! });
35//! ```

44/// Initialize TLS certificate manager
45///
46/// This must be called before any other TLS operations.
47/// It will load an existing CA from the keyring or generate a new one.
48#[tauri::command]
49pub async fn tls_initialize(app_handle:AppHandle) -> Result<String, String> {
50	log::info!("[TlsCommands] Initializing TLS certificate manager");
51
52	// Try to get existing certificate manager from state
53	let state = app_handle
54		.try_state::<Arc<Mutex<CertificateManager>>>()
55		.ok_or("Certificate manager not initialized in app state")?;
56	let cert_manager = state.clone(); // Clone the Arc from State
57
58	let mut manager = cert_manager.lock().map_err(|e| format!("Failed to acquire lock: {}", e))?;
59
60	manager
61		.initialize_ca()
62		.await
63		.map_err(|e| format!("Failed to initialize CA: {}", e))?;
64
65	log::info!("[TlsCommands] TLS certificate manager initialized successfully");
66	Ok("TLS certificate manager initialized".to_string())
67}

69/// Get the CA certificate in PEM format
70///
71/// This can be used to install the CA in the system trust store
72/// or configure the webview to trust it.
73///
74/// Returns the CA certificate PEM string, or an error if not initialized.
75#[tauri::command]
76pub async fn tls_get_ca_cert(app_handle:AppHandle) -> Result<String, String> {
77	log::debug!("[TlsCommands] Getting CA certificate");
78
79	let state = app_handle
80		.try_state::<Arc<Mutex<CertificateManager>>>()
81		.ok_or("Certificate manager not found")?;
82	let cert_manager = state.clone();
83
84	let manager = cert_manager.lock().map_err(|e| format!("Failed to acquire lock: {}", e))?;
85	let cert_pem = manager.get_ca_cert_pem().ok_or("CA certificate not initialized")?;
86
87	Ok(String::from_utf8(cert_pem).map_err(|e| format!("Invalid certificate UTF-8: {}", e))?)
88}

160/// Check if a certificate needs renewal
161///
162/// # Arguments
163///
164/// * `hostname` - The hostname to check
165///
166/// Returns true if the certificate is expiring within 30 days.
167#[tauri::command]
168pub async fn tls_check_cert_status(app_handle:AppHandle, hostname:String) -> Result<CertificateStatus, String> {
169	log::debug!("[TlsCommands] Checking certificate status for {}", hostname);
170
171	let state = app_handle
172		.try_state::<Arc<Mutex<CertificateManager>>>()
173		.ok_or("Certificate manager not found")?;
174	let cert_manager = state.clone();
175
176	let manager = cert_manager.lock().map_err(|e| format!("Failed to acquire lock: {}", e))?;
177
178	// Try to get certificate info
179	if let Some(cert_info) = manager.get_server_cert_info(&hostname) {
180		// Parse the expiry time
181		let valid_until = chrono::DateTime::parse_from_rfc3339(&cert_info.valid_until)
182			.map_err(|e| format!("Invalid certificate expiry time: {}", e))?
183			.with_timezone(&chrono::Utc);
184
185		let now = chrono::Utc::now();
186		let days_until_expiry = (valid_until - now).num_days();
187		let needs_renewal = days_until_expiry <= CertificateManager::RENEWAL_THRESHOLD_DAYS;
188
189		Ok(CertificateStatus {
190			exists:true,
191			is_valid:now <= valid_until,
192			days_until_expiry,
193			needs_renewal,
194			valid_until:cert_info.valid_until,
195		})
196	} else {
197		Ok(CertificateStatus {
198			exists:false,
199			is_valid:false,
200			days_until_expiry:0,
201			needs_renewal:true,
202			valid_until:String::new(),
203		})
204	}
205}

207/// Generate a server certificate for a hostname
208///
209/// This will generate a new certificate if one doesn't exist,
210/// or return an existing valid certificate.
211///
212/// # Arguments
213///
214/// * `hostname` - The hostname (e.g., "code.editor.land")
215///
216/// Returns success message with certificate details.
217#[tauri::command]
218pub async fn tls_generate_cert(app_handle:AppHandle, hostname:String) -> Result<CertificateGenerationResult, String> {
219	log::info!("[TlsCommands] Generating certificate for {}", hostname);
220
221	let state = app_handle
222		.try_state::<Arc<Mutex<CertificateManager>>>()
223		.ok_or("Certificate manager not found")?;
224
225	let cert_manager = state.clone();
226	let manager = cert_manager.lock().map_err(|e| format!("Failed to acquire lock: {}", e))?;
227	let hostname_clone = hostname.clone();
228
229	let _server_config = manager
230		.get_server_cert(&hostname)
231		.await
232		.map_err(|e| format!("Failed to generate certificate: {}", e))?;
233
234	// Get certificate info directly from previous call
235	let cert_info:CertificateInfo = manager
236		.get_server_cert_info(&hostname)
237		.ok_or_else(|| "Certificate not found after generation".to_string())?;
238
239	Ok(CertificateGenerationResult {
240		hostname:hostname_clone,
241		success:true,
242		valid_until:cert_info.valid_until,
243		message:format!("Certificate generated successfully for {}", hostname),
244	})
245}

275/// Certificate status information
276#[derive(Debug, Clone, Serialize, Deserialize)]
277pub struct CertificateStatus {
278	/// Whether the certificate exists
279	pub exists:bool,
280	/// Whether the certificate is currently valid
281	pub is_valid:bool,
282	/// Days until expiry (negative if expired)
283	pub days_until_expiry:i64,
284	/// Whether the certificate needs renewal
285	pub needs_renewal:bool,
286	/// Certificate expiry time (ISO 8601)
287	pub valid_until:String,
288}

295	/// Whether generation was successful

1//! # Window Build Module
2//!
3//! Creates and configures the main application window.
4
5use tauri::{App, WebviewUrl, WebviewWindowBuilder, Wry};
6
7use crate::IPC::WindServiceHandlers::Utilities::RecentlyOpened::ReadRecentlyOpened;
8
9/// Creates and configures the main application window.
10///
11/// # Arguments
12///
13/// * `Application` - The Tauri application instance
14/// * `LocalhostUrl` - The localhost URL for the webview content
15///
16/// # Returns
17///
18/// A configured `WebviewWindow<Wry>` instance.
19///
20/// # Platform-Specific Behavior
21///
22/// - **macOS**: `TitleBarStyle::Overlay` + `hidden_title(true)` keeps the
23///   traffic-light buttons at the top-left but hides the native title bar
24///   strip, so VS Code's custom titlebar (which has `-webkit-app-region:
25///   drag` baked into its CSS) lights up the entire top row as a drag
26///   region. The previous `maximized(true)` was the direct cause of "can't
27///   drag the editor around" - maximized macOS windows are pinned to the
28///   screen and refuse all drag events, regardless of `app-region` CSS.
29/// - **Windows / Linux**: `decorations(false)` keeps the window chrome-less
30///   so the workbench draws its own. We still start `resizable(true)` so
31///   the window can be moved by the drag region.
32/// - **Debug builds**: DevTools auto-open.
33pub fn WindowBuild(Application:&mut App, LocalhostUrl:String) -> tauri::WebviewWindow<Wry> {
34	// Restore the most-recently-opened folder so the webview boots
35	// directly into the workspace. Without this, every launch lands
36	// on the Welcome tab, the user clicks "Open Folder", and the
37	// `pickFolderAndOpen` handler fires `Window.navigate()` - a hard
38	// reload that wipes workbench state mid-initialisation and is
39	// the direct cause of the purple splash flash, stuttering paint,
40	// empty `@builtin` sidebar, and broken keybindings (every layer
41	// has to boot twice and the second pass often loses references
42	// to the first). Using `?folder=...` in the initial URL skips
43	// that destructive round-trip.
44	let InitialUrl = BuildInitialUrl(&LocalhostUrl);
45	let WindowUrl = WebviewUrl::External(
46		InitialUrl
47			.parse()
48			.expect("FATAL: Failed to parse initial webview URL"),
49	);
50
51	// Configure window builder with base settings.
52	//
53	// `visible(false)` is the hidden-until-ready pattern. Tauri's default
54	// is to show the window the instant it's built, which paints the native
55	// chrome + Base.astro's `#1e1e1e` inline background + VS Code theme CSS
56	// + workbench DOM in four separate repaints over the first ~200 ms -
57	// observed as the "purple/dark flash" and panel-pop flicker.
58	//
59	// Mountain shows the window explicitly when the frontend's
60	// `lifecycle:advancePhase(3)` (Restored) arrives, which fires after
61	// `.monaco-workbench` is attached and the first frame is ready. A 3 s
62	// safety timer in `AppLifecycle` guarantees the window appears even if
63	// Sky crashes before signalling phase 3.
64	let mut WindowBuilder = WebviewWindowBuilder::new(Application, "main", WindowUrl)
65		.use_https_scheme(false)
66		.initialization_script("")
67		.zoom_hotkeys_enabled(true)
68		.browser_extensions_enabled(false)
69		.title("Mountain")
70		.resizable(true)
71		.inner_size(1400.0, 900.0)
72		.shadow(true)
73		.visible(false);
74
75	#[cfg(target_os = "macos")]
76	{
77		// Overlay style lets VS Code's custom titlebar paint behind the
78		// traffic-light buttons. `hidden_title(true)` suppresses the OS
79		// title text so it doesn't collide with the workbench menubar.
80		// `decorations(true)` is REQUIRED for the traffic lights to
81		// render - turning decorations off also removes the buttons and
82		// breaks the native drag + resize handles entirely on macOS.
83		WindowBuilder = WindowBuilder
84			.title_bar_style(tauri::TitleBarStyle::Overlay)
85			.hidden_title(true)
86			.decorations(true);
87	}
88
89	#[cfg(any(target_os = "windows", target_os = "linux"))]
90	{
91		// Keep chrome-less on non-macOS - the workbench provides its
92		// own window controls via the WindowsTitleBar / LinuxTitleBar
93		// components. Drag works via `-webkit-app-region: drag` on the
94		// titlebar element.
95		WindowBuilder = WindowBuilder.decorations(false);
96	}
97
98	// Build the main window
99	let MainWindow = WindowBuilder.build().expect("FATAL: Main window build failed");
100
101	// Open DevTools in debug builds
102	#[cfg(debug_assertions)]
103	{
104		MainWindow.open_devtools();
105	}
106
107	MainWindow
108}

110/// Build the initial webview URL, optionally appending `?folder=<path>`
111/// when `~/.land/workspaces/RecentlyOpened.json` has an entry for the
112/// previous session's workspace. Falls back to plain `index.html` if
113/// the file is missing, malformed, or has no resolvable path.
114///
115/// The returned string is already URL-encoded and safe to feed to
116/// `WebviewUrl::External`.
117fn BuildInitialUrl(LocalhostUrl:&str) -> String {
118	let Base = format!("{}/index.html", LocalhostUrl);
119
120	let Recent = match ReadRecentlyOpened() {
121		Ok(Value) => Value,
122		Err(_) => return Base,
123	};
124	let Workspaces = match Recent.get("workspaces").and_then(|V| V.as_array()) {
125		Some(Array) if !Array.is_empty() => Array,
126		_ => return Base,
127	};
128
129	// VS Code's Recently-Opened record can store the folder under a few
130	// different shapes depending on whether the entry came from the
131	// extension host, the workbench, or a `$deltaWorkspaceFolders`
132	// broadcast. Probe them in the same priority order the workbench
133	// itself uses in `getRecentlyOpenedWorkspaces`.
134	let Probe = |Entry:&serde_json::Value| -> Option<String> {
135		// Mountain's own writer emits `{ uri: "file://…", label }` (see
136		// `RecentlyOpened.json` on a freshly closed window). VS Code's
137		// historical `folderUri` / `workspace.configPath` shapes are kept
138		// as fallbacks so imported profiles and third-party writers keep
139		// working.
140		if let Some(Uri) = Entry.get("uri").and_then(|V| V.as_str()) {
141			return Some(Uri.to_string());
142		}
143		if let Some(Uri) = Entry.get("folderUri").and_then(|V| V.as_str()) {
144			return Some(Uri.to_string());
145		}
146		if let Some(Path) = Entry.get("folderUri").and_then(|V| V.get("path")).and_then(|V| V.as_str()) {
147			return Some(Path.to_string());
148		}
149		if let Some(Path) =
150			Entry.get("workspace").and_then(|V| V.get("configPath")).and_then(|V| V.get("path")).and_then(|V| V.as_str())
151		{
152			return Some(Path.to_string());
153		}
154		None
155	};
156
157	let FolderPath = match Workspaces.iter().find_map(Probe) {
158		Some(Path) => Path,
159		None => return Base,
160	};
161
162	// Strip any `file://` scheme so the query param is a plain path
163	// the workbench will stringify into a `file:` URI itself; leaving
164	// the scheme in doubles up and breaks the URL-decode on the other
165	// side (observed as the second `?folder=` boot path appearing as
166	// `file:/Volumes/...` in `wb:boot`).
167	let Normalised = FolderPath
168		.strip_prefix("file://")
169		.unwrap_or(FolderPath.as_str())
170		.to_string();
171	if !std::path::Path::new(&Normalised).is_dir() {
172		return Base;
173	}
174
175	let Encoded = url::form_urlencoded::Serializer::new(String::new())
176		.append_pair("folder", &Normalised)
177		.finish();
178	format!("{}/?{}", LocalhostUrl, Encoded)
179}

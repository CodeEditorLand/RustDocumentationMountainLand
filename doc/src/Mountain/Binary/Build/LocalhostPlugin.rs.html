1//! # Localhost Plugin Module
2//!
3//! Configures and creates the Tauri localhost plugin with CORS headers for
4//! Service Workers and an OTLP proxy for build-baked telemetry.
5
6use std::{
7	io::{Read, Write},
8	net::TcpStream,
9	time::Duration,
10};
11
12use tauri::plugin::TauriPlugin;
13
14/// OTLP collector host:port. OTELBridge.ts sends to `/v1/traces` (same-origin),
15/// this proxy forwards to the real collector via raw TCP. Zero CORS issues.
16const OTLP_HOST:&str = "127.0.0.1:4318";
17
18/// Resolve the correct `Content-Type` for a request URL by its file extension.
19///
20/// The vendored `tauri-plugin-localhost` asset resolver sometimes reports
21/// `text/html` for disk-served `.js` / `.css` assets, which breaks module
22/// loading in the webview (the browser refuses JS with `'text/html' is not a
23/// valid JavaScript MIME type'`). By pre-setting `Content-Type` in
24/// `on_request`, we guarantee the right MIME for the extensions the workbench
25/// actually loads; the patched plugin keeps our value instead of overwriting.
26///
27/// Returns `None` for unknown extensions so the plugin's `asset.mime_type`
28/// fallback still applies (e.g. images, fonts, WASM - the asset resolver
29/// handles those correctly).
30fn MimeFromUrl(Url:&str) -> Option<&'static str> {
31	// Strip query string / fragment before extension match.
32	let Path = Url.split(['?', '#']).next().unwrap_or(Url);
33
34	let Extension = Path.rsplit('.').next()?.to_ascii_lowercase();
35
36	match Extension.as_str() {
37		"js" | "mjs" | "cjs" => Some("application/javascript; charset=utf-8"),
38		"css" => Some("text/css; charset=utf-8"),
39		"json" | "map" => Some("application/json; charset=utf-8"),
40		"html" | "htm" => Some("text/html; charset=utf-8"),
41		"svg" => Some("image/svg+xml"),
42		"wasm" => Some("application/wasm"),
43		"txt" => Some("text/plain; charset=utf-8"),
44		_ => None,
45	}
46}

48/// Forward a JSON body to the OTLP collector via raw HTTP/1.1 POST.
49/// Returns true if the collector accepted (2xx), false otherwise.
50fn ProxyToOTLP(Body:&[u8]) -> bool {
51	let Ok(mut Stream) = TcpStream::connect_timeout(&OTLP_HOST.parse().unwrap(), Duration::from_millis(500)) else {
52		return false;
53	};
54
55	let _ = Stream.set_write_timeout(Some(Duration::from_millis(500)));
56	let _ = Stream.set_read_timeout(Some(Duration::from_millis(500)));
57
58	let Request = format!(
59		"POST /v1/traces HTTP/1.1\r\nHost: {}\r\nContent-Type: application/json\r\nContent-Length: {}\r\nConnection: \
60		 close\r\n\r\n",
61		OTLP_HOST,
62		Body.len(),
63	);
64
65	if Stream.write_all(Request.as_bytes()).is_err() {
66		return false;
67	}
68	if Stream.write_all(Body).is_err() {
69		return false;
70	}
71
72	let mut ResponseBuffer = [0u8; 32];
73	let _ = Stream.read(&mut ResponseBuffer);
74	// Check for "HTTP/1.1 2" - any 2xx status
75	ResponseBuffer.starts_with(b"HTTP/1.1 2") || ResponseBuffer.starts_with(b"HTTP/1.0 2")
76}

78/// Creates and configures the localhost plugin with CORS headers preconfigured.
79///
80/// # CORS Configuration
81///
82/// - Access-Control-Allow-Origin: * (allows all origins)
83/// - Access-Control-Allow-Methods: GET, POST, OPTIONS, HEAD
84/// - Access-Control-Allow-Headers: Content-Type, Authorization, Origin, Accept
85///
86/// # OTLP Proxy
87///
88/// Requests to `/v1/traces` are forwarded to the local OTLP collector
89/// (Jaeger, OTEL Collector, etc.) so OTELBridge.ts can send telemetry
90/// without cross-origin issues. Uses raw TCP - no extra HTTP client dependency.
91pub fn LocalhostPlugin<R:tauri::Runtime>(ServerPort:u16) -> TauriPlugin<R> {
92	tauri_plugin_localhost::Builder::new(ServerPort)
93		.on_request(|Request, Response| {
94			// CORS headers for Service Workers and frontend integration
95			Response.add_header("Access-Control-Allow-Origin", "*");
96			Response.add_header("Access-Control-Allow-Methods", "GET, POST, OPTIONS, HEAD");
97			Response.add_header("Access-Control-Allow-Headers", "Content-Type, Authorization, Origin, Accept");
98
99			let Url = Request.url();
100
101			// OTLP proxy: forward /v1/traces to the local collector
102			if Url.contains("/v1/traces") {
103				Response.set_handled(true);
104				let Body = Request.body();
105				if ProxyToOTLP(Body) {
106					Response.set_status(200);
107					Response.add_header("Content-Type", "application/json");
108				} else {
109					// Collector not running - 204 silently.
110					// OTELBridge stops retrying after first failure.
111					Response.set_status(204);
112				}
113				return;
114			}
115
116			// Pre-set the correct `Content-Type` for known asset extensions.
117			// The patched localhost plugin only falls back to asset.mime_type
118			// when this header is absent, so setting it here makes our value
119			// authoritative. Fixes module-loading errors of the form
120			// `'text/html' is not a valid JavaScript MIME type`.
121			if let Some(Mime) = MimeFromUrl(Url) {
122				Response.add_header("Content-Type", Mime);
123			}
124		})
125		.build()
126}

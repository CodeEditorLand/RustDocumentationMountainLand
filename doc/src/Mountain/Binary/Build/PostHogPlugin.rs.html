1//! # PostHog Plugin Module
2//!
3//! Debug-only PostHog analytics integration for Mountain.
4//! Captures lifecycle events, IPC commands, errors, and performance metrics.
5//! Disabled in release builds (compile-time gated).
6
7use std::sync::OnceLock;
8
9use crate::dev_log;
10
11/// PostHog project token. Source of truth: `.env.Land.PostHog` LAND_POSTHOG_KEY;
12/// `build.rs` bakes the value via `cargo:rustc-env` so `env!` at compile
13/// time always resolves, even on a clean checkout.
14const POSTHOG_API_KEY:&str = env!("LAND_POSTHOG_KEY");
15
16/// PostHog region host (default EU Cloud; operators override via
17/// `.env.Land.PostHog` LAND_POSTHOG_HOST).
18const POSTHOG_HOST:&str = env!("LAND_POSTHOG_HOST");
19
20/// Per-tier enable flag baked from `.env.Land.PostHog`. Cheap early-exit in
21/// every capture path without forking the binary per env value.
22const POSTHOG_ENABLED:&str = env!("LAND_POSTHOG_MOUNTAIN_ENABLED");
23
24/// Optional pinned distinct-id seed (empty string → auto-generate per
25/// process). Useful for CI runs where correlating events across restarts
26/// matters more than per-dev isolation.
27const POSTHOG_DISTINCT_ID_SEED:&str = env!("LAND_POSTHOG_DISTINCT_ID");
28
29/// Global PostHog client instance.
30static CLIENT:OnceLock<posthog_rs::Client> = OnceLock::new();
31
32/// Machine-stable distinct ID for the dev session. When LAND_POSTHOG_DISTINCT_ID
33/// is set, it wins - same value across every process in the same dev run.
34fn DistinctId() -> String {
35	if !POSTHOG_DISTINCT_ID_SEED.is_empty() {
36		return POSTHOG_DISTINCT_ID_SEED.to_string();
37	}
38	let User = std::env::var("USER")
39		.or_else(|_| std::env::var("USERNAME"))
40		.unwrap_or_else(|_| "unknown".to_string());
41	format!("land-dev-{}", User)
42}
43
44/// Whether the Mountain tier should capture at all. Combines compile-time
45/// debug gate with the `.env.Land.PostHog` enable switch.
46fn CaptureAllowed() -> bool {
47	if !cfg!(debug_assertions) {
48		return false;
49	}
50	!matches!(POSTHOG_ENABLED, "false" | "0" | "off")
51}
52
53/// Initialize the PostHog client. Call once during app setup.
54/// No-op in release builds or when LAND_POSTHOG_MOUNTAIN_ENABLED=false.
55pub async fn Initialize() {
56	if !CaptureAllowed() {
57		return;
58	}
59
60	// posthog-rs 0.5 renamed `api_endpoint` → `host` on `ClientOptionsBuilder`
61	// to match the JS/Python SDK vocabulary. Behaviour unchanged: the value is
62	// the base URL the ingestion client POSTs to. Pass `String`-typed so the
63	// `#[builder(setter(into, strip_option))]` attribute on `host` wraps it in
64	// `Some(...)` without another hop.
65	let Options = posthog_rs::ClientOptionsBuilder::default()
66		.api_key(POSTHOG_API_KEY.to_string())
67		.host(POSTHOG_HOST.to_string())
68		.build()
69		.expect("PostHog client options");
70
71	let PostHogClient = posthog_rs::client(Options).await;
72	let _ = CLIENT.set(PostHogClient);
73	dev_log!("lifecycle", "[PostHog] Initialized (host={}, debug mode)", POSTHOG_HOST);
74	CaptureEvent("mountain:session:start", None);
75}

77/// Capture a named event with optional properties.
78pub fn CaptureEvent(EventName:&str, Properties:Option<Vec<(&str, &str)>>) {
79	if !CaptureAllowed() {
80		return;
81	}
82
83	let Some(Client) = CLIENT.get() else { return };
84
85	let mut Event = posthog_rs::Event::new(EventName, &DistinctId());
86
87	let _ = Event.insert_prop("$app", "land-editor");
88	let _ = Event.insert_prop("$app_version", "0.0.1");
89	let _ = Event.insert_prop("$build_mode", "debug");
90	let _ = Event.insert_prop("$component", "mountain");
91
92	if let Some(Props) = Properties {
93		for (Key, Value) in Props {
94			let _ = Event.insert_prop(Key, Value);
95		}
96	}
97
98	let _ = Client.capture(Event);
99}
100
101/// Capture an error event.
102pub fn CaptureError(Tag:&str, Message:&str) {
103	if !CaptureAllowed() {
104		return;
105	}
106
107	CaptureEvent("mountain:error", Some(vec![("error_tag", Tag), ("error_message", Message)]));
108}
109
110/// Capture an IPC command invocation.
111pub fn CaptureIPC(Method:&str) {
112	if !CaptureAllowed() {
113		return;
114	}
115
116	CaptureEvent("mountain:ipc:invoke", Some(vec![("method", Method)]));
117}

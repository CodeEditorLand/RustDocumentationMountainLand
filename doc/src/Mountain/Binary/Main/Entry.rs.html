3//! # Entry (Binary/Main)
4//!
5//! ## RESPONSIBILITIES
6//!
7//! Main application entry point that orchestrates the complete application
8//! lifecycle. This function coordinates:
9//! - Tokio runtime creation and management
10//! - CLI argument parsing
11//! - Application state initialization
12//! - Tauri application builder setup
13//! - Service initialization (Vine, Cocoon, Configuration)
14//! - Graceful shutdown handling
15//!
16//! ## ARCHITECTURAL ROLE
17//!
18//! The Entry module is the **primary entry point** in Mountain's architecture:
19//!
20//! ```text
21//! main.rs ──► Binary::Main::Entry::Fn()
22//!                                    │
23//!                                    ▼
24//! AppLifecycle ──► Service Initialization ──► Tauri App Run
25//!                                           │
26//!                                           ▼
27//!                                   Graceful Shutdown
28//! ```
29//!
30//! ## KEY COMPONENTS
31//!
32//! - **Fn()**: Main entry point exported as `Binary::Main::Fn()`
33//! - Tokio runtime management
34//! - Application state initialization via StateBuild
35//! - Tauri builder configuration via TauriBuild
36//! - Service orchestration (Vine, Cocoon, Configuration)
37//! - Event-driven lifecycle management
38//!
39//! ## ERROR HANDLING
40//!
41//! - Panics on fatal errors (Tokio runtime failure, Tauri build failure)
42//! - Logs errors for service initialization failures
43//! - Graceful degradation for non-critical service failures
44//!
45//! ## LOGGING
46//!
47//! Uses the TraceStep! macro for checkpoint logging at TRACE level.
48//! Additional logging at DEBUG, INFO, WARN, and ERROR levels throughout.
49//!
50//! ## PERFORMANCE CONSIDERATIONS
51//!
52//! - Tokio multi-threaded runtime for optimal performance
53//! - Asynchronous service initialization
54//! - Lazy initialization where possible
55//!
56//! ## TODO
57//! - [ ] Add crash recovery mechanism
58//! - [ ] Implement proper error dialog for startup failures
59//! - [ ] Add startup performance metrics

126/// The main function that orchestrates the application lifecycle.
127///
128/// This function:
129/// 1. Creates a Tokio runtime
130/// 2. Parses CLI arguments
131/// 3. Builds application state
132/// 4. Creates a scheduler
133/// 5. Selects a port for the local server
134/// 6. Resolves the log level
135/// 7. Sets up the Tauri builder
136/// 8. Configures the application lifecycle
137/// 9. Runs the Tauri application
138/// 10. Handles graceful shutdown
139pub fn Fn() {
140	// Open `Mountain.dev.log` up front. Forces `InitFileSink` to create
141	// the session log header on disk before any other code can panic, so
142	// an early crash still leaves a file with a timestamp + pid + tag
143	// context for post-mortem. Env vars are read from the shell here (the
144	// `.env.Land` load below may add MORE keys but never overrides
145	// Trace / Record because `set_var` only runs when a
146	// key is currently unset). Harmless to call: the inner `OnceLock`
147	// gates repeat invocations.
148	crate::IPC::DevLog::InitEager();
149
150	// -------------------------------------------------------------------------
151	// [Boot] [Env] Enhance the process environment with the user's
152	// interactive-shell PATH / NVM_DIR / HOMEBREW_PREFIX / JAVA_HOME / …
153	// before any child process is spawned. macOS GUI launches (Finder,
154	// Dock, Spotlight, `open <bundle>.app`) start the app with a minimal
155	// env where Homebrew, NVM, and similar are not on PATH; without this
156	// step the Cocoon node binary, language servers, and `git` calls
157	// from extensions all fall back to system defaults (or fail).
158	//
159	// No-op when launched from a TTY (terminal already has the right
160	// env) or when the shell probe fails / times out.
161	// -------------------------------------------------------------------------
162	crate::Environment::Utility::EnhanceShellEnvironment::Fn();
163
164	// -------------------------------------------------------------------------
165	// [Boot] [Env] Load .env.Land into process env so standalone binary
166	// invocations pick up Product*, Tier*, Network* vars without requiring
167	// the shell to pre-source the env file. Search order matches
168	// TierEnvironment.sh: cwd .env.Land → parent .env.Land → cwd
169	// .env.Land.Sample → parent .env.Land.Sample → repo-layout probe from
170	// the binary's canonical path (for `target/debug/...` launches where
171	// cwd is arbitrary).
172	// -------------------------------------------------------------------------
173	{
174		fn LoadEnvFile(Path:&std::path::Path) -> bool {
175			let Ok(Content) = std::fs::read_to_string(Path) else {
176				return false;
177			};
178			for Line in Content.lines() {
179				let Trimmed = Line.trim();
180				if Trimmed.is_empty() || Trimmed.starts_with('#') {
181					continue;
182				}
183				if let Some((Key, Value)) = Trimmed.split_once('=') {
184					let CleanKey = Key.trim();
185					let CleanValue = Value.trim().trim_matches('"').trim_matches('\'');
186					if std::env::var_os(CleanKey).is_none() {
187						// SAFETY: set_var is called once per key during bootstrap
188						// before any threads read env (Tokio runtime starts later
189						// in this function).
190						unsafe { std::env::set_var(CleanKey, CleanValue) };
191					}
192				}
193			}
194			true
195		}
196
197		let mut Candidates:Vec<std::path::PathBuf> = Vec::new();
198		if let Ok(Cwd) = std::env::current_dir() {
199			Candidates.push(Cwd.join(".env.Land"));
200			if let Some(Parent) = Cwd.parent() {
201				Candidates.push(Parent.join(".env.Land"));
202			}
203			Candidates.push(Cwd.join(".env.Land.Sample"));
204			if let Some(Parent) = Cwd.parent() {
205				Candidates.push(Parent.join(".env.Land.Sample"));
206			}
207		}
208		// Repo-layout probe: Target/debug/<bin> → four hops up lands at Land/.
209		if let Ok(Exe) = std::env::current_exe() {
210			let Ancestors:Vec<&std::path::Path> = Exe.ancestors().collect();
211			for Candidate in Ancestors.iter().take(6) {
212				Candidates.push(Candidate.join(".env.Land"));
213				Candidates.push(Candidate.join(".env.Land.Sample"));
214			}
215		}
216
217		let mut Loaded = false;
218		for Candidate in Candidates {
219			if Candidate.exists() && LoadEnvFile(&Candidate) {
220				crate::dev_log!("lifecycle", "[Boot] [Env] Loaded env from {}", Candidate.display());
221				Loaded = true;
222				break;
223			}
224		}
225		if !Loaded {
226			crate::dev_log!(
227				"lifecycle",
228				"[Boot] [Env] No .env.Land / .env.Land.Sample found - using defaults"
229			);
230		}
231	}
232
233	// -------------------------------------------------------------------------
234	// [Boot] [Tier] Resolved tier banner (Plan A Wave 1.7 runtime banner)
235	// -------------------------------------------------------------------------
236	crate::LandFixTier::LogResolvedTiers();
237
238	// -------------------------------------------------------------------------
239	// [Boot] [Profile] Self-report (BATCH-13 step 6)
240	//
241	// Build.sh exports `Browser`/`Mountain`/`Electron`/`Bundle`/`Compiler`/
242	// `Profile` into the shell that invokes cargo. `build.rs` captures
243	// those into `cargo:rustc-env=LAND_*` so they're baked into the binary -
244	// runtime env lookups don't survive launching the binary from Finder /
245	// another shell. `option_env!` falls back to "unknown" when the build
246	// ran outside Build.sh (e.g. plain `cargo build`).
247	// -------------------------------------------------------------------------
248	{
249		let NamedProfile = option_env!("Profile").unwrap_or("unknown");
250
251		let Workbench = option_env!("Pack").unwrap_or("Unknown");
252
253		let Bundle = option_env!("Bundle").unwrap_or("");
254
255		let Compiler = option_env!("Compiler").unwrap_or("default");
256
257		dev_log!(
258			"lifecycle",
259			"[LandFix:Profile] Active profile={} workbench={} bundle={} compiler={}",
260			NamedProfile,
261			Workbench,
262			Bundle,
263			Compiler
264		);
265	}
266
267	// -------------------------------------------------------------------------
268	// [Boot] [Runtime] Tokio runtime creation
269	// -------------------------------------------------------------------------
270	TraceStep!("[Boot] [Runtime] Building Tokio runtime...");
271
272	let Runtime = tokio::runtime::Builder::new_multi_thread()
273		.enable_all()
274		.build()
275		.expect("FATAL: Cannot build Tokio runtime.");
276
277	TraceStep!("[Boot] [Runtime] Tokio runtime built.");
278
279	Runtime.block_on(async {
280		// ---------------------------------------------------------------------
281		// [Boot] [PostHog] Initialize telemetry client first so any
282		// error captured during the rest of boot lands in the project.
283		// No-op in release builds or when Report=false.
284		// ---------------------------------------------------------------------
285		crate::Binary::Build::PostHogPlugin::Initialize::Fn().await;
286
287		// ---------------------------------------------------------------------
288		// [Boot] [Args] CLI parsing (using CliParse module)
289		// ---------------------------------------------------------------------
290		let _WorkspaceConfigurationPath = CliParseFn();
291		let _InitialFolders:Vec<String> = vec![];
292
293		// ---------------------------------------------------------------------
294		// [Boot] [State] ApplicationState (using StateBuild module)
295		// ---------------------------------------------------------------------
296		dev_log!("lifecycle", "[Boot] [State] Building ApplicationState...");
297
298		// Create application state directly (StateBuild::Build with default config)
299		let AppState = ApplicationState::default();
300
301		// -------------------------------------------------------------------
302		// [Boot] [Workspace] Seed initial workspace folders so every extension
303		// that calls `vscode.workspace.findFiles(...)` at activation has
304		// something to walk. Precedence: --folder flags → positional dirs →
305		// Open env → CWD fallback. See
306		// CliParse::ParseWorkspaceFolders.
307		// -------------------------------------------------------------------
308		{
309			let InitialFolderPaths = crate::Binary::Initialize::CliParse::ParseWorkspaceFolders();
310			if InitialFolderPaths.is_empty() {
311				dev_log!(
312					"lifecycle",
313					"[Boot] [Workspace] No initial folders resolved - editor will open in \"no folder\" mode."
314				);
315			} else {
316				use crate::ApplicationState::DTO::WorkspaceFolderStateDTO::WorkspaceFolderStateDTO;
317				let mut Folders:Vec<WorkspaceFolderStateDTO> = Vec::new();
318				for (Index, Path) in InitialFolderPaths.iter().enumerate() {
319					let Uri = match url::Url::from_directory_path(Path) {
320						Ok(U) => U,
321						Err(()) => {
322							dev_log!(
323								"lifecycle",
324								"warn: [Boot] [Workspace] Failed to build URL for {}; skipping",
325								Path.display()
326							);
327							continue;
328						},
329					};
330					let Name = Path
331						.file_name()
332						.and_then(|N| N.to_str())
333						.map(str::to_string)
334						.unwrap_or_else(|| Path.display().to_string());
335					match WorkspaceFolderStateDTO::New(Uri, Name, Index) {
336						Ok(Dto) => Folders.push(Dto),
337						Err(Error) => {
338							dev_log!(
339								"lifecycle",
340								"warn: [Boot] [Workspace] Failed to build folder DTO for {}: {}",
341								Path.display(),
342								Error
343							);
344						},
345					}
346				}
347				if !Folders.is_empty() {
348					// Seed state directly; Cocoon is not yet spawned at this
349					// point, so there is no sidecar to notify. The initial
350					// workspace makes it to Cocoon via `InitializeExtensionHost`'s
351					// `workspace` payload during its handshake instead.
352					AppState.Workspace.SetWorkspaceFolders(Folders);
353					dev_log!(
354						"lifecycle",
355						"[Boot] [Workspace] Seeded {} workspace folder(s).",
356						InitialFolderPaths.len()
357					);
358				}
359			}
360		}
361
362		dev_log!(
363			"lifecycle",
364			"[Boot] [State] ApplicationState created with {} workspace folders.",
365			AppState.Workspace.WorkspaceFolders.lock().map(|f| f.len()).unwrap_or(0)
366		);
367
368		// Create Arc for application state to be managed by Tauri
369		let AppStateArcForClosure = Arc::new(AppState.clone());
370
371		// ---------------------------------------------------------------------
372		// [Boot] [Runtime] Scheduler handles (using RuntimeBuild module)
373		// ---------------------------------------------------------------------
374		let Scheduler = Arc::new(SchedulerBuilder::Create().Build());
375		let SchedulerForClosure = Scheduler.clone();
376		TraceStep!("[Boot] [Echo] Scheduler handles prepared.");
377
378		// ---------------------------------------------------------------------
379		// [Boot] [Localhost] Port selection (using PortSelector module)
380		// ---------------------------------------------------------------------
381		let ServerPort = SelectPort();
382		let LocalhostUrl = BuildPortUrl(ServerPort);
383
384		// ---------------------------------------------------------------------
385		// [Boot] [Logging] Log level resolution (using LogLevel module)
386		// ---------------------------------------------------------------------
387		let log_level = ResolveLogLevel();
388
389		// ---------------------------------------------------------------------
390		// [Boot] [Tauri] Builder setup (using TauriBuild module)
391		// ---------------------------------------------------------------------
392		let Builder = TauriBuildFn();
393
394		Builder
395			.plugin(LoggingPluginFn(log_level))
396			.plugin(LocalhostPluginFn(ServerPort))
397			.manage(AppStateArcForClosure.clone())
398			.setup({
399				let LocalhostUrl = LocalhostUrl.clone();
400				let ServerPortForClosure = ServerPort;
401				move |app:&mut App| {
402					dev_log!("lifecycle", "[Lifecycle] [Setup] Setup hook started.");
403					dev_log!("lifecycle", "[Lifecycle] [Setup] LocalhostUrl={}", LocalhostUrl);
404
405					// ---------------------------------------------------------
406					// [Service Registry] Initialize service registry for land:// routing
407					// ---------------------------------------------------------
408					dev_log!(
409						"lifecycle",
410						"[Lifecycle] [Setup] Initializing ServiceRegistry for land:// scheme..."
411					);
412					let service_registry = ServiceRegistryFn::new();
413					init_service_registry(service_registry.clone());
414
415					// ---------------------------------------------------------
416					// [Service Registry] Register local HTTP services
417					// ---------------------------------------------------------
418					// Register the main code editor service
419					dev_log!(
420						"lifecycle",
421						"[Lifecycle] [Setup] Registering code.editor.land service on port {}",
422						ServerPortForClosure
423					);
424					register_land_service("code.editor.land", ServerPortForClosure);
425
426					// Register API editor service (same port for now, can be separate later)
427					register_land_service("api.editor.land", ServerPortForClosure);
428
429					// Register assets editor service (same port for now, can be separate later)
430					register_land_service("assets.editor.land", ServerPortForClosure);
431
432					// Make the registry available as managed state for Tauri commands
433					app.manage(service_registry);
434					dev_log!(
435						"lifecycle",
436						"[Lifecycle] [Setup] ServiceRegistry initialized and services registered."
437					);
438
439					// ---------------------------------------------------------
440					// [DNS Server] Start the Hickory DNS server
441					// ---------------------------------------------------------
442					// The DNS server must start BEFORE any webview loads to ensure
443					// that land:// protocol_resolution is available
444					dev_log!("lifecycle", "[Lifecycle] [Setup] Starting DNS server on preferred port 5380...");
445					let dns_port = Mist::start(5380).unwrap_or_else(|e| {
446						dev_log!(
447							"lifecycle",
448							"warn: [Lifecycle] [Setup] Failed to start DNS server on port 5380: {}",
449							e
450						);
451						// Fallback to random port if preferred port fails
452						Mist::start(0).unwrap_or_else(|e| {
453							dev_log!(
454								"lifecycle",
455								"error: [Lifecycle] [Setup] Completely failed to start DNS server: {}",
456								e
457							);
458							0 // Return 0 as error indicator
459						})
460					});
461
462					if dns_port == 0 {
463						dev_log!(
464							"lifecycle",
465							"warn: [Lifecycle] [Setup] DNS server failed to start, land:// protocol will not be \
466							 available"
467						);
468					} else {
469						dev_log!(
470							"lifecycle",
471							"[Lifecycle] [Setup] DNS server started successfully on port {}",
472							dns_port
473						);
474						// Initialize DNS startup time for tracking
475						crate::Binary::Build::DnsCommands::init_dns_startup_time();
476					}
477
478					// Register DnsPort as managed state for Tauri commands
479					app.manage(DnsPort(dns_port));
480
481					let AppHandle = app.handle().clone();
482					TraceStep!("[Lifecycle] [Setup] AppHandle acquired.");
483
484					// ---------------------------------------------------------
485					// Setup application lifecycle through AppLifecycle module
486					// ---------------------------------------------------------
487					let AppStateArcFromClosure = AppStateArcForClosure.clone();
488
489					if let Err(e) = AppLifecycleSetup(
490						app,
491						AppHandle.clone(),
492						LocalhostUrl.clone(),
493						SchedulerForClosure.clone(),
494						AppStateArcFromClosure,
495					) {
496						dev_log!("lifecycle", "error: [Lifecycle] [Setup] Failed to setup lifecycle: {}", e);
497					}
498
499					Ok(())
500				}
501			})
502			.register_asynchronous_uri_scheme_protocol("land", |_ctx, request, responder| {
503				// Implemented: delegate to synchronous scheme handler
504				let response = crate::Binary::Build::Scheme::land_scheme_handler(&request);
505				responder.respond(response);
506			})
507			.register_asynchronous_uri_scheme_protocol("vscode-file", |ctx, request, responder| {
508				// VS Code Electron workbench uses vscode-file:// to load assets.
509				// Maps to embedded frontend assets from Sky/Target.
510				let AppHandle = ctx.app_handle().clone();
511				std::thread::spawn(move || {
512					let response = crate::Binary::Build::Scheme::VscodeFileSchemeHandler(&AppHandle, &request);
513					responder.respond(response);
514				});
515			})
516			.register_asynchronous_uri_scheme_protocol("vscode-webview", |ctx, request, responder| {
517				// VS Code's `WebviewElement` wraps every extension webview in
518				// an iframe whose `src` is `vscode-webview://<authority>/index.html?...`.
519				// Without this handler the iframe stays blank and every
520				// extension that uses `webviewView` / `WebviewPanel` /
521				// `CustomEditor` (Roo Code, Claude, GitLens, custom editors)
522				// is dead. The handler serves the three-file `pre/`
523				// directory (`index.html`, `service-worker.js`, `fake.html`);
524				// extension HTML itself comes through later via the workbench's
525				// `swMessage` postMessage channel, not this scheme.
526				let AppHandle = ctx.app_handle().clone();
527				std::thread::spawn(move || {
528					let response = crate::Binary::Build::Scheme::VscodeWebviewSchemeHandler(&AppHandle, &request);
529					responder.respond(response);
530				});
531			})
532			.plugin(tauri_plugin_dialog::init())
533			.plugin(tauri_plugin_fs::init())
534			.invoke_handler(tauri::generate_handler![
535				crate::Binary::Tray::SwitchTrayIcon::SwitchTrayIcon,
536				crate::Binary::IPC::WorkbenchConfigurationCommand::MountainGetWorkbenchConfiguration,
537				Command::TreeView::GetTreeViewChildren,
538				Command::LanguageFeature::MountainProvideHover::MountainProvideHover,
539				Command::LanguageFeature::MountainProvideCompletions::MountainProvideCompletions,
540				Command::LanguageFeature::MountainProvideDefinition::MountainProvideDefinition,
541				Command::LanguageFeature::MountainProvideReferences::MountainProvideReferences,
542				Command::SourceControlManagement::GetAllSourceControlManagementState,
543				Command::Keybinding::GetResolvedKeybinding::GetResolvedKeybinding,
544				Track::FrontendCommand::DispatchFrontendCommand::DispatchFrontendCommand,
545				Track::UIRequest::ResolveUIRequest::ResolveUIRequest,
546				Track::Webview::MountainWebviewPostMessageFromGuest::MountainWebviewPostMessageFromGuest,
547				crate::Binary::IPC::MessageReceiveCommand::MountainIPCReceiveMessage,
548				crate::Binary::IPC::StatusGetCommand::MountainIPCGetStatus,
549				crate::Binary::IPC::InvokeCommand::MountainIPCInvoke,
550				crate::Binary::IPC::WindConfigurationCommand::MountainGetWindDesktopConfiguration,
551				crate::Binary::IPC::ConfigurationUpdateCommand::MountainUpdateConfigurationFromWind,
552				crate::Binary::IPC::ConfigurationSyncCommand::MountainSynchronizeConfiguration,
553				crate::Binary::IPC::ConfigurationStatusCommand::MountainGetConfigurationStatus,
554				crate::Binary::IPC::IPCStatusCommand::MountainGetIPCStatus,
555				crate::Binary::IPC::IPCStatusHistoryCommand::MountainGetIPCStatusHistory,
556				crate::Binary::IPC::IPCStatusReportingStartCommand::MountainStartIPCStatusReporting,
557				crate::Binary::IPC::PerformanceStatsCommand::MountainGetPerformanceStats,
558				crate::Binary::IPC::CacheStatsCommand::MountainGetCacheStats,
559				crate::Binary::IPC::CollaborationSessionCommand::MountainCreateCollaborationSession,
560				crate::Binary::IPC::CollaborationSessionCommand::MountainGetCollaborationSessions,
561				crate::Binary::IPC::DocumentSyncCommand::MountainAddDocumentForSync,
562				crate::Binary::IPC::DocumentSyncCommand::MountainGetSyncStatus,
563				crate::Binary::IPC::UpdateSubscriptionCommand::MountainSubscribeToUpdates,
564				crate::Binary::IPC::ConfigurationDataCommand::GetConfigurationData,
565				crate::Binary::IPC::ConfigurationDataCommand::SaveConfigurationData,
566				crate::Binary::IPC::WorkspaceFolderCommand::MountainWorkspaceOpenFolder,
567				crate::Binary::IPC::WorkspaceFolderCommand::MountainWorkspaceListFolders,
568				crate::Binary::IPC::WorkspaceFolderCommand::MountainWorkspaceCloseAllFolders,
569				crate::Binary::Build::DnsCommands::dns_get_server_info,
570				crate::Binary::Build::DnsCommands::dns_get_zone_info,
571				crate::Binary::Build::DnsCommands::dns_get_forward_allowlist,
572				crate::Binary::Build::DnsCommands::dns_get_health_status,
573				crate::Binary::Build::DnsCommands::dns_resolve,
574				crate::Binary::Build::DnsCommands::dns_test_resolution,
575				crate::Binary::Build::DnsCommands::dns_health_check,
576				// Process commands (direct Tauri invoke from ProcessPolyfill)
577				crate::Binary::IPC::ProcessCommand::process_get_exec_path::process_get_exec_path,

579				crate::Binary::IPC::ProcessCommand::process_get_arch::process_get_arch,

581				crate::Binary::IPC::ProcessCommand::process_get_shell_env::process_get_shell_env,
582				crate::Binary::IPC::ProcessCommand::process_get_memory_info::process_get_memory_info,
583				// Health check commands (direct Tauri invoke from SharedProcessProxy)
584				crate::Binary::IPC::HealthCommand::cocoon_extension_host_health::cocoon_extension_host_health,
585				crate::Binary::IPC::HealthCommand::cocoon_search_service_health::cocoon_search_service_health,

587				crate::Binary::IPC::HealthCommand::shared_process_service_health::shared_process_service_health,
588				crate::Binary::IPC::RenderDevLogCommand::RenderDevLog,
589				// LAND-PATCH B7-S6 P14.5: Vine notification broadcast

591				// Tauri Channel that drains the process-wide
592				// `Vine::Client` broadcast into the webview; Effect-TS
593				// `VineNotificationsLive` Layer wraps it as a

597				crate::Binary::IPC::VineSubscribeCommand::vine_subscribe_notifications,

599			])
600			.build(tauri::generate_context!())
601			.expect("FATAL: Error while building Mountain Tauri application")
602			.run(move |app_handle:&tauri::AppHandle, event:tauri::RunEvent| {
603				// Debug-only: log selected lifecycle events
604				if cfg!(debug_assertions) {
605					match &event {
606						RunEvent::MainEventsCleared => {},
607						RunEvent::WindowEvent { .. } => {},
608						_ => dev_log!("lifecycle", "[Lifecycle] [RunEvent] {:?}", event),
609					}
610				}
611
612				if let RunEvent::ExitRequested { api, .. } = event {
613					// Shutdown runs once. The graceful path ends with
614					// `app_handle.exit(0)`, which Tauri re-delivers as a
615					// second `ExitRequested { code: Some(0) }`. On re-entry
616					// we must NOT `prevent_exit` or spawn the shutdown task
617					// again - Cocoon has already been SIGKILLed and the
618					// second pass would log spurious "tcp connect error"
619					// warnings trying to notify a dead sidecar.
620					static SHUTTING_DOWN:AtomicBool = AtomicBool::new(false);
621					if SHUTTING_DOWN.swap(true, Ordering::SeqCst) {
622						return;
623					}
624
625					dev_log!(
626						"lifecycle",
627						"warn: [Lifecycle] [Shutdown] Exit requested. Starting graceful shutdown..."
628					);
629					api.prevent_exit();
630
631					let SchedulerHandle = Scheduler.clone();
632					let app_handle_clone = app_handle.clone();
633
634					tokio::spawn(async move {
635						dev_log!("lifecycle", "[Lifecycle] [Shutdown] Shutting down ApplicationRunTime...");
636						let _ = RuntimeShutdownFn(&app_handle_clone).await;
637
638						dev_log!("lifecycle", "[Lifecycle] [Shutdown] Stopping Echo scheduler...");
639						let _ = SchedulerShutdownFn(SchedulerHandle).await;
640
641						dev_log!("lifecycle", "[Lifecycle] [Shutdown] Done. Exiting process.");
642						app_handle_clone.exit(0);
643					});
644				}
645			});
646
647		dev_log!("lifecycle", "[Lifecycle] [Exit] Mountain application has shut down.");
648	});

3//! Cocoon legacy aliases: `openDocument`, `readFile`, `stat` - short-hand
4//! routes used by Cocoon's Effect-TS Workspace + FileSystem services before
5//! the canonical `FileSystem.*` naming was established. Backed by the same
6//! `FileSystemReader` provider.
7
8use std::{future::Future, pin::Pin, sync::Arc};
9
10use CommonLibrary::{Environment::Requires::Requires, FileSystem::FileSystemReader::FileSystemReader};
11use serde_json::{Value, json};
12use tauri::Runtime;
13
14use crate::{RunTime::ApplicationRunTime::ApplicationRunTime, Track::Effect::MappedEffectType::MappedEffect};
15
16pub fn CreateEffect<R:Runtime>(
17	MethodName:&str,
18	Parameters:Value,
19) -> Option<Result<MappedEffect, String>> {
20	match MethodName {
21		"openDocument" | "readFile" | "stat" => {
22			let MethodNameOwned = MethodName.to_string();
23			let effect =
24				move |run_time:Arc<ApplicationRunTime>| -> Pin<Box<dyn Future<Output = Result<Value, String>> + Send>> {
25					Box::pin(async move {
26						let fs_reader:Arc<dyn FileSystemReader> = run_time.Environment.Require();
27						let Path = if let Some(Object) = Parameters.as_object() {
28							Object
29								.get("uri")
30								.or_else(|| Object.get("path"))
31								.and_then(Value::as_str)
32								.unwrap_or("")
33								.to_string()
34						} else {
35							Parameters.get(0).and_then(Value::as_str).unwrap_or("").to_string()
36						};
37						let PathBuf_ = std::path::PathBuf::from(&Path);
38						match MethodNameOwned.as_str() {
39							"stat" => {
40								fs_reader
41									.StatFile(&PathBuf_)
42									.await
43									.map(|S| serde_json::to_value(S).unwrap_or(Value::Null))
44									.map_err(|e| e.to_string())
45							},
46							"readFile" | "openDocument" => {
47								fs_reader
48									.ReadFile(&PathBuf_)
49									.await
50									.map(|Bytes| {
51										let Text = String::from_utf8(Bytes).unwrap_or_default();
52										json!({ "uri": Path, "text": Text })
53									})
54									.map_err(|e| e.to_string())
55							},
56							_ => Ok(Value::Null),
57						}
58					})
59				};
60			Some(Ok(Box::new(effect)))
61		},
62
63		_ => None,
64	}
65}

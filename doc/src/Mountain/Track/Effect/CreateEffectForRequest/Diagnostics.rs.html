1use std::sync::Arc;
2
3use CommonLibrary::{Diagnostic::DiagnosticManager::DiagnosticManager, Environment::Requires::Requires};
4use serde_json::{Value, json};
5use tauri::Runtime;
6
7use crate::Track::Effect::{
8	CreateEffectForRequest::Utilities::Params::{string_at, val_at},
9	MappedEffectType::MappedEffect,
10};
11
12pub fn CreateEffect<R:Runtime>(MethodName:&str, Parameters:Value) -> Option<Result<MappedEffect, String>> {
13	match MethodName {
14		"Diagnostic.Set" => {
15			crate::effect!(run_time, {
16				let provider:Arc<dyn DiagnosticManager> = run_time.Environment.Require();
17				let owner = string_at(&Parameters, 0);
18				let entries = val_at(&Parameters, 1);
19
20				let Result = provider
21					.SetDiagnostics(owner.clone(), entries.clone())
22					.await
23					.map(|_| json!(null))
24					.map_err(|e| e.to_string());
25
26				// Fan back to Cocoon so peer extensions hooking
27				// `vscode.languages.onDidChangeDiagnostics` observe the
28				// change. The matching subscriber lives at
29				// `Languages/Namespace.ts:1140` on the
30				// `diagnostics.didChange` Emitter channel. Extract the
31				// list of changed URIs from the entries payload
32				// (`entries` is `[[uriString, diagnostics[]], ...]`).
33				let Uris:Vec<Value> = entries
34					.as_array()
35					.map(|Arr| {
36						Arr.iter()
37							.filter_map(|Pair| Pair.as_array().and_then(|P| P.first().cloned()))
38							.collect()
39					})
40					.unwrap_or_default();
41				let _ = crate::Vine::Client::SendNotification::Fn(
42					"cocoon-main".to_string(),
43					"$acceptDiagnosticsChanged".to_string(),
44					json!({ "owner": owner, "uris": Uris }),
45				)
46				.await;
47
48				Result
49			})
50		},
51
52		"Diagnostic.Clear" => {
53			crate::effect!(run_time, {
54				let provider:Arc<dyn DiagnosticManager> = run_time.Environment.Require();
55				let owner = string_at(&Parameters, 0);
56
57				let Result = provider
58					.ClearDiagnostics(owner.clone())
59					.await
60					.map(|_| json!(null))
61					.map_err(|e| e.to_string());
62
63				// Clear translates to "every URI previously held by this
64				// owner now has zero diagnostics". Without knowing the
65				// prior URI set we send an empty `uris` list - the
66				// `onDidChangeDiagnostics` subscriber should re-query
67				// `getDiagnostics(uri)` if it needs the new state.
68				let _ = crate::Vine::Client::SendNotification::Fn(
69					"cocoon-main".to_string(),
70					"$acceptDiagnosticsChanged".to_string(),
71					json!({ "owner": owner, "uris": [] }),
72				)
73				.await;
74
75				Result
76			})
77		},
78
79		_ => None,
80	}
81}

1//! # Languages Effect (CreateEffectForRequest)
2//!
3//! Effect constructor for the `Languages.GetAll` RPC method. Aggregates
4//! language contribution metadata from all scanned extensions (from
5//! `ApplicationState.Extension.ScannedExtensions`) and merges duplicate
6//! language IDs by unioning their aliases, extensions, filenames, filename
7//! patterns, and mimetypes. The result is returned as a flat JSON array
8//! matching the shape expected by Cocoon's `languages.getAll` API.
9//!
10//! ## Aggregation behavior
11//!
12//! - Each extension's `contributes.languages` array is scanned.
13//! - Languages with the same `id` are merged (first-wins for `configuration`).
14//! - Array fields (aliases, extensions, filenames, filenamePatterns, mimetypes)
15//!   are deduplicated across all contributing extensions.
16
17use std::{collections::HashMap, sync::Arc};
18
19use serde_json::{Value, json};
20use tauri::Runtime;
21
22use crate::{Track::Effect::MappedEffectType::MappedEffect, dev_log};
23
24pub fn CreateEffect<R:Runtime>(MethodName:&str, Parameters:Value) -> Option<Result<MappedEffect, String>> {
25	match MethodName {
26		"Languages.GetAll" => {
27			crate::effect!(run_time, {
28				let scanned = run_time
29					.Environment
30					.ApplicationState
31					.Extension
32					.ScannedExtensions
33					.ScannedExtensions
34					.clone();
35
36				let Snapshot = {
37					let Guard = match scanned.lock() {
38						Ok(g) => g,
39						Err(error) => {
40							return Err(format!("Languages.GetAll: scanned-extensions lock poisoned: {}", error));
41						},
42					};
43					Guard.clone()
44				};
45
46				let mut merged:HashMap<String, serde_json::Map<String, Value>> = HashMap::new();
47				for Dto in Snapshot.values() {
48					let Contributes = match Dto.Contributes.as_ref() {
49						Some(c) => c,
50						None => continue,
51					};
52					let Languages = Contributes.get("languages").and_then(Value::as_array);
53					let Some(Languages) = Languages else { continue };
54					for Entry in Languages {
55						let Id = match Entry.get("id").and_then(Value::as_str) {
56							Some(id) if !id.is_empty() => id.to_string(),
57							_ => continue,
58						};
59						let Existing = merged.entry(Id.clone()).or_insert_with(|| {
60							let mut seed = serde_json::Map::new();
61							seed.insert("id".to_string(), json!(Id));
62							seed.insert("aliases".to_string(), json!([]));
63							seed.insert("extensions".to_string(), json!([]));
64							seed.insert("filenames".to_string(), json!([]));
65							seed.insert("filenamePatterns".to_string(), json!([]));
66							seed.insert("mimetypes".to_string(), json!([]));
67							seed.insert("configuration".to_string(), Value::Null);
68							seed
69						});
70						let merge_array = |target:&mut serde_json::Map<String, Value>, key:&str, incoming:&Value| {
71							let Some(incoming_arr) = incoming.get(key).and_then(Value::as_array) else {
72								return;
73							};
74							let bucket = target.entry(key.to_string()).or_insert_with(|| json!([]));
75							if let Some(bucket_arr) = bucket.as_array_mut() {
76								for v in incoming_arr {
77									if !bucket_arr.iter().any(|e| e == v) {
78										bucket_arr.push(v.clone());
79									}
80								}
81							}
82						};
83						merge_array(Existing, "aliases", Entry);
84						merge_array(Existing, "extensions", Entry);
85						merge_array(Existing, "filenames", Entry);
86						merge_array(Existing, "filenamePatterns", Entry);
87						merge_array(Existing, "mimetypes", Entry);
88						if Existing.get("configuration").map(Value::is_null).unwrap_or(true) {
89							if let Some(cfg) = Entry.get("configuration") {
90								Existing.insert("configuration".to_string(), cfg.clone());
91							}
92						}
93					}
94				}
95
96				let result:Vec<Value> = merged.into_values().map(Value::Object).collect();
97				dev_log!("ipc", "[Languages.GetAll] returning {} languages", result.len());
98				Ok(json!(result))
99			})
100		},
101
102		_ => None,
103	}
104}

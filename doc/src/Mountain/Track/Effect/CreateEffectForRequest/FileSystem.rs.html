15/// Strip a leading `file://` (or `file:///`) scheme from the incoming path.
16/// Cocoon sends full URIs like `file:///Users/nikola/.land/extensions/...`
17/// through `FileSystem.ReadFile`/`WriteFile`/`ReadDirectory`; `PathBuf` from
18/// such a string treats the scheme literally and every read 404s. Without
19/// this the redhat.java activation (and any other extension that uses the
20/// gRPC fs.readFile path for its own package.json) fails with "Resource not
21/// found: file:///...".
22fn StripFileUriScheme(Input:&str) -> &str {
23	if let Some(Rest) = Input.strip_prefix("file://") {
24		// `file:///Users/...` - the third slash is part of the path, keep it.
25		if Rest.starts_with('/') {
26			return Rest;
27		}
28		// `file://localhost/Users/...` - rarely used, but normalise by
29		// stripping host-up-to-first-slash. Fall through on failure.
30		if let Some(Idx) = Rest.find('/') {
31			return &Rest[Idx..];
32		}
33	}
34	Input
35}

37pub fn CreateEffect<R:Runtime>(
38	MethodName:&str,
39	Parameters:Value,

41	match MethodName {
42		"FileSystem.ReadFile" => {
43			let effect =
44				move |run_time:Arc<ApplicationRunTime>| -> Pin<Box<dyn Future<Output = Result<Value, String>> + Send>> {
45					Box::pin(async move {
46						let path_str = Parameters.get(0).and_then(Value::as_str).unwrap_or("");
47						if path_str.starts_with("vscode://schemas-associations/") {
48							let payload = serde_json::to_vec(&json!({ "schemas": [] }))
49								.unwrap_or_else(|_| b"{\"schemas\":[]}".to_vec());
50							return Ok(json!(payload));
51						}
52						let fs_reader:Arc<dyn FileSystemReader> = run_time.Environment.Require();
53						let path = std::path::PathBuf::from(StripFileUriScheme(path_str));
54						fs_reader
55							.ReadFile(&path)
56							.await
57							.map(|bytes| json!(bytes))
58							.map_err(|e| e.to_string())
59					})
60				};
61			Some(Ok(Box::new(effect)))
62		},
63
64		"FileSystem.WriteFile" => {
65			let effect =
66				move |run_time:Arc<ApplicationRunTime>| -> Pin<Box<dyn Future<Output = Result<Value, String>> + Send>> {
67					Box::pin(async move {
68						let fs_writer:Arc<dyn FileSystemWriter> = run_time.Environment.Require();
69						let path_str = Parameters.get(0).and_then(Value::as_str).unwrap_or("");
70						let path = std::path::PathBuf::from(StripFileUriScheme(path_str));
71						let content = Parameters.get(1).cloned();
72						let content_bytes = match content {
73							Some(Value::Array(arr)) => {
74								arr.into_iter().filter_map(|v| v.as_u64().map(|n| n as u8)).collect()
75							},
76							Some(Value::String(s)) => STANDARD.decode(&s).unwrap_or_default(),
77							_ => vec![],
78						};
79						fs_writer
80							.WriteFile(&path, content_bytes, true, true)
81							.await
82							.map(|_| json!(null))
83							.map_err(|e| e.to_string())
84					})
85				};
86			Some(Ok(Box::new(effect)))
87		},

3//! Forward a completion request to the registered provider and project
4//! the suggestions into the gRPC `CompletionItem` shape.
5
6use tonic::{Response, Status};
7use url::Url;
8use CommonLibrary::LanguageFeature::{
9	DTO::{
10		CompletionContextDTO::{CompletionContextDTO, CompletionTriggerKindDTO},
11		PositionDTO::PositionDTO,
12	},
13	LanguageFeatureProviderRegistry::LanguageFeatureProviderRegistry,
14};
15
16use crate::{
17	RPC::CocoonService::CocoonServiceImpl,
18	Vine::Generated::{CompletionItem, ProvideCompletionItemsRequest, ProvideCompletionItemsResponse},
19	dev_log,
20};
21
22pub async fn Fn(
23	Service:&CocoonServiceImpl,
24
25	Request:ProvideCompletionItemsRequest,
26) -> Result<Response<ProvideCompletionItemsResponse>, Status> {
27	dev_log!(
28		"cocoon",
29		"[CocoonService] Providing completions for provider {}",
30		Request.provider_handle
31	);
32
33	let URI = Request.uri.as_ref().map(|U| U.value.as_str()).unwrap_or("");
34
35	let DocumentURI = Url::parse(URI).map_err(|E| Status::invalid_argument(format!("Invalid URI: {}", E)))?;
36
37	let Position_ = Request.position.as_ref();
38
39	let PositionDTO_ = PositionDTO {
40		LineNumber:Position_.map(|P| P.line).unwrap_or(0),
41
42		Column:Position_.map(|P| P.character).unwrap_or(0),
43	};
44
45	let ContextDTO = CompletionContextDTO {
46		TriggerKind:CompletionTriggerKindDTO::Invoke,
47
48		TriggerCharacter:if Request.trigger_character.is_empty() {
49			None
50		} else {
51			Some(Request.trigger_character.clone())
52		},
53	};
54
55	match Service
56		.environment
57		.ProvideCompletions(DocumentURI, PositionDTO_, ContextDTO, None)
58		.await
59	{
60		Ok(Some(List)) => {
61			let Items = List
62				.Suggestions
63				.iter()
64				.map(|S| {
65					CompletionItem {
66						label:S.Label.as_str().map(|L| L.to_string()).unwrap_or_default(),
67						kind:format!("{}", S.Kind),
68						detail:S.Detail.clone().unwrap_or_default(),
69						documentation:Vec::new(),
70						insert_text:S.InsertText.as_ref().and_then(|V| V.as_str()).unwrap_or("").to_string(),
71					}
72				})
73				.collect();
74
75			Ok(Response::new(ProvideCompletionItemsResponse { items:Items }))
76		},
77
78		Ok(None) => Ok(Response::new(ProvideCompletionItemsResponse { items:Vec::new() })),
79
80		Err(Error) => Err(Status::internal(format!("Completions failed: {}", Error))),
81	}
82}

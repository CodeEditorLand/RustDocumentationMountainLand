2use tonic::Response;

6pub async fn Fn(RequestId:u64, Params:Value) -> Response<GenericResponse> {
7	let Path = Params
8		.as_str()
9		.or_else(|| Params.get("path").and_then(|V| V.as_str()))
10		.unwrap_or("");
11
12	match tokio::fs::read_dir(Path).await {
13		Ok(mut Entries) => {
14			let mut Items:Vec<Value> = Vec::new();
15
16			while let Ok(Some(Entry)) = Entries.next_entry().await {
17				if let Some(Name) = Entry.file_name().to_str() {
18					let IsDir = Entry.file_type().await.map(|T| T.is_dir()).unwrap_or(false);
19
20					Items.push(json!({ "name": Name, "type": if IsDir { 2u32 } else { 1u32 } }));
21				}
22			}
23
24			super::OkResponse(RequestId, &Items)
25		},
26
27		Err(Error) => super::ErrResponse(RequestId, -32000, format!("fs.listDir: {}", Error)),
28	}
29}

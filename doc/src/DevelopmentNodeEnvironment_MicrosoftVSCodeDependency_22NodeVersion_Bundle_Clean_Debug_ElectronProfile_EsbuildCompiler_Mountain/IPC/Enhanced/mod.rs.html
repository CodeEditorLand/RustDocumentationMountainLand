100	/// Send a message using enhanced features
101	pub async fn send_enhanced_message<T:serde::Serialize>(
102		&self,
103
104		channel:&str,
105
106		message:&T,
107
108		use_compression:bool,
109
110		use_encryption:bool,
111	) -> Result<(), String> {
112		let start_time = std::time::Instant::now();
113
114		// Get connection from pool
115		let connection = self.connection_pool.get_connection().await?;
116
117		// Serialize message
118		let serialized = encode_to_vec(message, bincode::config::standard())
119			.map_err(|e| format!("Failed to serialize message: {}", e))?;
120
121		let result = if use_encryption {
122			// Use secure channel
123			let encrypted = self.secure_channel.encrypt_message(message).await?;
124
125			self.send_encrypted_message(channel, &encrypted).await
126		} else if use_compression {
127			// Use compression
128			self.send_compressed_message(channel, &serialized).await
129		} else {
130			// Send raw message
131			self.send_raw_message(channel, &serialized).await
132		};
133
134		// Record performance metrics
135		let duration = start_time.elapsed().as_millis() as f64;
136
137		let metric = PerformanceDashboard::Dashboard::Struct::create_metric(
138			MetricType::MessageProcessingTime,
139			duration,
140			Some(channel.to_string()),
141			HashMap::new(),
142		);
143
144		self.performance_dashboard.record_metric(metric).await;
145
146		// Release connection
147		self.connection_pool.release_connection(connection).await;
148
149		result
150	}

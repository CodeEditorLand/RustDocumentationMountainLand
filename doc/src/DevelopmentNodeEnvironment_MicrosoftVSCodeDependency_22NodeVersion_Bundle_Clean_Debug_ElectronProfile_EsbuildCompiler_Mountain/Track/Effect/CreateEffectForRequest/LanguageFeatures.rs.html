1//! # LanguageFeatures Effect (CreateEffectForRequest)
2//!
3//! Effect constructors for LSP-like language feature provider registration.
4//! Each method maps to a `ProviderType` variant and delegates to the
5//! `LanguageFeatureProviderRegistry::RegisterProvider` trait method.
6//!
7//! ## Provider types covered
8//!
9//! Hover, Completion, Definition, References, CodeAction, DocumentHighlight,
10//! DocumentSymbol, WorkspaceSymbol, Rename, DocumentFormatting,
11//! DocumentRangeFormatting, OnTypeFormatting, SignatureHelp, CodeLens,
12//! FoldingRange, SelectionRange, SemanticTokens, InlayHint, TypeHierarchy,
13//! CallHierarchy, LinkedEditingRange, DocumentLink, Color, Implementation,
14//! TypeDefinition, Declaration, EvaluatableExpression, InlineValues.
15

33fn CreateProviderEffect(Parameters:Value, ProviderKind:ProviderType) -> Option<Result<MappedEffect, String>> {
34	crate::effect!(run_time, {
35		let provider:Arc<dyn LanguageFeatureProviderRegistry> = run_time.Environment.Require();
36		let id = obj_str(&Parameters, "handle").to_string();
37		let selector = obj_val(&Parameters, "language_selector");
38		let extension_id = obj_val(&Parameters, "extension_id");
39		let options = Parameters.get("options").cloned();
40		provider
41			.RegisterProvider(id, ProviderKind, selector, extension_id, options)
42			.await
43			.map(|handle| json!(handle))
44			.map_err(|e| e.to_string())
45	})
46}
